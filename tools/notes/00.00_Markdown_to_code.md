# Извлечение файлов проекта из Markdown

## Быстрый способ — скрипт на Python

Создай файл `extract.py`:

```python
#!/usr/bin/env python3
"""
Извлекает блоки кода из Markdown файлов в реальные файлы проекта.

Ищет паттерн:
  ## `path/to/file.lua`
  ```lua
  <content>
  ```

Или:
  ## `path/to/file.lua` — описание
  ```lua
  <content>
  ```

Использование:
  python3 extract.py step1.md step2.md ...
  python3 extract.py *.md
  python3 extract.py --all     # все .md в текущей директории
  python3 extract.py --dry-run step1.md   # только показать что будет создано
"""

import re
import sys
import os
from pathlib import Path

# ── Паттерны для определения пути файла ──

# ## `lua/anvil/init.lua`
# ## `lua/anvil/init.lua` — описание
# ### `lua/anvil/init.lua`
HEADER_PATH = re.compile(
    r'^#{1,4}\s+`([^`]+\.[a-zA-Z]+)`'
)

# Альтернативный формат:
# ## lua/anvil/init.lua
HEADER_PATH_BARE = re.compile(
    r'^#{1,4}\s+((?:lua|config|health)/\S+\.\w+|init\.lua|\.env\.\w+|stylua\.toml|selene\.toml|Makefile|\.luarc\.json)'
)

# Блок кода с языком
CODE_BLOCK_START = re.compile(r'^```(\w+)?')
CODE_BLOCK_END = re.compile(r'^```\s*$')


def extract_files(markdown_text: str) -> dict[str, str]:
    """Парсит markdown и возвращает {путь: содержимое}."""
    files = {}
    lines = markdown_text.split('\n')

    current_path = None
    in_code_block = False
    code_lines = []
    pending_path = None

    i = 0
    while i < len(lines):
        line = lines[i]

        # Проверяем заголовок с путём файла
        if not in_code_block:
            match = HEADER_PATH.match(line)
            if not match:
                match = HEADER_PATH_BARE.match(line)

            if match:
                pending_path = match.group(1)
                # Убираем лишние пробелы
                pending_path = pending_path.strip()
                i += 1
                continue

        # Начало блока кода
        if not in_code_block and CODE_BLOCK_START.match(line):
            in_code_block = True
            code_lines = []

            # Если перед блоком кода был заголовок с путём — используем его
            if pending_path:
                current_path = pending_path
                pending_path = None

            i += 1
            continue

        # Конец блока кода
        if in_code_block and CODE_BLOCK_END.match(line):
            in_code_block = False

            if current_path:
                content = '\n'.join(code_lines)
                # Убираем trailing newline но оставляем один
                content = content.rstrip('\n') + '\n'
                files[current_path] = content
                current_path = None

            code_lines = []
            i += 1
            continue

        # Внутри блока кода
        if in_code_block:
            code_lines.append(line)
        else:
            # Сбрасываем pending_path если встретили не-пустую строку
            # которая не является началом блока кода
            if pending_path and line.strip() and not line.startswith('#'):
                # Проверяем — может следующая строка это блок кода
                if i + 1 < len(lines) and CODE_BLOCK_START.match(lines[i + 1]):
                    pass  # Не сбрасываем, пустая строка между заголовком и блоком
                else:
                    pending_path = None

        i += 1

    return files


def write_files(files: dict[str, str], base_dir: str, dry_run: bool = False):
    """Записывает файлы на диск."""
    for path, content in sorted(files.items()):
        full_path = os.path.join(base_dir, path)

        if dry_run:
            lines_count = content.count('\n')
            print(f"  {'[NEW]' if not os.path.exists(full_path) else '[UPD]'} {path} ({lines_count} lines)")
            continue

        # Создаём директории
        os.makedirs(os.path.dirname(full_path), exist_ok=True)

        # Записываем
        with open(full_path, 'w') as f:
            f.write(content)

        print(f"  ✓ {path}")


def main():
    args = sys.argv[1:]

    if not args or '--help' in args or '-h' in args:
        print(__doc__)
        sys.exit(0)

    dry_run = '--dry-run' in args
    args = [a for a in args if a != '--dry-run']

    # Определяем целевую директорию
    base_dir = os.environ.get('ANVIL_DIR', os.path.expanduser('~/.config/anvil'))

    if '--dir' in args:
        idx = args.index('--dir')
        base_dir = args[idx + 1]
        args = args[:idx] + args[idx + 2:]

    # Собираем MD файлы
    if '--all' in args:
        md_files = sorted(Path('.').glob('*.md'))
    else:
        md_files = [Path(a) for a in args]

    if not md_files:
        print("No markdown files found.")
        sys.exit(1)

    all_files = {}

    for md_file in md_files:
        if not md_file.exists():
            print(f"⚠ File not found: {md_file}")
            continue

        print(f"\n📄 Parsing: {md_file}")
        text = md_file.read_text(encoding='utf-8')
        files = extract_files(text)

        if files:
            print(f"   Found {len(files)} file(s)")
            all_files.update(files)
        else:
            print(f"   No extractable files found")

    if not all_files:
        print("\nNo files to extract.")
        sys.exit(1)

    # Фильтруем — только файлы проекта, не случайные блоки кода
    project_files = {}
    for path, content in all_files.items():
        # Должен выглядеть как путь к файлу проекта
        if any([
            path.startswith('lua/'),
            path.startswith('config/'),
            path.startswith('health/'),
            path.startswith('scripts/'),
            path in ('init.lua', 'stylua.toml', 'selene.toml', 'Makefile', '.env.example', '.luarc.json'),
        ]):
            project_files[path] = content

    print(f"\n{'[DRY RUN] ' if dry_run else ''}Writing {len(project_files)} files to: {base_dir}\n")
    write_files(project_files, base_dir, dry_run=dry_run)

    if dry_run:
        print(f"\nDry run complete. Use without --dry-run to write files.")
    else:
        print(f"\n✅ Done! {len(project_files)} files written to {base_dir}")


if __name__ == '__main__':
    main()
```

## Проблема: формат заголовков в наших файлах

Наши Markdown файлы используют разные форматы. Скрипт выше обрабатывает основные, но для **точного** извлечения из конкретных ответов в этом чате лучше использовать усиленную версию:

```python
#!/usr/bin/env python3
"""
extract_anvil.py — извлекает файлы Anvil из Markdown.
Понимает форматы:

  ## `lua/anvil/init.lua`
  ## `lua/anvil/init.lua` — описание
  ## lua/plugins/core.lua — **описание**
  ### `config/servers.json`
"""

import re
import sys
import os
from pathlib import Path

# Все допустимые расширения файлов проекта
EXTENSIONS = r'\.(lua|json|toml|yaml|yml|sh|bash|md|txt|example)'

# Паттерн 1: ## `path/to/file.ext`
P_BACKTICK = re.compile(
    rf'^#{{1,4}}\s+`([^`]+{EXTENSIONS})`'
)

# Паттерн 2: ## path/to/file.ext — описание
P_BARE = re.compile(
    rf'^#{{1,4}}\s+([\w./-]+{EXTENSIONS})\s*(?:—|--|$)'
)

# Паттерн 3: ## path/to/file.ext (без описания, строка заканчивается)
P_BARE_EOL = re.compile(
    rf'^#{{1,4}}\s+([\w./-]+{EXTENSIONS})\s*$'
)

CODE_START = re.compile(r'^```(\w*)')
CODE_END = re.compile(r'^```\s*$')

# Пути которые точно являются файлами проекта
VALID_PREFIXES = (
    'lua/', 'config/', 'health/', 'scripts/', 'doc/', 'tests/',
)
VALID_ROOT_FILES = {
    'init.lua', 'stylua.toml', 'selene.toml', 'Makefile',
    '.env.example', '.luarc.json', '.editorconfig',
}


def is_project_path(path: str) -> bool:
    """Проверяет что путь похож на файл проекта."""
    if any(path.startswith(p) for p in VALID_PREFIXES):
        return True
    if path in VALID_ROOT_FILES:
        return True
    return False


def extract_path(line: str) -> str | None:
    """Пытается извлечь путь файла из строки заголовка."""
    for pattern in (P_BACKTICK, P_BARE, P_BARE_EOL):
        m = pattern.match(line)
        if m:
            path = m.group(1).strip()
            if is_project_path(path):
                return path
    return None


def extract_files(text: str) -> dict[str, str]:
    files = {}
    lines = text.split('\n')

    pending_path = None
    in_code = False
    code_lines = []
    current_path = None

    for i, line in enumerate(lines):
        if not in_code:
            # Ищем заголовок с путём
            path = extract_path(line)
            if path:
                pending_path = path
                continue

            # Начало блока кода
            if CODE_START.match(line):
                in_code = True
                code_lines = []
                if pending_path:
                    current_path = pending_path
                    pending_path = None
                continue

            # Пустые строки не сбрасывают pending_path
            if line.strip() == '':
                continue

            # Непустая строка без блока кода — сбрасываем
            if pending_path and not line.startswith('#'):
                pending_path = None

        else:
            # Внутри блока кода
            if CODE_END.match(line):
                in_code = False
                if current_path and code_lines:
                    content = '\n'.join(code_lines).rstrip('\n') + '\n'
                    files[current_path] = content
                current_path = None
                code_lines = []
            else:
                code_lines.append(line)

    return files


def main():
    args = sys.argv[1:]

    dry_run = '--dry-run' in args
    args = [a for a in args if not a.startswith('--dry')]

    base_dir = os.path.expanduser('~/.config/anvil')
    if '--dir' in args:
        idx = args.index('--dir')
        base_dir = args[idx + 1]
        args = args[:idx] + args[idx + 2:]

    if '--all' in args:
        md_files = sorted(Path('.').glob('*.md'))
        args = []
    else:
        md_files = [Path(a) for a in args if a.endswith('.md')]

    if not md_files:
        print("Usage: python3 extract_anvil.py [--dry-run] [--dir DIR] [--all | file1.md file2.md ...]")
        sys.exit(1)

    all_files = {}
    for md in md_files:
        if not md.exists():
            print(f"⚠ Not found: {md}")
            continue
        text = md.read_text(encoding='utf-8')
        found = extract_files(text)
        print(f"📄 {md.name}: {len(found)} files")
        for p in sorted(found):
            print(f"   {p}")
        all_files.update(found)

    print(f"\n{'[DRY RUN] ' if dry_run else ''}Total: {len(all_files)} files → {base_dir}\n")

    for path in sorted(all_files):
        full = os.path.join(base_dir, path)
        lines_count = all_files[path].count('\n')
        status = 'NEW' if not os.path.exists(full) else 'UPD'

        if dry_run:
            print(f"  [{status}] {path} ({lines_count}L)")
        else:
            os.makedirs(os.path.dirname(full), exist_ok=True)
            with open(full, 'w') as f:
                f.write(all_files[path])
            print(f"  ✓ [{status}] {path}")

    if not dry_run and all_files:
        print(f"\n✅ {len(all_files)} files written")


if __name__ == '__main__':
    main()
```

# Оптимальный формат для извлечения кода из Markdown

## Рекомендуемый формат

Каждый файл оформляется так:

````md
<!-- file: lua/anvil/init.lua -->
```lua
local M = {}
-- содержимое файла
return M
```
````

---

## Почему именно так

| Свойство | Преимущество |
|---|---|
| HTML-комментарий `<!-- file: ... -->` | Не рендерится визуально, не путается с обычными заголовками |
| Фиксированный префикс `file:` | Однозначный парсинг одним regex |
| Путь без кавычек и бэктиков | Нет вариативности, нечего нормализовать |
| Блок кода сразу после комментария | Однозначная связь «путь → содержимое» |
| Язык в блоке кода (`lua`, `json`...) | Подсветка в чате сохраняется |

## Парсер — 30 строк

````python
#!/usr/bin/env python3
"""extract.py — извлекает файлы из Markdown с маркерами <!-- file: path -->"""

import re, sys, os
from pathlib import Path

FILE_MARKER = re.compile(r'<!--\s*file:\s*(\S+)\s*-->')
CODE_FENCE = re.compile(r'^```')

def extract(text: str) -> dict[str, str]:
    files = {}
    lines = text.split('\n')
    i = 0
    while i < len(lines):
        m = FILE_MARKER.search(lines[i])
        if m:
            path = m.group(1)
            # Ищем открывающий ```
            i += 1
            while i < len(lines) and not CODE_FENCE.match(lines[i]):
                i += 1
            i += 1  # пропускаем ```lang
            code = []
            while i < len(lines) and not CODE_FENCE.match(lines[i]):
                code.append(lines[i])
                i += 1
            files[path] = '\n'.join(code).rstrip('\n') + '\n'
        i += 1
    return files

def main():
    dry_run = '--dry-run' in sys.argv
    base = os.path.expanduser(os.environ.get('OUT_DIR', '.'))
    md_files = [f for f in sys.argv[1:] if f.endswith('.md')]

    all_files = {}
    for f in md_files:
        all_files.update(extract(Path(f).read_text()))

    for path in sorted(all_files):
        full = os.path.join(base, path)
        if dry_run:
            print(f"  {'NEW' if not os.path.exists(full) else 'UPD'}  {path} ({all_files[path].count(chr(10))}L)")
        else:
            os.makedirs(os.path.dirname(full), exist_ok=True)
            Path(full).write_text(all_files[path])
            print(f"  ✓ {path}")

    print(f"\n{'[DRY RUN] ' if dry_run else ''}{len(all_files)} files")

if __name__ == '__main__':
    main()
````

```

## Промпт для LLM

````text
When outputting project files, use this exact format for each file:

<!-- file: path/to/file.ext -->
```lang
content
```

This enables automated extraction. Do not use any other format for file output.
Every code block that represents a project file MUST have the <!-- file: --> marker
on the line immediately before the opening fence. Code blocks without the marker
are treated as illustrative examples and will be ignored by the extractor.
````
