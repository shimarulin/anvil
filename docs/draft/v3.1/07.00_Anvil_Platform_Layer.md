# Anvil — Platform Layer

OS-specific settings for Linux, macOS, and WSL.

## Architecture

```text
lua/anvil/platform/
├── init.lua    # platform detection and dispatch
├── linux.lua   # Linux-specific settings
├── macos.lua   # macOS-specific settings
└── wsl.lua     # WSL-specific settings
```

### `lua/anvil/platform/init.lua`

<!-- file: lua/anvil/platform/init.lua -->
```lua
--- Anvil platform detection and OS-specific settings
---@class AnvilPlatform
local M = {}

--- Detect current platform
---@return "linux"|"macos"|"wsl"
function M.detect()
  if vim.fn.has("wsl") == 1 then
    return "wsl"
  elseif vim.fn.has("mac") == 1 then
    return "macos"
  else
    return "linux"
  end
end

--- Apply platform-specific settings
function M.apply()
  local platform = M.detect()
  local ok, mod = pcall(require, "anvil.platform." .. platform)
  if ok and mod.apply then
    mod.apply()
  end

  local logger_ok, logger = pcall(require, "anvil.core.logger")
  if logger_ok then
    logger.debug("Platform: %s", platform)
  end
end

return M
```

### `lua/anvil/platform/linux.lua`

<!-- file: lua/anvil/platform/linux.lua -->
```lua
--- Linux-specific settings
local M = {}

function M.apply()
  -- Clipboard: use xclip or xsel if available
  if vim.fn.executable("xclip") == 1 then
    vim.g.clipboard = {
      name = "xclip",
      copy = {
        ["+"] = "xclip -selection clipboard",
        ["*"] = "xclip -selection primary",
      },
      paste = {
        ["+"] = "xclip -selection clipboard -o",
        ["*"] = "xclip -selection primary -o",
      },
      cache_enabled = true,
    }
  end
end

return M
```

<!-- file: lua/anvil/platform/macos.lua -->
```lua
--- macOS-specific settings
local M = {}

function M.apply()
  -- macOS uses pbcopy/pbpaste by default (Neovim detects this)
  -- Nothing extra needed for clipboard

  -- Fix Python provider path if using Homebrew
  local brew_python = "/opt/homebrew/bin/python3"
  if vim.fn.executable(brew_python) == 1 then
    vim.g.python3_host_prog = brew_python
  end
end

return M
```

### `lua/anvil/platform/wsl.lua`

<!-- file: lua/anvil/platform/wsl.lua -->
```lua
--- WSL-specific settings
local M = {}

function M.apply()
  -- Use win32yank for clipboard in WSL
  if vim.fn.executable("win32yank.exe") == 1 then
    vim.g.clipboard = {
      name = "win32yank-wsl",
      copy = {
        ["+"] = "win32yank.exe -i --crlf",
        ["*"] = "win32yank.exe -i --crlf",
      },
      paste = {
        ["+"] = "win32yank.exe -o --lf",
        ["*"] = "win32yank.exe -o --lf",
      },
      cache_enabled = true,
    }
  end

  -- Reduce updatetime for better WSL performance
  vim.opt.updatetime = 200
end

return M
```
