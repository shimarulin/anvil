# Anvil — AI Ecosystem

The AI layer is a first-class subsystem in Anvil, designed to replace GitHub
Copilot, Cline, and Claude Code with open-source alternatives using any
OpenAI-compatible provider.

## Tool Selection Guide

| Task                          | Tool                  | Why                                |
|-------------------------------|-----------------------|------------------------------------|
| Quick question about code     | CodeCompanion chat    | Fast, integrated, buffer context   |
| Fix diagnostics / errors      | CodeCompanion /fix    | Sees LSP diagnostics directly      |
| Explain error under cursor    | wtf.nvim              | One keypress, instant popup        |
| Code review                   | CodeCompanion /review | Full file + diagnostics            |
| Generate tests                | CodeCompanion /gentest| Understands project test patterns  |
| Inline completion             | minuet-ai (F2 toggle) | Non-intrusive, opt-in             |
| Multi-file refactoring        | Aider                 | Best at editing multiple files     |
| Complex feature with planning | OpenCode              | Built-in planner + sub-agents      |
| Explore unfamiliar codebase   | CodeCompanion + MCP   | MCP reads files, fetches docs      |
| Database queries / schema     | CodeCompanion + MCP   | postgres MCP server                |
| Read external documentation   | CodeCompanion /url    | Fetches and converts to text       |

## Context Gathering

### `lua/anvil/ai/context.lua`

```lua
--- Anvil AI context builder
--- Collects context from editor state for AI prompts
---@class AnvilAIContext
local M = {}

local cache = require("anvil.core.cache")

--- Get diagnostics for a buffer
---@param bufnr? number buffer number (0 = current)
---@return string
function M.get_diagnostics(bufnr)
  bufnr = bufnr or 0
  local diagnostics = vim.diagnostic.get(bufnr)
  if #diagnostics == 0 then return "" end

  local lines = { "## Diagnostics\n" }
  for _, d in ipairs(diagnostics) do
    local severity = vim.diagnostic.severity[d.severity] or "UNKNOWN"
    lines[#lines + 1] = string.format(
      "- [%s] Line %d: %s%s",
      severity, d.lnum + 1, d.message,
      d.source and (" (%s)"):format(d.source) or ""
    )
  end
  return table.concat(lines, "\n")
end

--- Get all project diagnostics (errors only)
---@return string
function M.get_project_errors()
  local diagnostics = vim.diagnostic.get(nil, {
    severity = vim.diagnostic.severity.ERROR,
  })
  if #diagnostics == 0 then return "" end

  local lines = { "## Project Errors\n" }
  for _, d in ipairs(diagnostics) do
    local fname = vim.fn.fnamemodify(
      vim.api.nvim_buf_get_name(d.bufnr), ":~:."
    )
    lines[#lines + 1] = string.format(
      "- %s:%d — %s%s",
      fname, d.lnum + 1, d.message,
      d.source and (" (%s)"):format(d.source) or ""
    )
  end
  return table.concat(lines, "\n")
end

--- Get test results from neotest
---@return string
function M.get_test_results()
  local ok, neotest = pcall(require, "neotest")
  if not ok then return "" end

  local client = neotest.state
  if not client then return "" end

  local adapter_ids = client.adapter_ids()
  if #adapter_ids == 0 then return "" end

  local lines = { "## Test Results\n" }
  local has_failures = false

  for _, adapter_id in ipairs(adapter_ids) do
    local results = client.results(adapter_id)
    if results then
      for test_id, result in pairs(results) do
        if result.status == "failed" then
          has_failures = true
          lines[#lines + 1] = string.format("- FAILED: %s", test_id)
          if result.short then
            lines[#lines + 1] = "  ```\n  " .. result.short .. "\n  ```"
          end
          if result.errors then
            for _, err in ipairs(result.errors) do
              lines[#lines + 1] = string.format(
                "  Error at line %d: %s", err.line or 0, err.message
              )
            end
          end
        end
      end
    end
  end

  if not has_failures then
    lines[#lines + 1] = "All tests passed ✓"
  end

  return table.concat(lines, "\n")
end

--- Get treesitter context (current function/class)
---@return string
function M.get_treesitter_context()
  local node = vim.treesitter.get_node()
  if not node then return "" end

  local function_node = nil
  local class_node = nil
  local current = node

  while current do
    local ntype = current:type()
    if ntype:match("function") or ntype:match("method")
      or ntype == "function_definition" or ntype == "function_declaration"
      or ntype == "method_definition" or ntype == "arrow_function"
    then
      function_node = function_node or current
    elseif ntype:match("class") or ntype == "class_definition"
      or ntype == "class_declaration" or ntype == "impl_item"
      or ntype == "struct_item"
    then
      class_node = class_node or current
    end
    current = current:parent()
  end

  if not function_node and not class_node then return "" end

  local parts = { "## Current Context (Treesitter)\n" }

  if class_node then
    local start_row = class_node:start()
    local text = vim.treesitter.get_node_text(class_node, 0)
    local class_lines = vim.split(text, "\n")
    if #class_lines > 5 then
      text = table.concat(vim.list_slice(class_lines, 1, 5), "\n") .. "\n  ..."
    end
    parts[#parts + 1] = string.format("Class (line %d):\n```\n%s\n```\n", start_row + 1, text)
  end

  if function_node then
    local start_row = function_node:start()
    local text = vim.treesitter.get_node_text(function_node, 0)
    parts[#parts + 1] = string.format("Function (line %d):\n```\n%s\n```\n", start_row + 1, text)
  end

  return table.concat(parts, "\n")
end

--- Get git diff
---@return string
function M.get_git_diff()
  local diff = vim.fn.system("git diff --stat HEAD 2>/dev/null")
  if vim.v.shell_error ~= 0 or diff == "" then return "" end

  local detailed = vim.fn.system("git diff HEAD 2>/dev/null")
  if #detailed > 5000 then
    detailed = detailed:sub(1, 5000) .. "\n... (truncated)"
  end

  return "## Git Changes\n\n```diff\n" .. detailed .. "\n```\n"
end

--- Get project structure and key files (cached)
---@param root? string project root (default: cwd)
---@return string
function M.get_project_index(root)
  root = root or vim.fn.getcwd()
  local cache_key = "project_index:" .. root

  return cache.get_or_set(cache_key, function()
    return M._build_project_index(root)
  end, 3600) -- 1 hour TTL
end

--- Build project index (internal)
---@param root string
---@return string
function M._build_project_index(root)
  local parts = {}

  -- File tree
  local tree_cmd
  if vim.fn.executable("fd") == 1 then
    tree_cmd = string.format(
      "fd --base-directory %s --max-depth 4 --type f --strip-cwd-prefix "
      .. "--exclude .git --exclude node_modules --exclude __pycache__ "
      .. "--exclude .venv --exclude target --exclude dist --exclude build "
      .. "| head -500 | sort",
      vim.fn.shellescape(root)
    )
  else
    tree_cmd = string.format(
      "find %s -maxdepth 4 -type f "
      .. "-not -path '*/.git/*' -not -path '*/node_modules/*' "
      .. "-not -path '*/__pycache__/*' -not -path '*/.venv/*' "
      .. "| head -500 | sed 's|^%s/||' | sort",
      vim.fn.shellescape(root),
      vim.fn.escape(root, "|/")
    )
  end

  local tree = vim.fn.system(tree_cmd)
  parts[#parts + 1] = "## Project Structure\n\n```\n" .. tree .. "```\n"

  -- Key files
  local key_files = {
    "README.md", "README.rst", "package.json", "Cargo.toml", "go.mod",
    "pyproject.toml", "setup.py", "Makefile", "Dockerfile",
    "docker-compose.yml", "tsconfig.json", "biome.json", ".editorconfig",
  }

  for _, file in ipairs(key_files) do
    local path = root .. "/" .. file
    if vim.fn.filereadable(path) == 1 then
      local content = vim.fn.readfile(path)
      if #content > 100 then
        content = vim.list_slice(content, 1, 100)
        content[#content + 1] = "... (truncated)"
      end
      parts[#parts + 1] = string.format(
        "## %s\n\n```\n%s\n```\n", file, table.concat(content, "\n")
      )
    end
  end

  return table.concat(parts, "\n")
end

--- Force re-index project
---@param root? string
function M.index_project(root)
  root = root or vim.fn.getcwd()
  local cache_key = "project_index:" .. root
  cache.del(cache_key)
  M.get_project_index(root)

  local logger = require("anvil.core.logger")
  logger.info("Project indexed: %s", root)
end

--- Build full context from multiple sources
---@param opts? { diagnostics?: boolean, tests?: boolean, treesitter?: boolean, git?: boolean, project?: boolean, project_errors?: boolean }
---@return string
function M.build(opts)
  opts = vim.tbl_extend("force", {
    diagnostics    = true,
    tests          = false,
    treesitter     = true,
    git            = false,
    project        = false,
    project_errors = false,
  }, opts or {})

  local parts = {}

  if opts.project then
    local idx = M.get_project_index()
    if idx ~= "" then parts[#parts + 1] = idx end
  end

  if opts.treesitter then
    local ts = M.get_treesitter_context()
    if ts ~= "" then parts[#parts + 1] = ts end
  end

  if opts.diagnostics then
    local diag = M.get_diagnostics()
    if diag ~= "" then parts[#parts + 1] = diag end
  end

  if opts.project_errors then
    local errs = M.get_project_errors()
    if errs ~= "" then parts[#parts + 1] = errs end
  end

  if opts.tests then
    local tests = M.get_test_results()
    if tests ~= "" then parts[#parts + 1] = tests end
  end

  if opts.git then
    local git = M.get_git_diff()
    if git ~= "" then parts[#parts + 1] = git end
  end

  -- Add language-specific AI context
  local lang_ok, lang = pcall(require, "anvil.lang")
  if lang_ok then
    local ai_ctx = lang.ai_context()
    if ai_ctx and ai_ctx.system_prompt then
      parts[#parts + 1] = "## Language Guidelines\n\n" .. ai_ctx.system_prompt
    end
  end

  return table.concat(parts, "\n\n---\n\n")
end

return M
```

```lua
--- Anvil CodeCompanion configuration
--- Main AI agent: chat, inline, actions, MCP integration
---@class AnvilAIAgents
local M = {}

local providers = require("anvil.ai.providers")

--- Build CodeCompanion adapter from an Anvil provider
---@param name string adapter name
---@param provider AnvilAIProvider
---@return function adapter factory
local function make_adapter(name, provider)
  return function()
    return require("codecompanion.adapters").extend("openai_compatible", {
      name = name,
      env = {
        url     = provider.endpoint,
        api_key = provider.api_key,
      },
      headers = {
        ["Content-Type"]  = "application/json",
        ["Authorization"] = "Bearer " .. provider.api_key,

        ["X-Title"]       = "Anvil",

      schema = {
        model   = { default = provider.model },
        num_ctx = { default = provider.max_tokens },
      },
    })
  end
end

--- Slash commands for chat

      callback = "strategies.chat.slash_commands.file",
      description = "Insert a file",

    ["help"] = {
      callback = "strategies.chat.slash_commands.help",
      description = "Insert content from help tags",
      opts = { contains_code = false, provider = "telescope" },
    },
    ["terminal"] = {
      callback = "strategies.chat.slash_commands.terminal",
      description = "Insert terminal output",
      opts = { contains_code = false },
    },
    ["now"] = {
      callback = "strategies.chat.slash_commands.now",
      description = "Insert current date/time",
      opts = { contains_code = false },
    },

    -- ─── Custom slash commands ───
    ["project"] = {
      callback = function(chat)
        local ctx = require("anvil.ai.context")
        local content = ctx.get_project_index()
        chat:add_message({
          role = "user",
          content = "Here is the project context:\n\n" .. content,
        }, { visible = false })
        vim.notify("Project context added", vim.log.levels.INFO)
      end,
      description = "Add project structure and key files as context",
    },
    ["diagnostics"] = {
      callback = function(chat)
        local ctx = require("anvil.ai.context")
        local content = ctx.get_diagnostics()
        if content == "" then
          vim.notify("No diagnostics found", vim.log.levels.INFO)
          return
        end
        chat:add_message({
          role = "user",
          content = "Current buffer diagnostics:\n\n" .. content,
        }, { visible = false })
        vim.notify("Diagnostics added to context", vim.log.levels.INFO)
      end,
      description = "Add current buffer diagnostics",
    },
    ["tests"] = {
      callback = function(chat)
        local ctx = require("anvil.ai.context")
        local content = ctx.get_test_results()
        if content == "" then
          vim.notify("No test results available", vim.log.levels.INFO)
          return
        end
        chat:add_message({
          role = "user",
          content = "Test results:\n\n" .. content,
        }, { visible = false })
        vim.notify("Test results added to context", vim.log.levels.INFO)
      end,
      description = "Add test results from neotest",
    },
    ["diff"] = {
      callback = function(chat)
        local ctx = require("anvil.ai.context")
        local content = ctx.get_git_diff()
        if content == "" then
          vim.notify("No git changes", vim.log.levels.INFO)
          return
        end
        chat:add_message({
          role = "user",
          content = content,
        }, { visible = false })
        vim.notify("Git diff added to context", vim.log.levels.INFO)
      end,
      description = "Add git diff to context",
    },
    ["url"] = {
      callback = function(chat, args)
        local url = (args and args.content) or ""
        if url == "" then
          url = vim.fn.input("URL: ")
        end
        if url == "" then return end

        vim.notify("Fetching " .. url .. "...", vim.log.levels.INFO)
        local result = vim.fn.system({
          "curl", "-sL", "--max-time", "10",
          "-H", "User-Agent: Mozilla/5.0",
          url,
        })

        -- Convert HTML to text if possible
        if result:match("^%s*<") then
          if vim.fn.executable("pandoc") == 1 then
            local tmp = vim.fn.tempname() .. ".html"
            vim.fn.writefile(vim.split(result, "\n"), tmp)
            result = vim.fn.system("pandoc -f html -t plain --wrap=none " .. tmp)
            vim.fn.delete(tmp)
          elseif vim.fn.executable("lynx") == 1 then
            local tmp = vim.fn.tempname() .. ".html"
            vim.fn.writefile(vim.split(result, "\n"), tmp)
            result = vim.fn.system("lynx -dump -nolist " .. tmp)
            vim.fn.delete(tmp)
          end
        end

        -- Limit size
        if #result > 10000 then
          result = result:sub(1, 10000) .. "\n\n... (truncated)"
        end

        chat:add_message({
          role = "user",
          content = string.format("Content from %s:\n\n%s", url, result),
        }, { visible = true })
      end,
      description = "Fetch and add URL content",
    },
  }
end

return M
```

```lua
--- Anvil AI chat history
--- Persistence and browsing of past conversations
---@class AnvilAIHistory
local M = {}

local history_dir = vim.fn.stdpath("data") .. "/anvil-chat-history"

--- List saved chats
---@return table[] { path: string, name: string, date: string, timestamp: number }
function M.list()
  vim.fn.mkdir(history_dir, "p")
  local files = vim.fn.glob(history_dir .. "/*.md", false, true)

  local chats = {}
  for _, file in ipairs(files) do
    local name = vim.fn.fnamemodify(file, ":t:r")
    local mtime = vim.fn.getftime(file)
    chats[#chats + 1] = {
      path = file, name = name,
      date = os.date("%Y-%m-%d %H:%M", mtime),
      timestamp = mtime,
    }
  end

  table.sort(chats, function(a, b) return a.timestamp > b.timestamp end)
  return chats
end

--- Browse history with telescope or vim.ui.select
function M.pick()
  local chats = M.list()
  if #chats == 0 then
    vim.notify("No chat history found", vim.log.levels.INFO)
    return
  end

  local telescope_ok = pcall(require, "telescope.pickers")
  if telescope_ok then
    M._telescope_pick(chats)
  else
    M._select_pick(chats)
  end
end

--- Telescope picker (internal)
---@param chats table[]
function M._telescope_pick(chats)
  local pickers = require("telescope.pickers")
  local finders = require("telescope.finders")
  local conf = require("telescope.config").values
  local actions = require("telescope.actions")
  local action_state = require("telescope.actions.state")

  pickers.new({}, {
    prompt_title = "Anvil Chat History",
    finder = finders.new_table({
      results = chats,
      entry_maker = function(entry)
        return {
          value = entry,
          display = entry.date .. " — " .. entry.name,
          ordinal = entry.name .. " " .. entry.date,
        }
      end,
    }),
    sorter = conf.generic_sorter({}),
    attach_mappings = function(prompt_bufnr)
      actions.select_default:replace(function()
        actions.close(prompt_bufnr)
        local selection = action_state.get_selected_entry()
        if selection then
          M.load_into_chat(selection.value.path)
        end
      end)
      return true
    end,
  }):find()
end

--- vim.ui.select fallback (internal)
---@param chats table[]
function M._select_pick(chats)
  local items = {}
  for _, chat in ipairs(chats) do
    items[#items + 1] = chat.date .. " — " .. chat.name
  end
  vim.ui.select(items, { prompt = "Load chat history:" }, function(_, idx)
    if idx then M.load_into_chat(chats[idx].path) end
  end)
end

--- Load a chat into CodeCompanion
---@param path string
function M.load_into_chat(path)
  local lines = vim.fn.readfile(path)
  local content = table.concat(lines, "\n")

  vim.cmd("CodeCompanionChat")
  vim.defer_fn(function()
    local cc_ok, cc = pcall(require, "codecompanion")
    if not cc_ok then return end
    local chat = cc.last_chat()
    if chat then
      chat:add_message({
        role = "user",
        content = "Previous conversation context:\n\n" .. content
          .. "\n\n---\nContinue from where we left off.",
      }, { visible = false })
      vim.notify("Chat history loaded", vim.log.levels.INFO)
    end
  end, 500)
end

--- Clean up old history
---@param days? number delete chats older than this (default: 30)
function M.cleanup(days)
  days = days or 30
  local cutoff = os.time() - (days * 86400)
  local chats = M.list()
  local removed = 0

  for _, chat in ipairs(chats) do
    if chat.timestamp < cutoff then
      vim.fn.delete(chat.path)
      removed = removed + 1
    end
  end

  vim.notify(
    string.format("Cleaned up %d chats older than %d days", removed, days),
    vim.log.levels.INFO
  )
end

return M
```

```lua
  })

  vim.bo.buftype = "nofile"
  vim.bo.filetype = "markdown"
  vim.api.nvim_buf_set_name(0, "Anvil AI Context")

end

  end
end

  vim.cmd("MCPHub")
end

end

end
```

```lua
--- Anvil AI health check
--- :checkhealth anvil_ai
local M = {}

function M.check()
  vim.health.start("Anvil AI Ecosystem")

  -- ─── Provider ───
  local prov_ok, providers = pcall(require, "anvil.ai.providers")
  if prov_ok then
    vim.health.ok("Provider config loaded")
    vim.health.info("Main model: " .. providers.main.model)
    vim.health.info("Fast model: " .. providers.fast.model)
    vim.health.info("Endpoint:   " .. providers.main.endpoint)

    if providers.main.api_key == "" then
      vim.health.warn("AI_API_KEY is not set — set it in .env or environment")
    else
      vim.health.ok("API key configured (length: " .. #providers.main.api_key .. ")")
    end
  else
    vim.health.error("Failed to load provider config")
  end

  -- ─── Plugins ───
  vim.health.start("Anvil AI Plugins")
  local plugins = {
    { "codecompanion",    "CodeCompanion" },
    { "minuet",           "minuet-ai" },
    { "mcphub",           "MCPHub" },
    { "wtf",              "wtf.nvim" },
    { "toggleterm",       "ToggleTerm" },
    { "render-markdown",  "render-markdown" },
  }
  for _, p in ipairs(plugins) do
    if pcall(require, p[1]) then
      vim.health.ok(p[2] .. " loaded")
    else
      vim.health.warn(p[2] .. " not found — run :Lazy sync")
    end
  end

  -- ─── External Tools ───
  vim.health.start("Anvil AI External Tools")
  local tools = {
    { "aider",    "Aider (multi-file AI editing)" },
    { "opencode", "OpenCode (planning agent)" },
  }
  for _, t in ipairs(tools) do
    if vim.fn.executable(t[1]) == 1 then
      vim.health.ok(t[2])
    else
      vim.health.info(t[2] .. " not installed (optional)")
    end
  end

  -- ─── Utilities ───
  vim.health.start("Anvil AI Utilities")
  local utils = {
    { "pandoc", "pandoc (HTML→text)" },
    { "curl",   "curl (URL fetching)" },
    { "node",   "Node.js (for MCP servers)" },
    { "npm",    "npm (for MCP servers)" },
  }
  for _, u in ipairs(utils) do
    if vim.fn.executable(u[1]) == 1 then
      vim.health.ok(u[2])
    else
      vim.health.warn(u[2] .. " not found")
    end
  end

  -- ─── MCP Config ───
  vim.health.start("Anvil MCP Servers")
  local mcp_config = vim.fn.expand("~/.config/anvil/mcp-servers.json")
  if vim.fn.filereadable(mcp_config) == 1 then
    vim.health.ok("MCP config found: " .. mcp_config)
  else
    vim.health.warn("MCP config not found — create " .. mcp_config)
  end
end

return M
```
