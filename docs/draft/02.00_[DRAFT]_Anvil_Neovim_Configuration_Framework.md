## Ключевой механизм — Common Bridge

Основная идея: `src/common` не является самостоятельной neovim-конфигурацией. Это **Lua-библиотека**, которую каждый фреймворк подключает через добавление пути в `runtimepath` и затем вызывает нужные модули, адаптируя их к своему формату конфигурации.

### src/common/lua/anvil/init.lua

```lua
-- Anvil Common Configuration Core
-- Это библиотека, а не самостоятельная конфигурация.
-- Подключается через runtimepath из astronvim или nvchad.

local M = {}

M._loaded = false

--- Определяет, в каком фреймворке работаем
---@return "astronvim"|"nvchad"|"standalone"|"unknown"
function M.detect_framework()
  -- AstroNvim определяется по наличию модуля astrocore
  local has_astro = pcall(require, "astrocore")
  if has_astro then return "astronvim" end

  -- NvChad определяется по наличию модуля nvconfig или chadrc
  local has_nvchad = pcall(require, "nvconfig")
  if has_nvchad then return "nvchad" end

  -- Проверяем глобальную переменную, которую может выставить standalone
  if vim.g.anvil_framework then
    return vim.g.anvil_framework
  end

  return "unknown"
end

--- Основная точка входа. Вызывается из bridge-модуля конкретного фреймворка.
---@param opts? table
function M.setup(opts)
  if M._loaded then return end
  M._loaded = true

  opts = opts or {}

  -- Применяем общие vim options
  require("anvil.options").setup(opts.options or {})

  -- Применяем общие автокоманды
  require("anvil.autocmds").setup(opts.autocmds or {})

  -- Маппинги применяются по-разному в зависимости от фреймворка,
  -- поэтому возвращаем таблицу, а bridge сам решает как применить
  -- (но можно и напрямую через vim.keymap.set)
  if opts.keymaps ~= false then
    require("anvil.keymaps").setup(opts.keymaps or {})
  end
end

--- Возвращает общие спеки плагинов в формате lazy.nvim
--- Оба фреймворка используют lazy.nvim, поэтому формат единый
---@param opts? table Переопределения и фильтры
---@return table[] Массив lazy.nvim plugin specs
function M.plugin_specs(opts)
  opts = opts or {}
  local specs = {}

  local spec_modules = {
    "anvil.plugins.editor",
    "anvil.plugins.ui",
    "anvil.plugins.git",
    "anvil.plugins.lang",
  }

  -- Позволяем отключить определённые группы
  local disabled = opts.disabled_groups or {}
  local disabled_set = {}
  for _, g in ipairs(disabled) do
    disabled_set[g] = true
  end

  for _, mod_name in ipairs(spec_modules) do
    local group = mod_name:match("anvil%.plugins%.(.+)")
    if not disabled_set[group] then
      local ok, mod = pcall(require, mod_name)
      if ok then
        if type(mod) == "table" then
          -- Модуль может вернуть один спек или массив спеков
          if mod[1] or mod.dir or mod.url then
            -- Одиночный спек (имеет [1] как short url или dir/url)
            table.insert(specs, mod)
          else
            -- Таблица спеков (ключи — числовые)
            for _, spec in ipairs(mod) do
              table.insert(specs, spec)
            end
          end
        end
      end
    end
  end

  return specs
end

--- Возвращает общие маппинги в нормализованном формате
--- { mode = "n", lhs = "<leader>xx", rhs = ..., opts = { desc = "..." } }
---@return table[]
function M.get_keymaps()
  return require("anvil.keymaps").get_all()
end

--- Возвращает общие опции для LSP
---@return table
function M.lsp_opts()
  return require("anvil.plugins.lang").lsp_opts()
end

return M
```

local M = {}

-- Общие опции, не зависящие от фреймворка
local default_options = {
  -- Визуальные
  number = true,
  relativenumber = true,
  signcolumn = "yes",
  cursorline = true,
  termguicolors = true,
  showmode = false,
  wrap = false,
  linebreak = true,

  -- Отступы
  expandtab = true,
  shiftwidth = 2,
  tabstop = 2,
  softtabstop = 2,
  smartindent = true,

  -- Поиск
  ignorecase = true,
  smartcase = true,
  hlsearch = true,
  incsearch = true,

  -- Поведение
  splitbelow = true,
  splitright = true,
  scrolloff = 8,
  sidescrolloff = 8,
  mouse = "a",
  clipboard = "unnamedplus",
  undofile = true,
  updatetime = 250,
  timeoutlen = 300,

  -- Completion
  completeopt = "menu,menuone,noselect",
}

---@param overrides? table
function M.setup(overrides)
  overrides = overrides or {}

  local opts = vim.tbl_deep_extend("force", default_options, overrides)

  for k, v in pairs(opts) do
    -- Пропускаем нестандартные ключи
    if type(k) == "string" then
      local ok = pcall(function() vim.opt[k] = v end)
      if not ok then
        vim.notify(
          string.format("[anvil] Failed to set option: %s = %s", k, vim.inspect(v)),
          vim.log.levels.WARN
        )
      end
    end
  end
end

return M

### src/common/lua/anvil/keymaps.lua

```lua
local M = {}

-- Формат: { mode, lhs, rhs, opts }
-- Используем <Leader> через описание — конкретный leader задаётся фреймворком
local keymaps = {
  -- ── Навигация между окнами ──
  { "n", "<C-h>", "<C-w>h", { desc = "Window left" } },
  { "n", "<C-j>", "<C-w>j", { desc = "Window down" } },
  { "n", "<C-k>", "<C-w>k", { desc = "Window up" } },
  { "n", "<C-l>", "<C-w>l", { desc = "Window right" } },

  -- ── Resize окон ──
  { "n", "<C-Up>", "<cmd>resize +2<cr>", { desc = "Increase window height" } },
  { "n", "<C-Down>", "<cmd>resize -2<cr>", { desc = "Decrease window height" } },
  { "n", "<C-Left>", "<cmd>vertical resize -2<cr>", { desc = "Decrease window width" } },
  { "n", "<C-Right>", "<cmd>vertical resize +2<cr>", { desc = "Increase window width" } },

  -- ── Перемещение строк ──
  { "n", "<A-j>", "<cmd>m .+1<cr>==", { desc = "Move line down" } },
  { "n", "<A-k>", "<cmd>m .-2<cr>==", { desc = "Move line up" } },
  { "v", "<A-j>", ":m '>+1<cr>gv=gv", { desc = "Move selection down" } },
  { "v", "<A-k>", ":m '<-2<cr>gv=gv", { desc = "Move selection up" } },

  -- ── Буферы ──
  { "n", "<S-h>", "<cmd>bprevious<cr>", { desc = "Previous buffer" } },
  { "n", "<S-l>", "<cmd>bnext<cr>", { desc = "Next buffer" } },

  -- ── Общие удобства ──
  { "n", "<Esc>", "<cmd>nohlsearch<cr>", { desc = "Clear search highlights" } },
  { "i", "jk", "<Esc>", { desc = "Exit insert mode" } },
  { "v", "<", "<gv", { desc = "Indent left and reselect" } },
  { "v", ">", ">gv", { desc = "Indent right and reselect" } },

  -- ── Диагностика (базовые, фреймворк может переопределить) ──
  { "n", "]d", function() vim.diagnostic.goto_next() end, { desc = "Next diagnostic" } },
  { "n", "[d", function() vim.diagnostic.goto_prev() end, { desc = "Previous diagnostic" } },
}

---@param overrides? table
function M.setup(overrides)
  overrides = overrides or {}

  for _, map in ipairs(keymaps) do
    local mode, lhs, rhs, opts = map[1], map[2], map[3], map[4] or {}
    opts = vim.tbl_extend("force", { silent = true, noremap = true }, opts)
    vim.keymap.set(mode, lhs, rhs, opts)
  end

  -- Применяем пользовательские переопределения
  for _, map in ipairs(overrides) do
    local mode, lhs, rhs, opts = map[1], map[2], map[3], map[4] or {}
    opts = vim.tbl_extend("force", { silent = true, noremap = true }, opts)
    vim.keymap.set(mode, lhs, rhs, opts)
  end
end

--- Возвращает все маппинги в структурированном виде
--- (для фреймворков, которые хотят применять маппинги через свой API)
---@return table[]
function M.get_all()
  return keymaps
end

--- Конвертирует маппинги в формат AstroCore mappings
---@return table Формат { n = { ["<lhs>"] = { rhs, desc = "..." } } }
function M.to_astrocore_format()
  local result = {}
  for _, map in ipairs(keymaps) do
    local mode, lhs, rhs, opts = map[1], map[2], map[3], map[4] or {}
    if not result[mode] then result[mode] = {} end
    result[mode][lhs] = {
      rhs,
      desc = opts.desc or "",
    }
  end
  return result
end

--- Конвертирует маппинги в формат NvChad mappings
---@return table Формат, совместимый с nvchad mappings
function M.to_nvchad_format()
  local result = {}
  for _, map in ipairs(keymaps) do
    local mode, lhs, rhs, opts = map[1], map[2], map[3], map[4] or {}
    if not result[mode] then result[mode] = {} end
    result[mode][lhs] = { rhs, opts.desc or "" }
  end
  return result
end

return M
```

### src/common/lua/anvil/autocmds.lua

```lua
local M = {}

---@param overrides? table
function M.setup(overrides)
  overrides = overrides or {}

  local augroup = vim.api.nvim_create_augroup("AnvilCommon", { clear = true })

  -- Подсветка при yank
  vim.api.nvim_create_autocmd("TextYankPost", {
    group = augroup,
    desc = "Highlight on yank",
    callback = function()
      vim.highlight.on_yank({ higroup = "IncSearch", timeout = 200 })
    end,
  })

  -- Убираем trailing whitespace при сохранении
  vim.api.nvim_create_autocmd("BufWritePre", {
    group = augroup,
    desc = "Remove trailing whitespace",
    pattern = "*",
    callback = function()
      if vim.bo.filetype == "diff" then return end
      local save = vim.fn.winsaveview()
      vim.cmd([[%s/\s\+$//e]])
      vim.fn.winrestview(save)
    end,
  })

  -- Автоматически возвращаемся к последней позиции в файле
  vim.api.nvim_create_autocmd("BufReadPost", {
    group = augroup,
    desc = "Restore cursor position",
    callback = function(args)
      local mark = vim.api.nvim_buf_get_mark(args.buf, '"')
      local line_count = vim.api.nvim_buf_line_count(args.buf)
      if mark[1] > 0 and mark[1] <= line_count then
        pcall(vim.api.nvim_win_set_cursor, 0, mark)
      end
    end,
  })

  -- Автоматический resize при изменении размера терминала
  vim.api.nvim_create_autocmd("VimResized", {
    group = augroup,
    desc = "Auto resize panes",
    callback = function()
      vim.cmd("tabdo wincmd =")
    end,
  })

  -- Пользовательские автокоманды
  for _, ac in ipairs(overrides) do
    vim.api.nvim_create_autocmd(ac.event, vim.tbl_extend("force", {
      group = augroup,
    }, ac.opts or {}))
  end
end

return M
```

### src/common/lua/anvil/utils.lua

```lua
local M = {}

--- Проверяет, установлен ли исполняемый файл
---@param name string
---@return boolean
function M.is_executable(name)
  return vim.fn.executable(name) == 1
end

--- Безопасный require с fallback
---@param mod string
---@return any|nil
function M.safe_require(mod)
  local ok, result = pcall(require, mod)
  if ok then return result end
  return nil
end

--- Путь к директории common (определяется автоматически)
---@return string
function M.common_path()
  -- Ищем путь к текущему файлу и поднимаемся до common/
  local source = debug.getinfo(1, "S").source:sub(2) -- убираем @
  -- source = .../src/common/lua/anvil/utils.lua
  local common_lua = vim.fn.fnamemodify(source, ":h:h") -- .../src/common/lua
  local common = vim.fn.fnamemodify(common_lua, ":h")    -- .../src/common
  return common
end

--- Добавляет common в runtimepath если ещё не добавлен
function M.ensure_rtp()
  local common = M.common_path()
  if not vim.tbl_contains(vim.opt.runtimepath:get(), common) then
    vim.opt.runtimepath:prepend(common)
  end
end

--- Мержит несколько таблиц lazy.nvim спеков
---@param ... table[] Массивы спеков
---@return table[] Объединённый массив
function M.merge_specs(...)
  local result = {}
  for _, specs in ipairs({ ... }) do
    for _, spec in ipairs(specs) do
      table.insert(result, spec)
    end
  end
  return result
end

return M
```

-- Мост между AstroNvim и anvil common
-- Этот файл подключает common runtimepath и загружает общую конфигурацию

-- Определяем путь к common относительно astronvim конфигурации
-- Структура: src/astronvim/lua/plugins/anvil_bridge.lua

local this_file = debug.getinfo(1, "S").source:sub(2)
local plugins_dir = vim.fn.fnamemodify(this_file, ":h")     -- .../lua/plugins

local common_dir = src_dir .. "/common"

vim.opt.runtimepath:prepend(common_dir)

-- Загружаем anvil (без применения keymaps — их мы передадим через astrocore)

  keymaps = false, -- Маппинги применяем через AstroCore, не напрямую
})

-- Возвращаем спеки как lazy.nvim плагины
-- Каждый спек из common будет загружен как обычный lazy.nvim плагин

    keymaps = true,
  })
end

### install.sh

```bash
#!/usr/bin/env bash
set -euo pipefail

# ─── Цвета ───
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
CYAN='\033[0;36m'
NC='\033[0m' # No Color

# ─── Определяем директорию скрипта (корень репозитория) ───
SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
SRC_DIR="${SCRIPT_DIR}/src"

# ─── Платформо-зависимые пути ───
case "$(uname -s)" in
  Linux|Darwin)
    NVIM_CONFIG_HOME="${XDG_CONFIG_HOME:-$HOME/.config}"
    NVIM_DATA_HOME="${XDG_DATA_HOME:-$HOME/.local/share}"
    NVIM_STATE_HOME="${XDG_STATE_HOME:-$HOME/.local/state}"
    NVIM_CACHE_HOME="${XDG_CACHE_HOME:-$HOME/.cache}"
    ;;
  MINGW*|MSYS*|CYGWIN*)
    NVIM_CONFIG_HOME="${LOCALAPPDATA}"
    NVIM_DATA_HOME="${LOCALAPPDATA}"
    NVIM_STATE_HOME="${LOCALAPPDATA}"
    NVIM_CACHE_HOME="${TEMP:-/tmp}"
    ;;
  *)
    echo -e "${RED}Unsupported platform$(uname -s)${NC}"
    exit 1
    ;;
esac

# ─── Функции ───

print_header() {
  echo -e "${CYAN}"
  echo "  ╔═══════════════════════════════════════╗"
  echo "  ║           ⚒  ANVIL  ⚒                ║"
  echo "  ║   Unified Neovim Configuration        ║"
  echo "  ╚═══════════════════════════════════════╝"
  echo -e "${NC}"
}

log_info() { echo -e "${BLUE}[INFO]${NC} $*"; }
log_ok()   { echo -e "${GREEN}[OK]${NC} $*"; }
log_warn() { echo -e "${YELLOW}[WARN]${NC} $*"; }
log_err()  { echo -e "${RED}[ERROR]${NC} $*"; }

# Проверяет зависимости
check_deps() {
  local missing=()

  command -v git    &>/dev/null || missing+=("git")
  command -v nvim   &>/dev/null || missing+=("nvim")

  if [[ ${#missing[@]} -gt 0 ]]; then
    log_err "Missing required dependencies: ${missing[*]}"
    exit 1
  fi

  # Проверяем версию neovim
  local nvim_version
  nvim_version=$(nvim --version | head -1 | grep -oP 'v\K[0-9]+\.[0-9]+')
  local major minor
  major=$(echo "$nvim_version" | cut -d. -f1)
  minor=$(echo "$nvim_version" | cut -d. -f2)

  if [[ $major -lt 1 && $minor -lt 10 ]]; then
    log_err "Neovim v0.10+ required (found v${nvim_version})"
    exit 1
  fi

  log_ok "Dependencies OK (nvim v${nvim_version})"
}

# Создаёт симлинк с проверками
create_symlink() {
  local source="$1"
  local target="$2"

  if [[ -L "$target" ]]; then
    local current_target
    current_target=$(readlink -f "$target")
    local resolved_source
    resolved_source=$(readlink -f "$source")

    if [[ "$current_target" == "$resolved_source" ]]; then
      log_ok "Symlink already correct: $target -> $source"
      return 0
    else
      log_warn "Symlink exists but points elsewhere: $target -> $current_target"
      log_info "Updating to: $target -> $source"
      rm "$target"
    fi
  elif [[ -e "$target" ]]; then
    local backup="${target}.bak.$(date +%Y%m%d%H%M%S)"
    log_warn "Target exists and is not a symlink: $target"
    log_info "Backing up to: $backup"
    mv "$target" "$backup"
  fi

  # Создаём родительскую директорию если нужно
  mkdir -p "$(dirname "$target")"

  ln -s "$source" "$target"
  log_ok "Created symlink: $target -> $source"
}

# Очищает данные для конкретного NVIM_APPNAME
clean_data() {
  local appname="$1"

  local dirs=(
    "${NVIM_DATA_HOME}/${appname}"
    "${NVIM_STATE_HOME}/${appname}"
    "${NVIM_CACHE_HOME}/${appname}"
  )

  for dir in "${dirs[@]}"; do
    if [[ -d "$dir" ]]; then
      log_info "Removing: $dir"
      rm -rf "$dir"
    fi
  done
}

# ─── Команда: install ───
# Устанавливает конфигурацию как изолированный NVIM_APPNAME
cmd_install() {
  local framework="$1"
  local appname="${2:-anvil-${framework}}"

  if [[ ! -d "${SRC_DIR}/${framework}" ]]; then
    log_err "Framework '${framework}' not found in ${SRC_DIR}/"
    log_info "Available: $(ls -1 "${SRC_DIR}" | grep -v common | tr '\n' ' ')"
    exit 1
  fi

  log_info "Installing '${framework}' as NVIM_APPNAME='${appname}'"

  local target="${NVIM_CONFIG_HOME}/${appname}"
  create_symlink "${SRC_DIR}/${framework}" "$target"

  # Также нужен доступ к common — он подключается через runtimepath
  # в bridge-файлах, используя относительные пути от framework dir.
  # Поскольку мы делаем симлинк на директорию framework, а common
  # лежит рядом в src/, относительные пути в bridge будут работать
  # только если симлинк разрешается. Проверяем:
  if [[ ! -d "${SRC_DIR}/common" ]]; then
    log_warn "Common directory not found at ${SRC_DIR}/common"
    log_warn "Shared configuration will not be available"
  else
    log_ok "Common config available at ${SRC_DIR}/common"
  fi

  log_ok "Installation complete!"
  echo ""
  echo -e "  Run with: ${GREEN}NVIM_APPNAME=${appname} nvim${NC}"
  echo ""
  echo -e "  Or add alias to your shell config:"
  echo -e "  ${CYAN}alias ${appname}='NVIM_APPNAME=${appname} nvim'${NC}"
}

# ─── Команда: install-default ───
# Устанавливает как основную конфигурацию nvim
cmd_install_default() {
  local framework="$1"

  if [[ ! -d "${SRC_DIR}/${framework}" ]]; then
    log_err "Framework '${framework}' not found"
    exit 1
  fi

  log_info "Installing '${framework}' as default nvim config"

  local target="${NVIM_CONFIG_HOME}/nvim"
  create_symlink "${SRC_DIR}/${framework}" "$target"

  log_ok "Installed as default nvim configuration"
  echo ""
  echo -e "  Run with: ${GREEN}nvim${NC}"
}

# ─── Команда: install-all ───
# Устанавливает все фреймворки как изолированные конфигурации
cmd_install_all() {
  log_info "Installing all frameworks as isolated configurations"

  for dir in "${SRC_DIR}"/*/; do
    local name
    name=$(basename "$dir")
    [[ "$name" == "common" ]] && continue
    [[ "$name" == "standalone" ]] && continue

    cmd_install "$name" "anvil-${name}"
    echo ""
  done

  log_ok "All frameworks installed!"
  echo ""
  echo "  Available configurations:"
  for dir in "${SRC_DIR}"/*/; do
    local name
    name=$(basename "$dir")
    [[ "$name" == "common" ]] && continue
    [[ "$name" == "standalone" ]] && continue
    echo -e "    ${GREEN}NVIM_APPNAME=anvil-${name} nvim${NC}"
  done
}

# ─── Команда: uninstall ───
cmd_uninstall() {
  local appname="$1"
  local target="${NVIM_CONFIG_HOME}/${appname}"

  if [[ -L "$target" ]]; then
    rm "$target"
    log_ok "Removed symlink: $target"
  elif [[ -e "$target" ]]; then
    log_warn "$target is not a symlink, refusing to remove"
    log_info "Remove manually if needed: rm -rf $target"
    return 1
  else
    log_warn "Nothing to remove: $target does not exist"
  fi

  read -rp "Clean data/state/cache for '${appname}'? [y/N] " answer
  if [[ "$answer" =~ ^[Yy]$ ]]; then
    clean_data "$appname"
    log_ok "Cleaned data for '${appname}'"
  fi
}

# ─── Команда: status ───
cmd_status() {
  echo -e "${CYAN}Anvil Installation Status${NC}"
  echo ""

  # Проверяем default nvim
  local nvim_config="${NVIM_CONFIG_HOME}/nvim"
  if [[ -L "$nvim_config" ]]; then
    local target
    target=$(readlink -f "$nvim_config")
    echo -e "  Default (nvim):  ${GREEN}→ ${target}${NC}"
  elif [[ -d "$nvim_config" ]]; then
    echo -e "  Default (nvim):  ${YELLOW}exists (not anvil symlink)${NC}"
  else
    echo -e "  Default (nvim):  ${RED}not configured${NC}"
  fi

  # Проверяем anvil-* конфигурации
  for dir in "${NVIM_CONFIG_HOME}"/anvil-*/; do
    [[ -d "$dir" ]] || continue
    local name
    name=$(basename "$dir")
    if [[ -L "$dir" ]]; then
      # Убираем trailing slash для readlink
      local target
      target=$(readlink -f "${dir%/}")
      echo -e "  ${name}:  ${GREEN}→ ${target}${NC}"
    else
      echo -e "  ${name}:  ${YELLOW}exists (not symlink)${NC}"
    fi
  done

  echo ""

  # Проверяем source
  echo -e "${CYAN}Source Frameworks:${NC}"
  for dir in "${SRC_DIR}"/*/; do
    local name
    name=$(basename "$dir")
    echo -e "  ${name}: ${GREEN}${dir}${NC}"
  done
}

# ─── Команда: bootstrap ───
# Запускает headless nvim для инициализации плагинов
cmd_bootstrap() {
  local appname="${1:-nvim}"

  log_info "Bootstrapping '${appname}'..."
  NVIM_APPNAME="$appname" nvim --headless -c 'quitall' 2>&1 || true
  log_ok "Bootstrap complete for '${appname}'"
}

# ─── Команда: shell-setup ───
# Выводит алиасы для добавления в shell config
cmd_shell_setup() {
  echo "# Anvil Neovim aliases"
  echo "# Add these to your ~/.bashrc, ~/.zshrc, or equivalent"
  echo ""

  for dir in "${SRC_DIR}"/*/; do
    local name
    name=$(basename "$dir")
    [[ "$name" == "common" ]] && continue

    echo "alias anvil-${name}='NVIM_APPNAME=anvil-${name} nvim'"
  done

  echo ""
  echo "# Function to switch between configurations"
  echo 'anvil() {'
  echo '  local config="${1:-astronvim}"'
  echo '  shift 2>/dev/null || true'
  echo '  NVIM_APPNAME="anvil-${config}" nvim "$@"'
  echo '}'
}

# ─── Usage ───
usage() {
  echo "Usage: $0 <command> [options]"
  echo ""
  echo "Commands:"
  echo "  install <framework> [appname]  Install framework as isolated config"
  echo "                                 (default appname: anvil-<framework>)"
  echo "  install-default <framework>    Install framework as default nvim config"
  echo "  install-all                    Install all frameworks as isolated configs"
  echo "  uninstall <appname>            Remove installed configuration"
  echo "  status                         Show installation status"
  echo "  bootstrap [appname]            Initialize plugins for a configuration"
  echo "  shell-setup                    Print shell aliases for configurations"
  echo "  clean <appname>                Clean data/state/cache for a config"
  echo ""
  echo "Frameworks:"
  for dir in "${SRC_DIR}"/*/; do
    local name
    name=$(basename "$dir")
    [[ "$name" == "common" ]] && echo "  ${name} (shared library)" && continue
    echo "  ${name}"
  done
  echo ""
  echo "Examples:"
  echo "  $0 install astronvim              # Install as anvil-astronvim"
  echo "  $0 install nvchad                 # Install as anvil-nvchad"
  echo "  $0 install-all                    # Install all"
  echo "  $0 install-default astronvim      # Replace default nvim config"
  echo "  $0 bootstrap anvil-astronvim      # Initialize plugins"
  echo "  $0 status                         # Show what's installed"
  echo "  $0 shell-setup >> ~/.bashrc       # Add shell aliases"
}

# ─── Main ───
main() {
  print_header
  check_deps

  local cmd="${1:-}"
  shift 2>/dev/null || true

  case "$cmd" in
    install)
      [[ $# -lt 1 ]] && { log_err "Framework name required"; usage; exit 1; }
      cmd_install "$@"
      ;;

      [[ $# -lt 1 ]] && { log_err "Framework name required"; usage; exit 1; }
      cmd_install_default "$@"
      ;;
    install-all)
      cmd_install_all

    uninstall)

}

## Как это работает

### Механизм подключения common

```
┌─────────────────────────────────────────────────────┐
│                   NVIM_APPNAME                       │
│              (anvil-astronvim или anvil-nvchad)      │
└──────────────────────┬──────────────────────────────┘
                       │
                       ▼
┌──────────────────────────────────────────────────────┐
│  src/astronvim/init.lua   или   src/nvchad/init.lua  │
│  (bootstrap lazy.nvim)                                │
└──────────────────────┬───────────────────────────────┘
                       │
                       ▼
┌──────────────────────────────────────────────────────┐
│  anvil_bridge.lua                                     │
│  1. Вычисляет путь к src/common через debug.getinfo  │
│  2. Добавляет src/common в vim.opt.runtimepath       │
│  3. Вызывает require("anvil").setup()                │
│  4. Возвращает common plugin specs                    │
└──────────┬──────────────────────┬────────────────────┘
           │                      │
           ▼                      ▼
┌─────────────────┐    ┌────────────────────┐
│  src/common/    │    │  Framework-specific │
│  lua/anvil/     │    │  plugins & config   │
│  - options      │    │  (astrocore, etc.)  │
│  - keymaps      │    │                     │
│  - autocmds     │    │                     │
│  - plugins/     │    │                     │
└─────────────────┘    └────────────────────┘
```

### Почему симлинки, а не копирование

Симлинк `~/.config/anvil-astronvim → repo/src/astronvim` означает:

1. **Относительные пути в bridge работают** — `debug.getinfo` возвращает реальный путь через симлинк, и `../common` разрешается корректно
2. **Изменения в репозитории сразу видны** — не нужно пересинхронизировать
3. **Git отслеживает всё в одном месте**

### Изоляция через NVIM_APPNAME

Neovim 0.10+ поддерживает `NVIM_APPNAME`, что даёт полную изоляцию:

```
NVIM_APPNAME=anvil-astronvim
  config: ~/.config/anvil-astronvim/
  data:   ~/.local/share/anvil-astronvim/
  state:  ~/.local/state/anvil-astronvim/
  cache:  ~/.cache/anvil-astronvim/

NVIM_APPNAME=anvil-nvchad
  config: ~/.config/anvil-nvchad/
  data:   ~/.local/share/anvil-nvchad/
  state:  ~/.local/state/anvil-nvchad/
  cache:  ~/.cache/anvil-nvchad/

nvim (default, без NVIM_APPNAME)
  config: ~/.config/nvim/
  data:   ~/.local/share/nvim/
  ...
```

Все три конфигурации работают **полностью независимо** — свои плагины, свой кэш, своё состояние.

```

## Расширение общей конфигурации

Чтобы добавить новый общий плагин, достаточно создать файл в `src/common/lua/anvil/plugins/` или дополнить существующий:

```lua
-- src/common/lua/anvil/plugins/editor.lua
-- Добавляем новый плагин в массив:
return {
  -- ... существующие плагины ...

  {
    "folke/flash.nvim",
    event = "VeryLazy",
    opts = {},
    keys = {
      { "s", mode = { "n", "x", "o" }, function() require("flash").jump() end, desc = "Flash" },
    },
  },
}
```

Этот плагин автоматически появится в обоих фреймворках при следующем запуске, потому что `anvil_bridge.lua` вызывает `anvil.plugin_specs()`, который собирает все модули из `anvil.plugins.*`.
