# Anvil — Workflow Layer

Task management, terminals, sessions, project management, and file marks.

## Architecture

```text
lua/anvil/workflow/
├── init.lua      # workflow subsystem setup
├── tasks.lua     # overseer task configuration
├── terminal.lua  # terminal profiles
├── project.lua   # project detection and switching
├── session.lua   # session management
├── marks.lua     # file marks (harpoon is in editor layer)
└── notes.lua     # project-local notes

lua/plugins/
└── workflow.lua  # workflow plugin specs
```

## Task Management

### `lua/anvil/workflow/tasks.lua`

<!-- file: lua/anvil/workflow/tasks.lua -->
```lua
--- Anvil task management (overseer.nvim)
--- Tasks come from language packs + auto-detection
---@class AnvilWorkflowTasks
local M = {}

--- Get overseer opts
---@return table
function M.opts()
  return {
    strategy = "toggleterm",
    task_list = {
      direction = "bottom",
      min_height = 15,
      max_height = 25,
      default_detail = 1,
      bindings = {
        ["<CR>"]  = "RunAction",
        ["<C-e>"] = "Edit",
        ["<C-v>"] = false,
        ["<C-s>"] = false,
        ["<C-q>"] = "QuickAction",
        ["q"]     = "Close",
      },
    },
    form = { border = "rounded", win_opts = { winblend = 0 } },
    confirm = { border = "rounded", win_opts = { winblend = 0 } },
    task_win = { border = "rounded", win_opts = { winblend = 0 } },
  }
end

--- Register language pack tasks with overseer
--- Called after overseer is loaded
function M.register_lang_tasks()
  local lang_ok, lang = pcall(require, "anvil.lang")
  if not lang_ok then return end

  local overseer_ok, overseer = pcall(require, "overseer")
  if not overseer_ok then return end

  local tasks = lang.tasks()
  for _, task in ipairs(tasks) do
    overseer.register_template({
      name = task.name,
      builder = function()
        return {
          cmd = type(task.cmd) == "table" and task.cmd or { task.cmd },
          name = task.name,
          cwd = vim.fn.getcwd(),
        }
      end,
      desc = task.name .. " (" .. task.lang .. ")",
      condition = {
        filetype = task.filetypes,
      },
    })
  end
end

return M
```

## Session Management

### `lua/anvil/workflow/session.lua`

<!-- file: lua/anvil/workflow/session.lua -->
```lua
--- Anvil session management
--- Auto-save and restore sessions per project
---@class AnvilWorkflowSession
local M = {}

--- Get persistence.nvim opts
---@return table
function M.opts()
  return {
    dir = vim.fn.stdpath("state") .. "/sessions/",
    options = {
      "buffers", "curdir", "tabpages", "winsize", "help", "globals", "skiprtp",
    },
    pre_save = function()
      -- Close ephemeral windows before saving
      for _, win in ipairs(vim.api.nvim_list_wins()) do
        local buf = vim.api.nvim_win_get_buf(win)
        local bt = vim.bo[buf].buftype
        if bt == "nofile" or bt == "terminal" or bt == "prompt" then
          pcall(vim.api.nvim_win_close, win, true)
        end
      end
    end,
  }
end

return M
```

## Project Detection

### `lua/anvil/workflow/project.lua`

<!-- file: lua/anvil/workflow/project.lua -->
```lua
--- Anvil project detection
--- Finds project root via markers
---@class AnvilWorkflowProject
local M = {}

--- Root markers in priority order
---@type string[]
M.markers = {
  ".git",
  "Cargo.toml",
  "go.mod",
  "package.json",
  "pyproject.toml",
  "Makefile",
  "CMakeLists.txt",
  ".editorconfig",
}

--- Find project root from a given path
---@param path? string starting path (default: current file or cwd)
---@return string root project root path
function M.find_root(path)
  path = path or vim.fn.expand("%:p:h")
  if path == "" then path = vim.fn.getcwd() end

  -- Use vim.fs.find (Neovim 0.10+)
  local found = vim.fs.find(M.markers, {
    path = path,
    upward = true,
    stop = vim.uv.os_homedir(),
  })

  if #found > 0 then
    return vim.fn.fnamemodify(found[1], ":h")
  end

  return vim.fn.getcwd()
end

--- Get project name from root
---@param root? string project root
---@return string name
function M.name(root)
  root = root or M.find_root()
  return vim.fn.fnamemodify(root, ":t")
end

--- Get project type based on marker files
---@param root? string project root
---@return string type e.g. "rust", "node", "python", "go", "unknown"
function M.type(root)
  root = root or M.find_root()

  local checks = {
    { "Cargo.toml",     "rust" },
    { "go.mod",         "go" },
    { "package.json",   "node" },
    { "pyproject.toml", "python" },
    { "setup.py",       "python" },
    { "CMakeLists.txt", "cmake" },
    { "Makefile",       "make" },
  }

  for _, check in ipairs(checks) do
    if vim.fn.filereadable(root .. "/" .. check[1]) == 1 then
      return check[2]
    end
  end

  return "unknown"
end

return M
```

## Workflow Plugin Specs

### `lua/plugins/workflow.lua`

<!-- file: lua/plugins/workflow.lua -->
```lua
--- Anvil workflow plugin specs
---@type LazySpec
return {
  -- ╭──────────────────────────────────────────────────────────╮
  -- │ Overseer — task runner                                   │
  -- ╰──────────────────────────────────────────────────────────╯
  {
    "stevearc/overseer.nvim",
    cmd = {
      "OverseerRun", "OverseerToggle", "OverseerInfo",
      "OverseerBuild", "OverseerQuickAction",
    },
    opts = function()
      return require("anvil.workflow.tasks").opts()
    end,
    config = function(_, opts)
      require("overseer").setup(opts)
      -- Register language pack tasks after overseer loads
      vim.defer_fn(function()
        require("anvil.workflow.tasks").register_lang_tasks()
      end, 100)
    end,
    keys = {
      { "<leader>or", "<cmd>OverseerRun<cr>",         desc = "Run task" },
      { "<leader>ot", "<cmd>OverseerToggle<cr>",      desc = "Toggle task list" },
      { "<leader>ob", "<cmd>OverseerBuild<cr>",       desc = "Build task" },
      { "<leader>oq", "<cmd>OverseerQuickAction<cr>", desc = "Quick action" },
      { "<leader>oi", "<cmd>OverseerInfo<cr>",        desc = "Overseer info" },
    },
  },

  -- ╭──────────────────────────────────────────────────────────╮
  -- │ persistence.nvim — session management                    │
  -- ╰──────────────────────────────────────────────────────────╯
  {
    "folke/persistence.nvim",
    event = "BufReadPre",
    opts = function()
      return require("anvil.workflow.session").opts()
    end,
    keys = {
      { "<leader>qs", function() require("persistence").load() end,                desc = "Restore session" },
      { "<leader>ql", function() require("persistence").load({ last = true }) end, desc = "Restore last session" },
      { "<leader>qd", function() require("persistence").stop() end,                desc = "Don't save session" },
    },
  },

  -- ╭──────────────────────────────────────────────────────────╮
  -- │ project.nvim — project detection + telescope             │
  -- ╰──────────────────────────────────────────────────────────╯
  {
    "ahmedkhalf/project.nvim",
    event = "VeryLazy",
    config = function()
      require("project_nvim").setup({
        detection_methods = { "pattern", "lsp" },
        patterns = require("anvil.workflow.project").markers,
        show_hidden = true,
        silent_chdir = true,
        scope_chdir = "global",
      })
      -- Register telescope extension
      pcall(function()
        require("telescope").load_extension("projects")
      end)
    end,
    keys = {
      { "<leader>fp", "<cmd>Telescope projects<cr>", desc = "Projects" },
    },
  },

  -- ╭──────────────────────────────────────────────────────────╮
  -- │ which-key groups                                         │
  -- ╰──────────────────────────────────────────────────────────╯
  {
    "folke/which-key.nvim",
    optional = true,
    opts = {
      spec = {
        { "<leader>o", group = "Overseer" },
        { "<leader>q", group = "Session" },
      },
    },
  },
}
```
