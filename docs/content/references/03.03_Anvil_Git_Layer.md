# Anvil — Git Layer

Git integration: signs, blame, diff, conflict resolution, lazygit, worktrees, and AI-powered review.

## Architecture

```text
lua/anvil/git/
├── init.lua       # git subsystem setup
├── signs.lua      # gitsigns configuration
├── blame.lua      # blame utilities
├── diff.lua       # diffview configuration
├── conflict.lua   # conflict resolution

lua/plugins/
└── git.lua        # git plugin specs
```

## Git Subsystem Entry Point

### `lua/anvil/git/init.lua`

<!-- file: lua/anvil/git/init.lua -->
```lua
--- Anvil git subsystem setup
--- Coordinates git-related modules
--- Depends on: core/events (optional)
---@class AnvilGit
local M = {}

--- Setup git subsystem
--- Called lazily when git plugins load
function M.setup()
  local events_ok, events = pcall(require, "anvil.core.events")
  if events_ok then
    events.emit("git:ready")
  end
end

--- Check if current directory is a git repository
---@return boolean
function M.is_repo()
  local result = vim.fn.system("git rev-parse --is-inside-work-tree 2>/dev/null")
  return vim.v.shell_error == 0 and vim.trim(result) == "true"
end

--- Get current branch name
---@return string branch name or empty string
function M.branch()
  if not M.is_repo() then return "" end
  local result = vim.fn.system("git branch --show-current 2>/dev/null")
  return vim.v.shell_error == 0 and vim.trim(result) or ""
end

--- Get repository root
---@return string|nil
function M.root()
  if not M.is_repo() then return nil end
  local result = vim.fn.system("git rev-parse --show-toplevel 2>/dev/null")
  return vim.v.shell_error == 0 and vim.trim(result) or nil
end

--- Get short status summary
---@return { staged: number, modified: number, untracked: number }
function M.status_counts()
  local counts = { staged = 0, modified = 0, untracked = 0 }
  if not M.is_repo() then return counts end

  local status = vim.fn.system("git status --porcelain 2>/dev/null")
  if vim.v.shell_error ~= 0 then return counts end

  for line in status:gmatch("[^\n]+") do
    local index = line:sub(1, 1)
    local worktree = line:sub(2, 2)
    if index ~= " " and index ~= "?" then
      counts.staged = counts.staged + 1
    end
    if worktree ~= " " and worktree ~= "?" then
      counts.modified = counts.modified + 1
    end
    if index == "?" then
      counts.untracked = counts.untracked + 1
    end
  end

  return counts
end

return M
```

```lua
      end, "Stage hunk (visual)")
      map("v", "<leader>gr", function()
        gs.reset_hunk({ vim.fn.line("."), vim.fn.line("v") })
      end, "Reset hunk (visual)")
      map("n", "<leader>gS", gs.stage_buffer, "Stage buffer")
      map("n", "<leader>gu", gs.undo_stage_hunk, "Undo stage hunk")
      map("n", "<leader>gR", gs.reset_buffer, "Reset buffer")
      map("n", "<leader>gp", gs.preview_hunk, "Preview hunk")
      map("n", "<leader>gb", function() gs.blame_line({ full = true }) end, "Blame line")

      map("n", "<leader>gd", gs.diffthis, "Diff this")

      map({ "o", "x" }, "ih", ":<C-U>Gitsigns select_hunk<CR>", "Select hunk")
    end,
  }
```

## Blame Utilities

### `lua/anvil/git/blame.lua`

```lua
--- Anvil git blame utilities
--- Provides blame information beyond what gitsigns offers
--- Depends on: nothing (uses vim.fn.system)
---@class AnvilGitBlame
local M = {}

--- Get full blame info for a specific line
---@param file? string file path (default: current buffer)
---@param line? number line number (default: cursor line)
---@return table|nil { hash: string, author: string, date: string, summary: string }
function M.line_info(file, line)
  file = file or vim.fn.expand("%:p")
  line = line or vim.fn.line(".")

  if file == "" or vim.fn.filereadable(file) ~= 1 then
    return nil
  end

  local result = vim.fn.system(string.format(
    "git blame -L %d,%d --porcelain %s 2>/dev/null",
    line, line, vim.fn.shellescape(file)
  ))

  if vim.v.shell_error ~= 0 or result == "" then
    return nil
  end

  local info = {}
  info.hash = result:match("^(%x+)")
  info.author = result:match("author (.-)%\n")
  info.date = result:match("author%-time (%d+)")
  info.summary = result:match("summary (.-)%\n")

  if info.date then
    info.date = os.date("%Y-%m-%d %H:%M", tonumber(info.date))
  end

  return info
end

--- Show blame for current line in a floating window
function M.show_line_blame()
  local info = M.line_info()
  if not info then
    vim.notify("No blame info available", vim.log.levels.INFO)
    return
  end

  local text = string.format(
    "%s\n\nAuthor: %s\nDate:   %s\nCommit: %s",
    info.summary or "No summary",
    info.author or "Unknown",
    info.date or "Unknown",
    info.hash and info.hash:sub(1, 8) or "Unknown"
  )

  vim.lsp.util.open_floating_preview(
    vim.split(text, "\n"),
    "markdown",
    { border = "rounded", title = "Git Blame", title_pos = "center" }
  )
end

--- Open the commit that last changed the current line
function M.open_commit()
  local info = M.line_info()
  if not info or not info.hash then
    vim.notify("No blame info available", vim.log.levels.INFO)
    return
  end

  -- Skip uncommitted changes
  if info.hash:match("^0+$") then
    vim.notify("Line is uncommitted", vim.log.levels.INFO)
    return
  end

  vim.cmd("DiffviewOpen " .. info.hash .. "^.." .. info.hash)
end

return M
```

## Conflict Resolution

### `lua/anvil/git/conflict.lua`

```lua
--- Anvil git conflict resolution utilities
--- Complements git-conflict.nvim with additional helpers
--- Depends on: nothing
---@class AnvilGitConflict
local M = {}

--- Check if current buffer has unresolved conflicts
---@param bufnr? number buffer number (default: current)
---@return boolean
function M.has_conflicts(bufnr)
  bufnr = bufnr or 0
  local lines = vim.api.nvim_buf_get_lines(bufnr, 0, -1, false)
  for _, line in ipairs(lines) do
    if line:match("^<<<<<<<") or line:match("^>>>>>>>") then
      return true
    end
  end
  return false
end

--- Count unresolved conflict markers in a buffer
---@param bufnr? number
---@return number
function M.count_conflicts(bufnr)
  bufnr = bufnr or 0
  local lines = vim.api.nvim_buf_get_lines(bufnr, 0, -1, false)
  local count = 0
  for _, line in ipairs(lines) do
    if line:match("^<<<<<<<") then
      count = count + 1
    end
  end
  return count
end

--- Jump to the next conflict marker
---@param direction? "next"|"prev" default: "next"
function M.jump(direction)
  direction = direction or "next"
  local pattern = "^<<<<<<< "
  if direction == "next" then
    vim.fn.search(pattern, "W")
  else
    vim.fn.search(pattern, "bW")
  end
end

return M
```

## Worktree Management

### `lua/anvil/git/worktree.lua`

```lua
--- Anvil git worktree management
--- Helpers for working with git worktrees
--- Depends on: nothing (uses vim.fn.system)
---@class AnvilGitWorktree
local M = {}

--- List all worktrees
---@return table[] { path: string, branch: string, head: string, bare: boolean }
function M.list()
  local result = vim.fn.system("git worktree list --porcelain 2>/dev/null")
  if vim.v.shell_error ~= 0 then return {} end

  local worktrees = {}
  local current = {}

  for line in result:gmatch("[^\n]+") do
    if line:match("^worktree ") then
      if current.path then
        worktrees[#worktrees + 1] = current
      end
      current = { path = line:match("^worktree (.+)") }
    elseif line:match("^HEAD ") then
      current.head = line:match("^HEAD (.+)")
    elseif line:match("^branch ") then
      current.branch = line:match("^branch refs/heads/(.+)")
    elseif line == "bare" then
      current.bare = true
    end
  end

  if current.path then
    worktrees[#worktrees + 1] = current
  end

  return worktrees
end

--- Create a new worktree
---@param path string worktree path
---@param branch string branch name
---@param create_branch? boolean create new branch (default: false)
function M.add(path, branch, create_branch)
  local cmd = create_branch
    and string.format("git worktree add -b %s %s", vim.fn.shellescape(branch), vim.fn.shellescape(path))
    or string.format("git worktree add %s %s", vim.fn.shellescape(path), vim.fn.shellescape(branch))

  local result = vim.fn.system(cmd .. " 2>&1")
  if vim.v.shell_error ~= 0 then
    vim.notify("Failed to create worktree: " .. result, vim.log.levels.ERROR)
  else
    vim.notify("Worktree created: " .. path, vim.log.levels.INFO)
  end
end

--- Remove a worktree
---@param path string worktree path
---@param force? boolean force removal
function M.remove(path, force)
  local cmd = force
    and "git worktree remove --force " .. vim.fn.shellescape(path)
    or "git worktree remove " .. vim.fn.shellescape(path)

  local result = vim.fn.system(cmd .. " 2>&1")
  if vim.v.shell_error ~= 0 then
    vim.notify("Failed to remove worktree: " .. result, vim.log.levels.ERROR)
  else
    vim.notify("Worktree removed: " .. path, vim.log.levels.INFO)
  end
end

--- Interactive worktree picker
function M.pick()
  local worktrees = M.list()
  if #worktrees == 0 then
    vim.notify("No worktrees found", vim.log.levels.INFO)
    return
  end

  local items = {}
  for _, wt in ipairs(worktrees) do
    items[#items + 1] = string.format(
      "%s [%s]",
      wt.branch or wt.head and wt.head:sub(1, 8) or "detached",
      wt.path
    )
  end

  vim.ui.select(items, { prompt = "Switch worktree:" }, function(_, idx)
    if idx then
      vim.cmd("cd " .. worktrees[idx].path)
      vim.notify("Switched to: " .. worktrees[idx].path, vim.log.levels.INFO)
    end
  end)
end

return M
```
