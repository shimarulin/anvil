# Anvil — Git Layer

Git integration: signs, blame, diff, conflict resolution, lazygit, worktrees, and AI-powered review.

## Architecture

```text
lua/anvil/git/
├── init.lua       # git subsystem setup
├── signs.lua      # gitsigns configuration
├── blame.lua      # blame utilities
├── diff.lua       # diffview configuration
├── conflict.lua   # conflict resolution
└── worktree.lua   # git worktree management

lua/plugins/
└── git.lua        # git plugin specs
```

## Gitsigns Configuration

### `lua/anvil/git/signs.lua`

<!-- file: lua/anvil/git/signs.lua -->
```lua
--- Anvil gitsigns configuration
---@class AnvilGitSigns
local M = {}

local icons = require("anvil.core.icons")

--- Return gitsigns plugin opts
---@return table
function M.opts()
  return {
    signs = {
      add          = { text = "▎" },
      change       = { text = "▎" },
      delete       = { text = "" },
      topdelete    = { text = "" },
      changedelete = { text = "▎" },
      untracked    = { text = "▎" },
    },
    signs_staged = {
      add          = { text = "▎" },
      change       = { text = "▎" },
      delete       = { text = "" },
      topdelete    = { text = "" },
      changedelete = { text = "▎" },
    },
    current_line_blame = false,
    current_line_blame_opts = {
      virt_text = true,
      virt_text_pos = "eol",
      delay = 500,
    },
    current_line_blame_formatter = "<author>, <author_time:%Y-%m-%d> — <summary>",
    preview_config = {
      border = "rounded",
      style = "minimal",
    },
    on_attach = function(bufnr)
      local gs = package.loaded.gitsigns

      local function map(mode, l, r, desc)
        vim.keymap.set(mode, l, r, { buffer = bufnr, desc = desc })
      end

      -- Navigation
      map("n", "]h", function()
        if vim.wo.diff then return "]c" end
        vim.schedule(function() gs.next_hunk() end)
        return "<Ignore>"
      end, "Next hunk")

      map("n", "[h", function()
        if vim.wo.diff then return "[c" end
        vim.schedule(function() gs.prev_hunk() end)
        return "<Ignore>"
      end, "Previous hunk")

      -- Actions
      map("n", "<leader>gs", gs.stage_hunk, "Stage hunk")
      map("n", "<leader>gr", gs.reset_hunk, "Reset hunk")
      map("v", "<leader>gs", function()
        gs.stage_hunk({ vim.fn.line("."), vim.fn.line("v") })
      end, "Stage hunk (visual)")
      map("v", "<leader>gr", function()
        gs.reset_hunk({ vim.fn.line("."), vim.fn.line("v") })
      end, "Reset hunk (visual)")
      map("n", "<leader>gS", gs.stage_buffer, "Stage buffer")
      map("n", "<leader>gu", gs.undo_stage_hunk, "Undo stage hunk")
      map("n", "<leader>gR", gs.reset_buffer, "Reset buffer")
      map("n", "<leader>gp", gs.preview_hunk, "Preview hunk")
      map("n", "<leader>gb", function() gs.blame_line({ full = true }) end, "Blame line")
      map("n", "<leader>gB", gs.toggle_current_line_blame, "Toggle line blame")
      map("n", "<leader>gd", gs.diffthis, "Diff this")
      map("n", "<leader>gD", function() gs.diffthis("~") end, "Diff this (~)")

      -- Text object
      map({ "o", "x" }, "ih", ":<C-U>Gitsigns select_hunk<CR>", "Select hunk")
    end,
  }
end

return M
```

## Diffview Configuration

### `lua/anvil/git/diff.lua`

<!-- file: lua/anvil/git/diff.lua -->
```lua
--- Anvil diffview configuration
---@class AnvilGitDiff
local M = {}

--- Return diffview plugin opts
---@return table
function M.opts()
  return {
    enhanced_diff_hl = true,
    view = {
      default = { layout = "diff2_horizontal" },
      merge_tool = {
        layout = "diff3_mixed",
        disable_diagnostics = true,
      },
    },
    file_panel = {
      listing_style = "tree",
      tree_options = {
        flatten_dirs = true,
        folder_statuses = "only_folded",
      },
      win_config = { position = "left", width = 35 },
    },
    hooks = {
      diff_buf_read = function(bufnr)
        vim.opt_local.wrap = false
        vim.opt_local.list = false
        vim.opt_local.colorcolumn = ""
      end,
    },
  }
end

return M
```

## Git Plugin Specs

### `lua/plugins/git.lua`

<!-- file: lua/plugins/git.lua -->
```lua
--- Anvil git plugin specs
---@type LazySpec
return {
  -- ╭──────────────────────────────────────────────────────────╮
  -- │ Gitsigns                                                 │
  -- ╰──────────────────────────────────────────────────────────╯
  {
    "lewis6991/gitsigns.nvim",
    event = { "BufReadPre", "BufNewFile" },
    opts = function()
      return require("anvil.git.signs").opts()
    end,
  },

  -- ╭──────────────────────────────────────────────────────────╮
  -- │ Diffview                                                 │
  -- ╰──────────────────────────────────────────────────────────╯
  {
    "sindrets/diffview.nvim",
    dependencies = { "nvim-lua/plenary.nvim" },
    cmd = { "DiffviewOpen", "DiffviewClose", "DiffviewFileHistory" },
    opts = function()
      return require("anvil.git.diff").opts()
    end,
    keys = {
      { "<leader>gv", "<cmd>DiffviewOpen<cr>",                       desc = "Diffview open" },
      { "<leader>gc", "<cmd>DiffviewClose<cr>",                      desc = "Diffview close" },
      { "<leader>gh", "<cmd>DiffviewFileHistory %<cr>",              desc = "File history" },
      { "<leader>gH", "<cmd>DiffviewFileHistory<cr>",                desc = "Branch history" },
    },
  },

  -- ╭──────────────────────────────────────────────────────────╮
  -- │ Lazygit                                                  │
  -- ╰──────────────────────────────────────────────────────────╯
  {
    "kdheepak/lazygit.nvim",
    dependencies = { "nvim-lua/plenary.nvim" },
    cmd = { "LazyGit", "LazyGitConfig", "LazyGitCurrentFile" },
    keys = {
      { "<leader>gg", "<cmd>LazyGit<cr>",            desc = "LazyGit" },
      { "<leader>gf", "<cmd>LazyGitCurrentFile<cr>", desc = "LazyGit current file" },
    },
  },

  -- ╭──────────────────────────────────────────────────────────╮
  -- │ git-conflict.nvim                                        │
  -- ╰──────────────────────────────────────────────────────────╯
  {
    "akinsho/git-conflict.nvim",
    version = "*",
    event = "BufReadPost",
    opts = {
      default_mappings = true,
      disable_diagnostics = true,
    },
  },

  -- ╭──────────────────────────────────────────────────────────╮
  -- │ which-key group                                          │
  -- ╰──────────────────────────────────────────────────────────╯
  {
    "folke/which-key.nvim",
    optional = true,
    opts = {
      spec = {
        { "<leader>g", group = "Git" },
      },
    },
  },
}
```
