# Anvil — Core Layer

The core layer provides foundational services used by all other Anvil modules. It has zero plugin dependencies — only Neovim built-in APIs.

## Architecture

```text
lua/anvil/core/
├── env.lua        # OS environment variable access (simplified)
├── cache.lua      # in-memory cache with TTL
├── logger.lua     # structured logging
├── events.lua     # internal event bus
├── options.lua    # vim.opt / vim.g settings
├── autocmds.lua   # global autocommands
├── keys.lua       # centralized keymap registration
└── icons.lua      # unified icon set

lua/anvil/
├── init.lua       # boot sequence
└── config.lua     # user-facing configuration schema
```

## Dependency Rules

```text
core/*  → depends on NOTHING (only vim.* APIs)
config  → depends on nothing
init    → depends on config, core/*, lang, platform
ai/*    → depends on core/cache, core/logger, core/events
voice/* → depends on core/events, core/logger, ai (for interpretation)
lang/*  → depends on core/logger, core/events
editor/ → depends on core/keys, core/icons
ui/     → depends on core/icons, core/events, config
```

No circular dependencies. Core modules may depend on each other only in this order: `logger → cache → events → everything else`.

## Environment Variables

### `lua/anvil/core/env.lua`

Simplified module — provides only OS environment variable access. All structured configuration (AI providers, models, presets) is handled by JSON config files.

<!-- file: lua/anvil/core/env.lua -->
```lua
--- Anvil environment variable access
--- Simplified: provides only OS environment variable access.
--- Structured configuration uses JSON files (config/ai-providers.json).
---@class AnvilCoreEnv
local M = {}

--- Get an OS environment variable
---@param name string variable name
---@param default? string default value if not found
---@return string
function M.get(name, default)
  local val = os.getenv(name)
  if val and val ~= "" then return val end
  return default or ""
end

--- Get an environment variable as a number
---@param name string variable name
---@param default? number default value
---@return number
function M.get_number(name, default)
  local val = M.get(name, "")
  if val == "" then return default or 0 end
  return tonumber(val) or default or 0
end

--- Get an environment variable as a boolean
---@param name string variable name
---@param default? boolean default value
---@return boolean
function M.get_bool(name, default)
  local val = M.get(name, "")
  if val == "" then return default or false end
  val = val:lower()
  return val == "1" or val == "true" or val == "yes" or val == "on"
end

--- Check if a variable is set (non-empty)
---@param name string
---@return boolean
function M.is_set(name)
  return M.get(name, "") ~= ""
end

return M
```

```lua
--- Anvil structured logger
--- Provides leveled logging with optional file output
--- Depends on: nothing (uses vim.notify and vim.fn)
---@class AnvilCoreLogger
local M = {}

---@alias AnvilLogLevel "trace"|"debug"|"info"|"warn"|"error"

---@type table<AnvilLogLevel, number>
local LEVELS = { trace = 0, debug = 1, info = 2, warn = 3, error = 4 }

---@type table<AnvilLogLevel, number>
local VIM_LEVELS = {
  trace = vim.log.levels.TRACE,
  debug = vim.log.levels.DEBUG,
  info  = vim.log.levels.INFO,
  warn  = vim.log.levels.WARN,
  error = vim.log.levels.ERROR,
}

---@class AnvilLoggerConfig
---@field level AnvilLogLevel
---@field to_file boolean
---@field file_path? string
---@field notify boolean
---@field prefix string
M._config = {
  level = "info",
  to_file = false,
  file_path = nil,
  notify = true,
  prefix = "Anvil",
}

--- Setup logger
---@param opts? AnvilLoggerConfig
function M.setup(opts)
  M._config = vim.tbl_deep_extend("force", M._config, opts or {})
  if M._config.to_file and not M._config.file_path then
    M._config.file_path = vim.fn.stdpath("log") .. "/anvil.log"
  end
end

--- Check if a level is enabled
---@param level AnvilLogLevel
---@return boolean
local function is_enabled(level)
  return LEVELS[level] >= LEVELS[M._config.level]
end

--- Format a log message
---@param level AnvilLogLevel
---@param msg string
---@param ... any format arguments
---@return string
local function format_msg(level, msg, ...)
  local formatted = select("#", ...) > 0 and string.format(msg, ...) or msg
  return string.format("[%s][%s] %s", M._config.prefix, level:upper(), formatted)
end

--- Write to log file
---@param formatted string
local function write_file(formatted)
  if not M._config.to_file or not M._config.file_path then return end
  local timestamp = os.date("%Y-%m-%d %H:%M:%S")
  local line = timestamp .. " " .. formatted
  local file = io.open(M._config.file_path, "a")
  if file then
    file:write(line .. "\n")
    file:close()
  end
end

--- Log at a specific level
---@param level AnvilLogLevel
---@param msg string
---@param ... any format arguments
local function log(level, msg, ...)
  if not is_enabled(level) then return end
  local formatted = format_msg(level, msg, ...)
  write_file(formatted)
  if M._config.notify and LEVELS[level] >= LEVELS["warn"] then
    vim.schedule(function()
      vim.notify(formatted, VIM_LEVELS[level], { title = M._config.prefix })
    end)
  end
end

function M.trace(msg, ...) log("trace", msg, ...) end
function M.debug(msg, ...) log("debug", msg, ...) end
function M.info(msg, ...)  log("info",  msg, ...) end
function M.warn(msg, ...)  log("warn",  msg, ...) end
function M.error(msg, ...) log("error", msg, ...) end

--- Create a child logger with a sub-prefix
---@param sub_prefix string
---@return table
function M.child(sub_prefix)
  local child = {}
  local full_prefix = M._config.prefix .. ":" .. sub_prefix
  for _, level in ipairs({ "trace", "debug", "info", "warn", "error" }) do
    child[level] = function(msg, ...)
      if not is_enabled(level) then return end
      local formatted = string.format(
        "[%s][%s] %s",
        full_prefix, level:upper(),
        select("#", ...) > 0 and string.format(msg, ...) or msg
      )
      write_file(formatted)
      if M._config.notify and LEVELS[level] >= LEVELS["warn"] then
        vim.schedule(function()
          vim.notify(formatted, VIM_LEVELS[level], { title = full_prefix })
        end)
      end
    end
  end
  return child
end

return M
```

## Cache

### `lua/anvil/core/cache.lua`

In-memory cache with TTL.

<!-- file: lua/anvil/core/cache.lua -->
```lua
--- Anvil in-memory cache with TTL
--- Depends on: nothing
---@class AnvilCoreCache
local M = {}

---@class AnvilCacheEntry
---@field value any
---@field expires number unix timestamp (0 = never expires)

---@type table<string, AnvilCacheEntry>
M._store = {}

--- Get a cached value
---@param key string
---@return any? value (nil if not found or expired)
function M.get(key)
  local entry = M._store[key]
  if not entry then return nil end
  if entry.expires > 0 and os.time() > entry.expires then
    M._store[key] = nil
    return nil
  end
  return entry.value
end

--- Set a cached value
---@param key string
---@param value any
---@param ttl? number time-to-live in seconds (0 or nil = never expires)
function M.set(key, value, ttl)
  M._store[key] = {
    value = value,
    expires = (ttl and ttl > 0) and (os.time() + ttl) or 0,
  }
end

--- Get a cached value, or compute and cache it
---@param key string
---@param fn function(): any computation function
---@param ttl? number time-to-live in seconds
---@return any
function M.get_or_set(key, fn, ttl)
  local value = M.get(key)
  if value ~= nil then return value end
  value = fn()
  M.set(key, value, ttl)
  return value
end

--- Delete a cached value
---@param key string
function M.del(key)
  M._store[key] = nil
end

--- Delete all entries matching a pattern
---@param pattern string Lua pattern
function M.del_pattern(pattern)
  local to_delete = {}
  for key in pairs(M._store) do
    if key:match(pattern) then
      to_delete[#to_delete + 1] = key
    end
  end
  for _, key in ipairs(to_delete) do
    M._store[key] = nil
  end
end

--- Clear all cached values
function M.clear()
  M._store = {}
end

--- Remove expired entries
---@return number count of removed entries
function M.gc()
  local now = os.time()
  local removed = 0
  local to_delete = {}
  for key, entry in pairs(M._store) do
    if entry.expires > 0 and now > entry.expires then
      to_delete[#to_delete + 1] = key
      removed = removed + 1
    end
  end
  for _, key in ipairs(to_delete) do
    M._store[key] = nil
  end
  return removed
end

--- Get cache statistics
---@return { total: number, expired: number, active: number }
function M.stats()
  local total = 0
  local expired = 0
  local now = os.time()
  for _, entry in pairs(M._store) do
    total = total + 1
    if entry.expires > 0 and now > entry.expires then
      expired = expired + 1
    end
  end
  return { total = total, expired = expired, active = total - expired }
end

return M
```

## Event Bus

### `lua/anvil/core/events.lua`

<!-- file: lua/anvil/core/events.lua -->
```lua
--- Anvil internal event bus
--- Decouples modules via publish/subscribe pattern
--- Uses Neovim's User autocmd under the hood
--- Depends on: nothing
---@class AnvilCoreEvents
local M = {}

---@type number
M._augroup = nil

--- Setup event system
function M.setup()
  M._augroup = vim.api.nvim_create_augroup("Anvil", { clear = true })
end

--- Emit an event
---@param event string event name
---@param data? table payload
function M.emit(event, data)
  vim.api.nvim_exec_autocmds("User", {
    pattern = "Anvil:" .. event,
    data = data,
  })
end

--- Subscribe to an event
---@param event string event name
---@param callback function(args: table) handler function
---@param opts? { once?: boolean, desc?: string }
---@return number autocmd id (for unsubscribing)
function M.on(event, callback, opts)
  opts = opts or {}
  return vim.api.nvim_create_autocmd("User", {
    group = M._augroup,
    pattern = "Anvil:" .. event,
    callback = function(args) callback(args) end,
    once = opts.once or false,
    desc = opts.desc or ("Anvil event: " .. event),
  })
end

--- Subscribe to an event (fires only once)
---@param event string event name
---@param callback function(args: table)
---@return number autocmd id
function M.once(event, callback)
  return M.on(event, callback, { once = true })
end

--- Unsubscribe from an event
---@param id number autocmd id returned by on()
function M.off(id)
  pcall(vim.api.nvim_del_autocmd, id)
end

return M
```

```lua
      end
    end

    for key, value in pairs(config.g) do
      vim.g[key] = value
    end
  end
```

```lua
--- Anvil global autocommands
--- Depends on: core/events (optional, for emitting)
---@class AnvilCoreAutocmds
local M = {}

--- Setup global autocommands
function M.setup()
  local augroup = vim.api.nvim_create_augroup("AnvilAutocmds", { clear = true })

  -- ─── Highlight on yank ───
  vim.api.nvim_create_autocmd("TextYankPost", {
    group = augroup,
    desc = "Highlight yanked text",
    callback = function()
      vim.highlight.on_yank({ higroup = "IncSearch", timeout = 200 })
    end,
  })

  -- ─── Auto-resize splits ───
  vim.api.nvim_create_autocmd("VimResized", {
    group = augroup,
    desc = "Auto-resize panes on terminal resize",
    callback = function()
      vim.cmd("tabdo wincmd =")
    end,
  })

  -- ─── Trim trailing whitespace ───
  vim.api.nvim_create_autocmd("BufWritePre", {
    group = augroup,
    desc = "Trim trailing whitespace on save",
    callback = function()
      local save_cursor = vim.fn.getpos(".")
      vim.cmd([[%s/\s\+$//e]])
      vim.fn.setpos(".", save_cursor)
    end,
  })

  -- ─── .env file detection ───
  vim.api.nvim_create_autocmd({ "BufRead", "BufNewFile" }, {
    group = augroup,
    pattern = { ".env", ".env.*" },
    desc = "Set filetype for .env files",
    callback = function()
      vim.bo.filetype = "sh"
    end,
  })

  -- ─── Large file handling ───
  vim.api.nvim_create_autocmd("BufReadPre", {
    group = augroup,
    desc = "Disable heavy features for large files (>1MB)",
    callback = function(args)
      local ok, stats = pcall(vim.uv.fs_stat, vim.api.nvim_buf_get_name(args.buf))
      if ok and stats and stats.size > 1024 * 1024 then
        vim.b[args.buf].large_buf = true
        vim.opt_local.foldmethod = "manual"
        vim.opt_local.spell = false
        vim.opt_local.syntax = "off"
      end
    end,
  })

  -- ─── Return to last edit position ───
  vim.api.nvim_create_autocmd("BufReadPost", {
    group = augroup,
    desc = "Return to last edit position when opening file",
    callback = function(args)
      local mark = vim.api.nvim_buf_get_mark(args.buf, '"')
      local line_count = vim.api.nvim_buf_line_count(args.buf)
      if mark[1] > 0 and mark[1] <= line_count then
        pcall(vim.api.nvim_win_set_cursor, 0, mark)
      end
    end,
  })

  -- ─── Auto-create parent directories on save ───
  vim.api.nvim_create_autocmd("BufWritePre", {
    group = augroup,
    desc = "Auto-create parent directories when saving",
    callback = function(args)
      local path = vim.fn.fnamemodify(args.file, ":p:h")
      if vim.fn.isdirectory(path) == 0 then
        vim.fn.mkdir(path, "p")
      end
    end,
  })

  -- ─── Custom filetypes ───
  vim.filetype.add({
    extension = {
      mdx = "mdx",
      env = "sh",
    },
    filename = {
      [".env"]                = "sh",
      [".env.local"]          = "sh",
      [".env.development"]    = "sh",
      [".env.production"]     = "sh",
      [".prettierrc"]         = "json",
      [".eslintrc"]           = "json",
      [".babelrc"]            = "json",
      ["tsconfig.json"]       = "jsonc",
      [".swcrc"]              = "json",
    },
    pattern = {
      ["%.env%.[%w_.-]+"] = "sh",
    },
  })
end

return M
```

```lua
--- Anvil centralized keymap registration
--- Core keymaps with no plugin dependencies
--- Plugin-dependent keymaps go in plugins/*.lua
--- Depends on: nothing
---@class AnvilCoreKeys
local M = {}

--- Leader key definitions (for documentation and which-key)
---@type table
M.leaders = {
  ["<leader>f"]  = "Find",
  ["<leader>g"]  = "Git",
  ["<leader>l"]  = "LSP",
  ["<leader>d"]  = "Debug",
  ["<leader>t"]  = "Test",
  ["<leader>a"]  = "AI",
  ["<leader>w"]  = "Workflow",
  ["<leader>v"]  = "Voice",
}

---@type table[]|nil
M._user_keys = nil

--- Setup core keymaps (non-plugin, always available)
---@param user_keys? table user keymap overrides
function M.setup(user_keys)
  local map = vim.keymap.set
  M._user_keys = user_keys or {}

  -- ─── Essential (no plugin dependency) ───
  map("n", "<Esc>", "<cmd>noh<CR>", { desc = "Clear search highlights" })
  map("n", "<leader>q", "<cmd>qa<CR>", { desc = "Quit all" })

  -- ─── Better window navigation ───
  map("n", "<C-h>", "<C-w>h", { desc = "Window left" })
  map("n", "<C-j>", "<C-w>j", { desc = "Window down" })
  map("n", "<C-k>", "<C-w>k", { desc = "Window up" })
  map("n", "<C-l>", "<C-w>l", { desc = "Window right" })

  -- ─── Window resize ───
  map("n", "<C-Up>",    "<cmd>resize +2<CR>",          { desc = "Resize up" })
  map("n", "<C-Down>",  "<cmd>resize -2<CR>",          { desc = "Resize down" })
  map("n", "<C-Left>",  "<cmd>vertical resize -2<CR>", { desc = "Resize left" })
  map("n", "<C-Right>", "<cmd>vertical resize +2<CR>", { desc = "Resize right" })

  -- ─── Move lines ───
  map("n", "<A-j>", "<cmd>m .+1<CR>==",     { desc = "Move line down" })
  map("n", "<A-k>", "<cmd>m .-2<CR>==",     { desc = "Move line up" })
  map("v", "<A-j>", ":m '>+1<CR>gv=gv",     { desc = "Move selection down" })
  map("v", "<A-k>", ":m '<-2<CR>gv=gv",     { desc = "Move selection up" })

  -- ─── Better visual paste (don't lose register) ───
  map("v", "p", '"_dP', { desc = "Paste without losing register" })

  -- ─── Stay in visual mode after indent ───
  map("v", "<", "<gv", { desc = "Indent left" })
  map("v", ">", ">gv", { desc = "Indent right" })

  -- ─── Splits ───
  map("n", "<leader>-",  "<cmd>split<CR>",  { desc = "Horizontal split" })
  map("n", "<leader>\\", "<cmd>vsplit<CR>", { desc = "Vertical split" })

  -- ─── Diagnostic navigation ───
  map("n", "]d", function() vim.diagnostic.goto_next() end, { desc = "Next diagnostic" })
  map("n", "[d", function() vim.diagnostic.goto_prev() end, { desc = "Previous diagnostic" })
  map("n", "<leader>e", function() vim.diagnostic.open_float() end, { desc = "Diagnostic float" })

  -- ─── Quickfix ───
  map("n", "]q", "<cmd>cnext<CR>", { desc = "Next quickfix" })
  map("n", "[q", "<cmd>cprev<CR>", { desc = "Previous quickfix" })

  -- ─── Toggles ───
  map("n", "<leader>uw", function()
    vim.wo.wrap = not vim.wo.wrap
    vim.notify("Wrap: " .. tostring(vim.wo.wrap))
  end, { desc = "Toggle wrap" })

  map("n", "<leader>un", function()
    vim.wo.relativenumber = not vim.wo.relativenumber
    vim.notify("Relative numbers: " .. tostring(vim.wo.relativenumber))
  end, { desc = "Toggle relative numbers" })

  map("n", "<leader>us", function()
    vim.wo.spell = not vim.wo.spell
    vim.notify("Spell: " .. tostring(vim.wo.spell))
  end, { desc = "Toggle spell check" })

  -- ─── Terminal navigation ───
  map("t", "<C-h>", "<C-\\><C-n><C-w>h", { desc = "Terminal: window left" })
  map("t", "<C-j>", "<C-\\><C-n><C-w>j", { desc = "Terminal: window down" })
  map("t", "<C-k>", "<C-\\><C-n><C-w>k", { desc = "Terminal: window up" })
  map("t", "<C-l>", "<C-\\><C-n><C-w>l", { desc = "Terminal: window right" })
end

--- Apply user keymap overrides after boot
--- Called from user/init.lua
---@param overrides table[] list of { mode, lhs, rhs, opts }
function M.apply_overrides(overrides)
  if not overrides then return end
  for _, keymap in ipairs(overrides) do
    vim.keymap.set(
      keymap.mode or "n",
      keymap.lhs,
      keymap.rhs,
      vim.tbl_extend("force", { desc = keymap.desc or "" }, keymap.opts or {})
    )
  end
end

return M
```

## Boot Sequence

### `lua/anvil/init.lua`

<!-- file: lua/anvil/init.lua -->
```lua
--- Anvil entry point
--- Orchestrates the boot sequence for all subsystems
---@class Anvil
local M = {}

M._version = "0.2.0"
M._ready = false

--- Boot Anvil
--- Called from the root init.lua: require("anvil").boot({...})
---@param opts? AnvilUserConfig
function M.boot(opts)
  if M._ready then return end

  -- ─── Phase 1: Core (synchronous, instant) ───
  local config = require("anvil.config")
  M.config = config.apply(opts or {})

  local logger = require("anvil.core.logger")
  logger.setup(M.config.log)

  local events = require("anvil.core.events")
  events.setup()

  local options = require("anvil.core.options")
  options.apply(M.config.options)

  require("anvil.core.autocmds").setup()
  require("anvil.core.keys").setup(M.config.keys)

  -- ─── Phase 2: Platform detection ───
  local platform_ok, platform = pcall(require, "anvil.platform")
  if platform_ok then platform.apply() end

  -- ─── Phase 3: Language registration ───
  -- Scans: anvil/lang/ → anvil-community/lang/ → user/lang/
  local lang_ok, lang = pcall(require, "anvil.lang")
  if lang_ok then lang.setup(M.config.languages) end

  -- ─── Phase 4: Commands ───
  M._register_commands()

  -- ─── Phase 5: Ready ───
  M._ready = true
  events.emit("ready", {
    version = M._version,
    languages = lang_ok and #lang.enabled() or 0,
  })

  -- ─── Phase 6: User layer ───
  local user_ok, user = pcall(require, "user")
  if user_ok and type(user) == "table" and user.setup then
    user.setup(M.config)
    logger.info("User layer loaded")
  end

  -- ─── Register voice commands from language packs ───
  -- Done after user layer so user voice commands have highest priority
  events.on("voice:initialized", function()
    local vc_ok, voice_commands = pcall(require, "anvil.voice.commands")
    if vc_ok then
      voice_commands.register_lang_commands()
    end
  end)

  logger.info(
    "Anvil v%s ready (%d languages, theme: %s, ui: %s)",
    M._version,
    lang_ok and #lang.enabled() or 0,
    M.config.ui.colorscheme,
    require("anvil.ui").detect_strategy()
  )
end

--- Register user commands
function M._register_commands()
  local cmd = vim.api.nvim_create_user_command

  cmd("Anvil", function(args)
    local sub = args.fargs[1]
    if not sub or sub == "" then
      M._show_menu()
    elseif sub == "health" then
      vim.cmd("checkhealth anvil")
    elseif sub == "version" then
      vim.notify("Anvil v" .. M._version, vim.log.levels.INFO)
    elseif sub == "config" then
      require("anvil.ai.providers").show_config()
    elseif sub == "theme" then
      local theme_name = args.fargs[2]
      if theme_name then
        pcall(vim.cmd.colorscheme, theme_name)
        vim.notify("Theme: " .. theme_name, vim.log.levels.INFO)
      else
        vim.notify("Current theme: " .. vim.g.colors_name, vim.log.levels.INFO)
      end
    elseif sub == "lang" then
      local lang = require("anvil.lang")
      local info = args.fargs[2]
      if info then
        lang.info(info)
      else
        vim.notify(
          "Languages: " .. table.concat(lang.enabled(), ", "),
          vim.log.levels.INFO
        )
      end
    elseif sub == "index" then
      require("anvil.ai.context").index_project()
    elseif sub == "cache" then
      local cache = require("anvil.core.cache")
      local stats = cache.stats()
      vim.notify(
        string.format(
          "Cache: %d active, %d expired, %d total",
          stats.active, stats.expired, stats.total
        ),
        vim.log.levels.INFO
      )
    elseif sub == "ui" then
      local ui = require("anvil.ui")
      vim.notify("UI strategy: " .. ui.detect_strategy(), vim.log.levels.INFO)
    else
      vim.notify("Unknown subcommand: " .. sub, vim.log.levels.WARN)
    end
  end, {
    nargs = "*",
    complete = function()
      return { "health", "version", "config", "theme", "lang", "index", "cache", "ui" }
    end,
    desc = "Anvil: Main command hub",
  })

  -- AI commands (delegate to ai.init)
  cmd("AnvilChat", function(a)
    require("anvil.ai").chat(a)
  end, { nargs = "?", range = true, desc = "Anvil: Open AI chat" })

  cmd("AnvilAsk", function(a)
    require("anvil.ai").ask(a.args)
  end, { nargs = "+", desc = "Anvil: Ask AI inline" })

  cmd("AnvilIndex", function()
    require("anvil.ai").index_project()
  end, { desc = "Anvil: Index current project" })

  cmd("AnvilCompletion", function()
    require("anvil.ai").toggle_completion()
  end, { desc = "Anvil: Toggle AI inline completions" })
end

--- Show main menu
function M._show_menu()
  local items = {
    "chat    — Open AI chat",
    "health  — Run health check",
    "version — Show version",
    "config  — Show AI provider config",
    "theme   — Show/change colorscheme",
    "lang    — Show enabled languages",
    "index   — Index current project",
    "cache   — Show cache statistics",
    "ui      — Show UI strategy",
  }
  vim.ui.select(items, { prompt = "Anvil:" }, function(choice)
    if not choice then return end
    local sub = choice:match("^(%S+)")
    if sub then vim.cmd("Anvil " .. sub) end
  end)
end

return M
```

```lua
--- Anvil core health check
--- :checkhealth anvil
local M = {}

function M.check()
  vim.health.start("Anvil Core")

  -- ─── Version ───
  local anvil_ok, anvil = pcall(require, "anvil")
  if anvil_ok then
    vim.health.ok("Anvil loaded (v" .. (anvil._version or "?") .. ")")
    if anvil._ready then
      vim.health.ok("Boot completed")
    else
      vim.health.warn("Boot not completed — was anvil.boot() called?")
    end
  else
    vim.health.error("Failed to load anvil module")
    return
  end

  -- ─── Core modules ───
  vim.health.start("Anvil Core Modules")
  local core_modules = {
    "anvil.core.env",
    "anvil.core.logger",
    "anvil.core.cache",
    "anvil.core.events",
    "anvil.core.options",
    "anvil.core.autocmds",
    "anvil.core.keys",
    "anvil.core.icons",
    "anvil.config",
  }
  for _, mod in ipairs(core_modules) do
    if pcall(require, mod) then
      vim.health.ok(mod)
    else
      vim.health.error(mod .. " — failed to load")
    end
  end

  -- ─── UI Strategy ───
  vim.health.start("Anvil UI")
  local ui_ok, ui = pcall(require, "anvil.ui")
  if ui_ok then
    local strategy = ui.detect_strategy()
    vim.health.ok("UI strategy: " .. strategy)
    vim.health.info("Colorscheme: " .. (vim.g.colors_name or "unknown"))
  end

  -- ─── AI Provider Configuration ───
  vim.health.start("Anvil AI Providers")
  local prov_ok, providers = pcall(require, "anvil.ai.providers")
  if prov_ok then
    local config_path = providers.config_path()
    if vim.fn.filereadable(config_path) == 1 then
      vim.health.ok("Config file: " .. config_path)
    else
      vim.health.info("No config file — using defaults")
    end
    providers.load()
    local preset = providers.active_preset()
    vim.health.ok("Active preset: " .. preset.name)
    for _, slot in ipairs(providers.slot_names()) do
      local resolved = providers.resolve(slot)
      if resolved then
        local has_key = resolved.provider.api_key ~= ""
        vim.health.ok(string.format(
          "  %-12s → %s (%s) %s",
          slot,
          resolved.model.model,
          resolved.provider.name,
          has_key and "✓" or "⚠ no API key"
        ))
      else
        vim.health.warn("  " .. slot .. " → not configured")
      end
    end
  end

  -- ─── Cache ───
  vim.health.start("Anvil Cache")
  local cache = require("anvil.core.cache")
  local stats = cache.stats()
  vim.health.ok(string.format(
    "%d entries (%d active, %d expired)",
    stats.total, stats.active, stats.expired
  ))

  -- ─── Languages ───
  vim.health.start("Anvil Languages")
  local lang_ok, lang = pcall(require, "anvil.lang")
  if lang_ok then
    local enabled = lang.enabled()
    local available = lang.available()
    vim.health.ok(string.format(
      "%d enabled out of %d available", #enabled, #available
    ))
    for _, name in ipairs(enabled) do
      local pack = lang.get(name)
      if pack then
        local parts = { name }
        local source = lang.source(name)
        if source and source ~= "builtin" then
          parts[1] = name .. " [" .. source .. "]"
        end
        if pack.lsp then
          parts[#parts + 1] = "LSP:" .. pack.lsp.server
          if pack.lsp.managed_by_plugin then
            parts[#parts] = parts[#parts] .. " (plugin)"
          end
        end
        if pack.formatter then
          parts[#parts + 1] = "fmt:" .. pack.formatter.tool
          if pack.formatter.managed_by_lsp then
            parts[#parts] = parts[#parts] .. " (LSP)"
          end
          if pack.formatter.chain then
            parts[#parts] = "fmt:" .. table.concat(pack.formatter.chain, "→")
          end
        end
        if pack.linter then
          parts[#parts + 1] = "lint:" .. pack.linter.tool
        end
        if pack.test then
          parts[#parts + 1] = "test:" .. pack.test.adapter
        end
        if pack.dap then
          parts[#parts + 1] = "dap:" .. pack.dap.adapter
        end
        if pack.voice_commands then
          parts[#parts + 1] = "voice:" .. #pack.voice_commands
        end
        vim.health.ok(table.concat(parts, " | "))
      end
    end
  else
    vim.health.warn("Language system not loaded")
  end

  -- ─── User layer ───
  vim.health.start("Anvil User Layer")
  local user_ok, _ = pcall(require, "user")
  if user_ok then
    vim.health.ok("User layer loaded (lua/user/init.lua)")
    local user_voice_ok, _ = pcall(require, "user.voice.commands")
    if user_voice_ok then
      vim.health.ok("User voice commands loaded")
    else
      vim.health.info("No user voice commands (lua/user/voice/commands.lua)")
    end
    local user_keys_ok, _ = pcall(require, "user.keys")
    if user_keys_ok then
      vim.health.ok("User keymap overrides loaded")
    else
      vim.health.info("No user keymap overrides (lua/user/keys.lua)")
    end
    local user_cmds_ok, _ = pcall(require, "user.commands")
    if user_cmds_ok then
      vim.health.ok("User text commands loaded")
    else
      vim.health.info("No user text commands (lua/user/commands.lua)")
    end
  else
    vim.health.info("No user layer (lua/user/init.lua not found)")
  end

  -- ─── Neovim version ───
  vim.health.start("Anvil Prerequisites")
  local v = vim.version()
  if v.major > 0 or v.minor >= 10 then
    vim.health.ok(string.format("Neovim %d.%d.%d", v.major, v.minor, v.patch))
  elseif v.minor >= 9 then
    vim.health.ok(string.format(
      "Neovim %d.%d.%d (minimum supported)", v.major, v.minor, v.patch
    ))
  else
    vim.health.error(string.format(
      "Neovim %d.%d.%d — need 0.9+ (recommend 0.10+)",
      v.major, v.minor, v.patch
    ))
  end

  -- Required external tools
  local required = {
    { "git",  "Git" },
    { "node", "Node.js (for MCP, Mason)" },
    { "npm",  "npm" },
  }
  for _, t in ipairs(required) do
    if vim.fn.executable(t[1]) == 1 then
      vim.health.ok(t[2])
    else
      vim.health.error(t[2] .. " — not found (required)")
    end
  end

  -- Recommended external tools
  local recommended = {
    { "rg",      "ripgrep (fast grep)" },
    { "fd",      "fd (fast find)" },
    { "lazygit", "lazygit (git TUI)" },
    { "delta",   "delta (git diff)" },
  }
  for _, t in ipairs(recommended) do
    if vim.fn.executable(t[1]) == 1 then
      vim.health.ok(t[2])
    else
      vim.health.info(t[2] .. " — not found (recommended)")
    end
  end
end

return M
```

```lua
    end)
```
