# Anvil — Core Layer

The core layer provides foundational services used by all other Anvil modules.
It has zero plugin dependencies — only Neovim built-in APIs.

## Architecture

```
lua/anvil/core/
├── env.lua        # .env file loader + environment variable access
├── cache.lua      # in-memory cache with TTL
├── logger.lua     # structured logging
├── events.lua     # internal event bus
├── options.lua    # vim.opt / vim.g settings
├── autocmds.lua   # global autocommands
├── keys.lua       # centralized keymap registration
└── icons.lua      # unified icon set

lua/anvil/
├── init.lua       # boot sequence
└── config.lua     # user-facing configuration schema
```

### Dependency Rules

```
core/*  →  depends on NOTHING (only vim.* APIs)
config  →  depends on core/env
init    →  depends on config, core/*, lang, platform
ai/*    →  depends on core/env, core/cache, core/logger, core/events
lang/*  →  depends on core/logger, core/events
editor/ →  depends on core/keys, core/icons
ui/     →  depends on core/icons, core/events
```

No circular dependencies. Core modules may depend on each other only
in this order: `env` → `logger` → `cache` → `events` → everything else.

## Environment Variables

### `lua/anvil/core/env.lua`

```lua
--- Anvil environment variable loader
--- Loads from OS environment and .env files
--- No dependencies on other Anvil modules
---@class AnvilCoreEnv
local M = {}

---@type table<string, string>
M._cache = {}

---@type boolean
M._loaded = false

--- Load .env file into cache
--- Supports: KEY=VALUE, KEY="VALUE", KEY='VALUE', comments (#), empty lines
---@param path string
---@return boolean success
local function load_dotenv(path)
  if vim.fn.filereadable(path) ~= 1 then
    return false
  end

  for _, line in ipairs(vim.fn.readfile(path)) do
    -- Skip empty lines and comments
    line = vim.trim(line)
    if line ~= "" and not line:match("^#") then
      -- Match KEY=VALUE (with optional quotes)
      local key, value = line:match("^([%w_]+)%s*=%s*(.*)$")
      if key and value then
        -- Remove surrounding quotes
        value = value:gsub("^[\"'](.-)[\"']$", "%1")
        -- Remove inline comments (only if not inside quotes)
        -- Simple heuristic: if the original didn't start with a quote
        if not line:match("^[%w_]+%s*=%s*[\"']") then
          value = value:gsub("%s+#.*$", "")
        end
        M._cache[key] = value
      end
    end
  end

  return true
end

--- Load environment from multiple .env files
--- Priority: OS env > project .env.local > project .env > config .env
function M.load()
  if M._loaded then return end

  -- Load in reverse priority order (later values DON'T override earlier)
  -- so we load lowest priority first, highest priority last
  local config_env = vim.fn.stdpath("config") .. "/.env"
  local project_env = vim.fn.getcwd() .. "/.env"
  local project_env_local = vim.fn.getcwd() .. "/.env.local"

  load_dotenv(config_env)
  load_dotenv(project_env)
  load_dotenv(project_env_local) -- highest file priority

  M._loaded = true
end

--- Get an environment variable
--- Priority: OS environment > .env files > default
---@param name string variable name
---@param default? string default value if not found
---@return string
function M.get(name, default)
  -- OS environment always wins
  local os_val = os.getenv(name)
  if os_val and os_val ~= "" then
    return os_val
  end

  -- Then .env files
  if not M._loaded then
    M.load()
  end

  local env_val = M._cache[name]
  if env_val and env_val ~= "" then
    return env_val
  end

  return default or ""
end

--- Get an environment variable as a number
---@param name string variable name
---@param default? number default value
---@return number
function M.get_number(name, default)
  local val = M.get(name, "")
  if val == "" then
    return default or 0
  end
  return tonumber(val) or default or 0
end

--- Get an environment variable as a boolean
--- Truthy: "1", "true", "yes", "on"
---@param name string variable name
---@param default? boolean default value
---@return boolean
function M.get_bool(name, default)
  local val = M.get(name, "")
  if val == "" then
    return default or false
  end
  val = val:lower()
  return val == "1" or val == "true" or val == "yes" or val == "on"
end

--- Check if a variable is set (non-empty)
---@param name string
---@return boolean
function M.is_set(name)
  return M.get(name, "") ~= ""
end

--- Reload .env files (useful after changing directory)
function M.reload()
  M._cache = {}
  M._loaded = false
  M.load()
end

--- Get all loaded variables (for debugging)
---@return table<string, string>
function M.all()
  if not M._loaded then M.load() end
  return vim.deepcopy(M._cache)
end

return M
```

## Logger

### `lua/anvil/core/logger.lua`

```lua
--- Anvil structured logger
--- Provides leveled logging with optional file output
--- Depends on: nothing (uses vim.notify and vim.fn)
---@class AnvilCoreLogger
local M = {}

---@alias AnvilLogLevel "trace"|"debug"|"info"|"warn"|"error"

---@type table<AnvilLogLevel, number>
local LEVELS = {
  trace = 0,
  debug = 1,
  info  = 2,
  warn  = 3,
  error = 4,
}

---@type table<AnvilLogLevel, number>
local VIM_LEVELS = {
  trace = vim.log.levels.TRACE,
  debug = vim.log.levels.DEBUG,
  info  = vim.log.levels.INFO,
  warn  = vim.log.levels.WARN,
  error = vim.log.levels.ERROR,
}

---@class AnvilLoggerConfig
---@field level AnvilLogLevel
---@field to_file boolean
---@field file_path? string
---@field notify boolean
---@field prefix string
M._config = {
  level = "info",
  to_file = false,
  file_path = nil,
  notify = true,
  prefix = "Anvil",
}

--- Setup logger
---@param opts? AnvilLoggerConfig
function M.setup(opts)
  M._config = vim.tbl_deep_extend("force", M._config, opts or {})
  if M._config.to_file and not M._config.file_path then
    M._config.file_path = vim.fn.stdpath("log") .. "/anvil.log"
  end
end

--- Check if a level is enabled
---@param level AnvilLogLevel
---@return boolean
local function is_enabled(level)
  return LEVELS[level] >= LEVELS[M._config.level]
end

--- Format a log message
---@param level AnvilLogLevel
---@param msg string
---@param ... any format arguments
---@return string
local function format_msg(level, msg, ...)
  local formatted = select("#", ...) > 0 and string.format(msg, ...) or msg
  return string.format("[%s][%s] %s", M._config.prefix, level:upper(), formatted)
end

--- Write to log file
---@param formatted string
local function write_file(formatted)
  if not M._config.to_file or not M._config.file_path then return end

  local timestamp = os.date("%Y-%m-%d %H:%M:%S")
  local line = timestamp .. " " .. formatted

  local file = io.open(M._config.file_path, "a")
  if file then
    file:write(line .. "\n")
    file:close()
  end
end

--- Log at a specific level
---@param level AnvilLogLevel
---@param msg string
---@param ... any format arguments
local function log(level, msg, ...)
  if not is_enabled(level) then return end

  local formatted = format_msg(level, msg, ...)

  -- File output (always, if enabled)
  write_file(formatted)

  -- Notify (only for warn and above, or if explicitly enabled)
  if M._config.notify and LEVELS[level] >= LEVELS["warn"] then
    vim.schedule(function()
      vim.notify(formatted, VIM_LEVELS[level], { title = M._config.prefix })
    end)
  end
end

--- Log at trace level
---@param msg string
---@param ... any
function M.trace(msg, ...) log("trace", msg, ...) end

--- Log at debug level
---@param msg string
---@param ... any
function M.debug(msg, ...) log("debug", msg, ...) end

--- Log at info level
---@param msg string
---@param ... any
function M.info(msg, ...) log("info", msg, ...) end

--- Log at warn level (shows notification)
---@param msg string
---@param ... any
function M.warn(msg, ...) log("warn", msg, ...) end

--- Log at error level (shows notification)
---@param msg string
---@param ... any
function M.error(msg, ...) log("error", msg, ...) end

--- Create a child logger with a sub-prefix
---@param sub_prefix string
---@return table logger-like table with trace/debug/info/warn/error
function M.child(sub_prefix)
  local child = {}
  local full_prefix = M._config.prefix .. ":" .. sub_prefix

  for _, level in ipairs({ "trace", "debug", "info", "warn", "error" }) do
    child[level] = function(msg, ...)
      if not is_enabled(level) then return end
      local formatted = string.format(
        "[%s][%s] %s",
        full_prefix,
        level:upper(),
        select("#", ...) > 0 and string.format(msg, ...) or msg
      )
      write_file(formatted)
      if M._config.notify and LEVELS[level] >= LEVELS["warn"] then
        vim.schedule(function()
          vim.notify(formatted, VIM_LEVELS[level], { title = full_prefix })
        end)
      end
    end
  end

  return child
end

return M
```

## Cache

### `lua/anvil/core/cache.lua`

```lua
--- Anvil in-memory cache with TTL
--- Depends on: nothing
---@class AnvilCoreCache
local M = {}

---@class AnvilCacheEntry
---@field value any
---@field expires number unix timestamp (0 = never expires)

---@type table<string, AnvilCacheEntry>
M._store = {}

--- Get a cached value
---@param key string
---@return any? value (nil if not found or expired)
function M.get(key)
  local entry = M._store[key]
  if not entry then return nil end

  -- Check expiration
  if entry.expires > 0 and os.time() > entry.expires then
    M._store[key] = nil
    return nil
  end

  return entry.value
end

--- Set a cached value
---@param key string
---@param value any
---@param ttl? number time-to-live in seconds (0 or nil = never expires)
function M.set(key, value, ttl)
  M._store[key] = {
    value = value,
    expires = (ttl and ttl > 0) and (os.time() + ttl) or 0,
  }
end

--- Get a cached value, or compute and cache it
---@param key string
---@param fn function() -> any  computation function
---@param ttl? number time-to-live in seconds
---@return any
function M.get_or_set(key, fn, ttl)
  local value = M.get(key)
  if value ~= nil then
    return value
  end

  value = fn()
  M.set(key, value, ttl)
  return value
end

--- Delete a cached value
---@param key string
function M.del(key)
  M._store[key] = nil
end

--- Delete all entries matching a pattern
---@param pattern string Lua pattern
function M.del_pattern(pattern)
  local to_delete = {}
  for key, _ in pairs(M._store) do
    if key:match(pattern) then
      to_delete[#to_delete + 1] = key
    end
  end
  for _, key in ipairs(to_delete) do
    M._store[key] = nil
  end
end

--- Clear all cached values
function M.clear()
  M._store = {}
end

--- Remove expired entries
---@return number count of removed entries
function M.gc()
  local now = os.time()
  local removed = 0
  local to_delete = {}

  for key, entry in pairs(M._store) do
    if entry.expires > 0 and now > entry.expires then
      to_delete[#to_delete + 1] = key
      removed = removed + 1
    end
  end

  for _, key in ipairs(to_delete) do
    M._store[key] = nil
  end

  return removed
end

--- Get cache statistics
---@return { total: number, expired: number, active: number }
function M.stats()
  local total = 0
  local expired = 0
  local now = os.time()

  for _, entry in pairs(M._store) do
    total = total + 1
    if entry.expires > 0 and now > entry.expires then
      expired = expired + 1
    end
  end

  return {
    total = total,
    expired = expired,
    active = total - expired,
  }
end

return M
```

## Event Bus

### `lua/anvil/core/events.lua`

```lua
--- Anvil internal event bus
--- Decouples modules via publish/subscribe pattern
--- Uses Neovim's User autocmd under the hood
--- Depends on: nothing
---@class AnvilCoreEvents
local M = {}

---@type number
M._augroup = nil

--- Setup event system
function M.setup()
  M._augroup = vim.api.nvim_create_augroup("Anvil", { clear = true })
end

--- Emit an event
--- Creates a Neovim User autocmd event with pattern "Anvil:<event_name>"
---@param event string event name (e.g., "ready", "ai:chat:opened")
---@param data? table event data (passed via autocmd callback args)
function M.emit(event, data)
  vim.api.nvim_exec_autocmds("User", {
    pattern = "Anvil:" .. event,
    data = data,
  })
end

--- Subscribe to an event
---@param event string event name
---@param callback function(args: table) handler function
---@param opts? { once?: boolean, desc?: string }
---@return number autocmd id (for unsubscribing)
function M.on(event, callback, opts)
  opts = opts or {}
  return vim.api.nvim_create_autocmd("User", {
    group = M._augroup,
    pattern = "Anvil:" .. event,
    callback = function(args)
      callback(args)
    end,
    once = opts.once or false,
    desc = opts.desc or ("Anvil event: " .. event),
  })
end

--- Subscribe to an event (fires only once)
---@param event string event name
---@param callback function(args: table)
---@return number autocmd id
function M.once(event, callback)
  return M.on(event, callback, { once = true })
end

--- Unsubscribe from an event
---@param id number autocmd id returned by on()
function M.off(id)
  pcall(vim.api.nvim_del_autocmd, id)
end

return M
```

```lua
--- Apply options
---@param overrides? table user overrides in the same format
function M.apply(overrides)
  local config = vim.tbl_deep_extend("force", M.defaults, overrides or {})

  -- Apply vim.opt
  if config.opt then
    for key, value in pairs(config.opt) do
      local ok, err = pcall(function()
        vim.opt[key] = value
      end)
      if not ok then
        vim.notify(
          string.format("[Anvil] Failed to set option '%s': %s", key, err),
          vim.log.levels.WARN
        )
      end
    end
  end

  -- Apply vim.g
  if config.g then
    for key, value in pairs(config.g) do
      vim.g[key] = value
    end
  end
end

return M
```

## Autocommands

### `lua/anvil/core/autocmds.lua`

```lua
--- Anvil global autocommands
--- Depends on: core/events (optional, for emitting)
---@class AnvilCoreAutocmds
local M = {}

--- Setup global autocommands
function M.setup()
  local augroup = vim.api.nvim_create_augroup("AnvilAutocmds", { clear = true })

  -- ─── Highlight on yank ───
  vim.api.nvim_create_autocmd("TextYankPost", {
    group = augroup,
    desc = "Highlight yanked text",
    callback = function()
      vim.highlight.on_yank({ higroup = "IncSearch", timeout = 200 })
    end,
  })

  -- ─── Auto-resize splits ───
  vim.api.nvim_create_autocmd("VimResized", {
    group = augroup,
    desc = "Auto-resize panes on terminal resize",
    callback = function()
      vim.cmd("tabdo wincmd =")
    end,
  })

  -- ─── Trim trailing whitespace ───
  vim.api.nvim_create_autocmd("BufWritePre", {
    group = augroup,
    desc = "Trim trailing whitespace on save",
    callback = function()
      local save_cursor = vim.fn.getpos(".")
      vim.cmd([[%s/\s\+$//e]])
      vim.fn.setpos(".", save_cursor)
    end,
  })

  -- ─── .env file detection ───
  vim.api.nvim_create_autocmd({ "BufRead", "BufNewFile" }, {
    group = augroup,
    pattern = { ".env", ".env.*" },
    desc = "Set filetype for .env files",
    callback = function()
      vim.bo.filetype = "sh"
    end,
  })

  -- ─── Large file handling ───
  vim.api.nvim_create_autocmd("BufReadPre", {
    group = augroup,
    desc = "Disable heavy features for large files (>1MB)",
    callback = function(args)
      local ok, stats = pcall(vim.uv.fs_stat, vim.api.nvim_buf_get_name(args.buf))
      if ok and stats and stats.size > 1024 * 1024 then
        vim.b[args.buf].large_buf = true
        vim.opt_local.foldmethod = "manual"
        vim.opt_local.spell = false
        vim.opt_local.syntax = "off"
      end
    end,
  })

  -- ─── Return to last edit position ───
  vim.api.nvim_create_autocmd("BufReadPost", {
    group = augroup,
    desc = "Return to last edit position when opening file",
    callback = function(args)
      local mark = vim.api.nvim_buf_get_mark(args.buf, '"')
      local line_count = vim.api.nvim_buf_line_count(args.buf)
      if mark[1] > 0 and mark[1] <= line_count then
        pcall(vim.api.nvim_win_set_cursor, 0, mark)
      end
    end,
  })

  -- ─── Auto-create parent directories on save ───
  vim.api.nvim_create_autocmd("BufWritePre", {
    group = augroup,
    desc = "Auto-create parent directories when saving",
    callback = function(args)
      local path = vim.fn.fnamemodify(args.file, ":p:h")
      if vim.fn.isdirectory(path) == 0 then
        vim.fn.mkdir(path, "p")
      end
    end,
  })

  -- ─── Reload .env on directory change ───
  vim.api.nvim_create_autocmd("DirChanged", {
    group = augroup,
    desc = "Reload .env when changing directory",
    callback = function()
      local env_ok, env = pcall(require, "anvil.core.env")
      if env_ok then
        env.reload()
      end
    end,
  })

  -- ─── Custom filetypes ───
  vim.filetype.add({
    extension = {
      mdx = "mdx",
      env = "sh",
    },
    filename = {
      [".env"]             = "sh",
      [".env.local"]       = "sh",
      [".env.development"] = "sh",
      [".env.production"]  = "sh",
      [".prettierrc"]      = "json",
      [".eslintrc"]        = "json",
      [".babelrc"]         = "json",
      ["tsconfig.json"]    = "jsonc",
      [".swcrc"]           = "json",
    },
    pattern = {
      ["%.env%.[%w_.-]+"] = "sh",
    },
  })
end

return M
```

```lua
--- Anvil centralized keymap registration
--- Provides helpers for consistent keymap definition
--- Actual keymaps are defined in plugins/ via AstroCore opts.mappings
--- This module provides utilities and shared definitions
--- Depends on: nothing
---@class AnvilCoreKeys
local M = {}

--- Leader key definitions (for documentation and which-key)
---@type table<string, string>
M.groups = {
  ["<leader>a"]  = "󰚩 AI",
  ["<leader>A"]  = "󰚩 AI External",
  ["<leader>b"]  = " Buffers",
  ["<leader>d"]  = " Debug",
  ["<leader>f"]  = " Find",
  ["<leader>g"]  = "󰊢 Git",
  ["<leader>l"]  = " LSP",
  ["<leader>n"]  = " Packages",
  ["<leader>o"]  = " Overseer",
  ["<leader>s"]  = " Search/Replace",
  ["<leader>t"]  = "󰙨 Test",
  ["<leader>u"]  = " Toggle",
  ["<leader>x"]  = " Diagnostics",
}

--- Setup core keymaps (non-plugin, always available)
---@param user_keys? table user keymap overrides
function M.setup(user_keys)
  -- These are minimal keymaps that don't depend on any plugin
  -- Plugin-dependent keymaps go in plugins/*.lua via AstroCore

  local map = vim.keymap.set

  -- ─── Essential (no plugin dependency) ───

  map("n", "<C-h>", "<C-w>h", { desc = "Move to left window" })

  map("n", "<leader>ld", function() vim.diagnostic.open_float() end, { desc = "Line diagnostics" })
```

```lua
    to_file = false,
```

## Boot Sequence

### `lua/anvil/init.lua`

```lua
--- Anvil entry point
--- Orchestrates the boot sequence for all subsystems
---@class Anvil
local M = {}

M._version = "0.1.0"
M._ready = false

--- Boot Anvil
--- Called from the root init.lua: require("anvil").boot({...})
---@param opts? AnvilUserConfig
function M.boot(opts)
  if M._ready then return end

  -- ─── Phase 1: Core (synchronous, instant) ───
  -- Order matters: env → logger → cache → events → options → autocmds → keys

  local env = require("anvil.core.env")
  env.load()

  local config = require("anvil.config")
  M.config = config.apply(opts or {})

  local logger = require("anvil.core.logger")
  logger.setup(M.config.log)

  -- cache needs no setup (stateless module)

  local events = require("anvil.core.events")
  events.setup()

  local options = require("anvil.core.options")
  options.apply(M.config.options)

  require("anvil.core.autocmds").setup()
  require("anvil.core.keys").setup(M.config.keys)

  -- ─── Phase 2: Platform detection ───
  local platform_ok, platform = pcall(require, "anvil.platform")
  if platform_ok then
    platform.apply()
  end

  -- ─── Phase 3: Language registration ───
  local lang_ok, lang = pcall(require, "anvil.lang")
  if lang_ok then
    lang.setup(M.config.languages)
  end

  -- ─── Phase 4: AI (lazy — initialized on first use) ───
  -- Nothing to do here; anvil.ai.init loads on demand

  -- ─── Phase 5: Commands ───
  M._register_commands()

  -- ─── Done ───
  M._ready = true
  events.emit("ready", {
    version = M._version,
    languages = lang_ok and #lang.enabled() or 0,
  })

  logger.info("Anvil v%s ready (%d languages)",
    M._version,
    lang_ok and #lang.enabled() or 0
  )
end

--- Register user commands
function M._register_commands()
  local cmd = vim.api.nvim_create_user_command

  -- ─── Main hub ───
  cmd("Anvil", function(args)
    local sub = args.fargs[1]
    if not sub or sub == "" then
      M._show_menu()
    elseif sub == "health" then
      vim.cmd("checkhealth anvil")
    elseif sub == "version" then
      vim.notify("Anvil v" .. M._version, vim.log.levels.INFO)
    elseif sub == "config" then
      require("anvil.ai.providers").show_config()
    elseif sub == "lang" then
      local lang = require("anvil.lang")
      local info = args.fargs[2]
      if info then
        lang.info(info)
      else
        vim.notify(
          "Languages: " .. table.concat(lang.enabled(), ", "),
          vim.log.levels.INFO
        )
      end
    elseif sub == "index" then
      require("anvil.ai.context").index_project()
    elseif sub == "cache" then
      local cache = require("anvil.core.cache")
      local stats = cache.stats()
      vim.notify(
        string.format("Cache: %d active, %d expired, %d total",
          stats.active, stats.expired, stats.total),
        vim.log.levels.INFO
      )
    else
      vim.notify("Unknown subcommand: " .. sub, vim.log.levels.WARN)
    end
  end, {
    nargs = "*",
    complete = function()
      return {
        "health", "version", "config", "lang", "index", "cache",
      }
    end,
    desc = "Anvil: Main command hub",
  })

  -- ─── AI commands (delegate to ai.init) ───
  cmd("AnvilChat", function(a)
    require("anvil.ai").chat(a)
  end, { nargs = "?", range = true, desc = "Anvil: Open AI chat" })

  cmd("AnvilAsk", function(a)
    require("anvil.ai").ask(a.args)
  end, { nargs = "+", desc = "Anvil: Ask AI inline" })

  cmd("AnvilIndex", function()
    require("anvil.ai").index_project()
  end, { desc = "Anvil: Index current project" })

  cmd("AnvilCompletion", function()
    require("anvil.ai").toggle_completion()
  end, { desc = "Anvil: Toggle AI inline completions" })
end

--- Show main menu (simple version — can be enhanced with telescope later)
function M._show_menu()
  local items = {
    "chat     — Open AI chat",
    "health   — Run health check",
    "version  — Show version",
    "config   — Show AI provider config",
    "lang     — Show enabled languages",
    "index    — Index current project",
    "cache    — Show cache statistics",
  }

  vim.ui.select(items, { prompt = "Anvil:" }, function(choice)
    if not choice then return end
    local sub = choice:match("^(%S+)")
    if sub then
      vim.cmd("Anvil " .. sub)
    end
  end)
end

return M
```

```lua
        if pack.dap then parts[#parts + 1] = "dap:" .. pack.dap.adapter end

    vim.health.ok(string.format("Neovim %d.%d.%d", v.major, v.minor, v.patch))
```
