# Anvil — Lang: Python

Python language pack with dual LSP support (basedpyright + ruff).

## Summary

| Feature | Tool |
|---|---|
| LSP | `basedpyright` (types) + `ruff` (lint/format actions) |
| Treesitter | `python`, `toml`, `requirements` |
| Formatter | `ruff` |
| Linter | `ruff` (via LSP, not nvim-lint) |
| DAP | `debugpy` |
| Test | `neotest-python` |

## Design Decisions

- **basedpyright** — community fork of pyright with better defaults and additional features. Primary LSP for type checking, completions, and go-to-definition.
- **ruff** — replaces black + flake8 + isort (ADR-015). 10-100x faster (written in Rust). Runs as a second LSP server for code actions (fix imports, fix lint errors).
- **Dual LSP** (ADR-013) — basedpyright handles types, ruff handles lint code actions. Capability conflicts resolved in `on_attach`: ruff's hover is disabled so basedpyright's hover is used.
- **debugpy** — Microsoft's Python debug adapter. Supports Django, Flask, pytest, and remote debugging.
- **neotest-python** — supports pytest and unittest. Configured for pytest by default.

## File

### `lua/anvil/lang/python.lua`

<!-- file: lua/anvil/lang/python.lua -->
```lua
--- Anvil language pack: Python
--- Dual LSP: basedpyright (types) + ruff (lint/format)
---@type AnvilLangPack
local M = {
  name = "python",
  filetypes = { "python" },

  treesitter = { "python", "toml", "requirements" },

  lsp = {
    server = "basedpyright",
    settings = {
      basedpyright = {
        analysis = {
          autoSearchPaths = true,
          useLibraryCodeForTypes = true,
          diagnosticMode = "openFilesOnly",
          typeCheckingMode = "standard",
        },
      },
    },
    extra_servers = {
      {
        server = "ruff",
        on_attach = function(client, _)
          -- Disable hover — basedpyright handles it
          client.server_capabilities.hoverProvider = false
        end,
      },
    },
  },

  formatter = {
    tool = "ruff",
    -- ruff format is invoked via conform; ruff LSP provides code actions
  },

  -- Linting is handled by ruff LSP, not nvim-lint
  -- linter = nil,

  dap = {
    adapter = "python",
    adapter_config = {
      type = "executable",
      command = "python",
      args = { "-m", "debugpy.adapter" },
    },
    configurations = {
      {
        type = "python",
        request = "launch",
        name = "Launch file",
        program = "${file}",
        pythonPath = function()
          -- Use virtualenv if available
          local venv = os.getenv("VIRTUAL_ENV")
          if venv then
            return venv .. "/bin/python"
          end
          local cwd = vim.fn.getcwd()
          if vim.fn.executable(cwd .. "/.venv/bin/python") == 1 then
            return cwd .. "/.venv/bin/python"
          end
          return "python3"
        end,
      },
      {
        type = "python",
        request = "launch",
        name = "Launch with arguments",
        program = "${file}",
        args = function()
          local input = vim.fn.input("Arguments: ")
          return vim.split(input, " ", { trimempty = true })
        end,
      },
      {
        type = "python",
        request = "launch",
        name = "Django",
        program = "${workspaceFolder}/manage.py",
        args = { "runserver", "--noreload" },
        django = true,
      },
      {
        type = "python",
        request = "launch",
        name = "pytest",
        module = "pytest",
        args = { "-xvs" },
      },
    },
    dependencies = { "debugpy" },
  },

  test = {
    adapter = "neotest-python",
    config = {
      runner = "pytest",
      python = function()
        local venv = os.getenv("VIRTUAL_ENV")
        if venv then
          return venv .. "/bin/python"
        end
        local cwd = vim.fn.getcwd()
        if vim.fn.executable(cwd .. "/.venv/bin/python") == 1 then
          return cwd .. "/.venv/bin/python"
        end
        return "python3"
      end,
    },
  },

  tasks = {
    {
      name = "Python: Run current file",
      cmd = { "python3", vim.fn.expand("%:p") },
      lang = "python",
      filetypes = { "python" },
    },
    {
      name = "Python: pytest",
      cmd = "python3 -m pytest -xvs",
      lang = "python",
      filetypes = { "python" },
    },
    {
      name = "Python: pytest (current file)",
      cmd = { "python3", "-m", "pytest", "-xvs", vim.fn.expand("%:p") },
      lang = "python",
      filetypes = { "python" },
    },
    {
      name = "Python: pip install -e .",
      cmd = "pip install -e .",
      lang = "python",
      filetypes = { "python" },
    },
    {
      name = "Python: ruff check",
      cmd = "ruff check .",
      lang = "python",
      filetypes = { "python" },
    },
  },

  ai = {
    system_prompt = "This is a Python project. Use modern Python 3.10+ syntax. "
      .. "Use type hints everywhere. Prefer pathlib over os.path. "
      .. "Use ruff for linting/formatting. Follow PEP 8 conventions.",
    file_patterns = {
      "pyproject.toml", "setup.py", "setup.cfg",
      "requirements.txt", "requirements*.txt",
      "ruff.toml", ".python-version", "Pipfile",
    },
    conventions = "snake_case for functions/variables, PascalCase for classes. "
      .. "Use dataclasses or Pydantic for data structures. "
      .. "Use pytest for testing. Type hints on all public functions.",
  },

  keys = {
    {
      mode = "n",
      lhs = "<leader>lv",
      rhs = function()
        local venv = os.getenv("VIRTUAL_ENV")
        if venv then
          vim.notify("Venv: " .. venv, vim.log.levels.INFO)
        else
          vim.notify("No virtualenv active", vim.log.levels.WARN)
        end
      end,
      desc = "Python: Show virtualenv",
    },
  },
}

return M
```
