# Anvil — Lang: Rust

Rust language pack using rustaceanvim for full rust-analyzer integration.

## Summary

| Feature | Tool |
|---|---|
| LSP | `rust_analyzer` (managed by rustaceanvim) |
| Treesitter | `rust`, `toml` |
| Formatter | `rustfmt` (managed by LSP) |
| Linter | `clippy` (via LSP) |
| DAP | `codelldb` |
| Test | `neotest-rust` |
| Extra plugins | `rustaceanvim`, `crates.nvim` |

## Design Decisions

- **rustaceanvim** — successor to rust-tools.nvim (ADR-012). Manages rust-analyzer directly (not via lspconfig), hence `managed_by_plugin = true`. Configuration via `vim.g.rustaceanvim`.
- **rustfmt** — formatting is handled by rust-analyzer (LSP), hence `managed_by_lsp = true`. No conform registration needed.
- **clippy** — linting is done by rust-analyzer with `check.command = "clippy"`. No separate linter needed.
- **codelldb** — LLDB-based debug adapter. Supports Rust-specific pretty-printing of types.
- **crates.nvim** — inline Cargo.toml crate version info and update actions.

### `lua/anvil/lang/rust.lua`

```lua
--- Anvil language pack: Rust
--- Uses rustaceanvim for full rust-analyzer integration
---@type AnvilLangPack
local M = {
  name = "rust",
  filetypes = { "rust" },

  treesitter = { "rust", "toml" },

  lsp = {
    server = "rust_analyzer",
    managed_by_plugin = true, -- rustaceanvim manages rust-analyzer
    settings = {
      ["rust-analyzer"] = {
        cargo = {
          allFeatures = true,
          loadOutDirsFromCheck = true,
          buildScripts = { enable = true },
        },
        checkOnSave = true,
        check = {
          command = "clippy",
          extraArgs = { "--no-deps" },
        },
        procMacro = { enable = true },
        inlayHints = {
          bindingModeHints = { enable = false },
          chainingHints = { enable = true },
          closingBraceHints = { enable = true, minLines = 25 },
          closureReturnTypeHints = { enable = "with_block" },
          lifetimeElisionHints = { enable = "never" },
          maxLength = 25,
          parameterHints = { enable = true },
          reborrowHints = { enable = "never" },
          renderColons = true,
          typeHints = {
            enable = true,
            hideClosureInitialization = false,
            hideNamedConstructor = false,
          },
        },
      },
    },
  },

  formatter = {
    tool = "rustfmt",
    managed_by_lsp = true, -- rust-analyzer handles formatting
  },

  -- Clippy runs via rust-analyzer check, not as a separate linter
  -- linter = nil,

  dap = {
    adapter = "codelldb",
    adapter_config = {
      type = "server",
      port = "${port}",
      executable = {
        command = "codelldb",
        args = { "--port", "${port}" },
      },
    },
    configurations = {
      {
        type = "codelldb",
        request = "launch",
        name = "Launch (cargo build)",
        program = function()
          -- Build first, then find the binary
          vim.fn.system("cargo build 2>&1")
          local metadata = vim.fn.system("cargo metadata --format-version 1 --no-deps 2>/dev/null")
          local ok, decoded = pcall(vim.json.decode, metadata)
          if ok and decoded.packages and #decoded.packages > 0 then
            local target_dir = decoded.target_directory or "target"
            local name = decoded.packages[1].name
            return target_dir .. "/debug/" .. name
          end
          return vim.fn.input("Binary path: ", vim.fn.getcwd() .. "/target/debug/", "file")
        end,
        cwd = "${workspaceFolder}",
        stopOnEntry = false,
      },
      {
        type = "codelldb",
        request = "launch",
        name = "Launch (select binary)",
        program = function()
          return vim.fn.input("Binary path: ", vim.fn.getcwd() .. "/target/debug/", "file")
        end,
        cwd = "${workspaceFolder}",
        stopOnEntry = false,
      },
    },
    dependencies = { "codelldb" },
  },

  test = {
    adapter = "neotest-rust",
    config = {
      args = { "--no-capture" },
    },
  },

  tasks = {
    {
      name = "Rust: Build",
      cmd = "cargo build",
      lang = "rust",
      filetypes = { "rust" },
    },
    {
      name = "Rust: Build (release)",
      cmd = "cargo build --release",
      lang = "rust",
      filetypes = { "rust" },
    },
    {
      name = "Rust: Run",
      cmd = "cargo run",
      lang = "rust",
      filetypes = { "rust" },
    },
    {
      name = "Rust: Test",
      cmd = "cargo test",
      lang = "rust",
      filetypes = { "rust" },
    },
    {
      name = "Rust: Clippy",
      cmd = "cargo clippy --all-targets --all-features",
      lang = "rust",
      filetypes = { "rust" },
    },
    {
      name = "Rust: Check",
      cmd = "cargo check",
      lang = "rust",
      filetypes = { "rust" },
    },
    {
      name = "Rust: Doc",
      cmd = "cargo doc --open",
      lang = "rust",
      filetypes = { "rust" },
    },
  },

  ai = {
    system_prompt = "This is a Rust project. Follow Rust idioms: "
      .. "use Result<T, E> for error handling, avoid unwrap() in production code, "
      .. "prefer iterators over manual loops, use derive macros. "
      .. "Follow clippy recommendations.",
    file_patterns = { "Cargo.toml", "Cargo.lock", "rust-toolchain.toml", ".cargo/config.toml" },
    conventions = "snake_case for functions/variables, PascalCase for types/traits, "
      .. "SCREAMING_SNAKE_CASE for constants. Use '?' operator for error propagation. "
      .. "Document public items with /// doc comments.",
  },

  keys = {
    {
      mode = "n",
      lhs = "<leader>lc",
      rhs = function()
        vim.cmd("RustLsp openCargo")
      end,
      desc = "Rust: Open Cargo.toml",
    },
    {
      mode = "n",
      lhs = "<leader>le",
      rhs = function()
        vim.cmd("RustLsp explainError")
      end,
      desc = "Rust: Explain error",
    },
    {
      mode = "n",
      lhs = "<leader>lE",
      rhs = function()
        vim.cmd("RustLsp renderDiagnostic")
      end,
      desc = "Rust: Render diagnostic",
    },
    {
      mode = "n",
      lhs = "<leader>lp",
      rhs = function()
        vim.cmd("RustLsp parentModule")
      end,
      desc = "Rust: Parent module",
    },
    {
      mode = "n",
      lhs = "<leader>lj",
      rhs = function()
        vim.cmd("RustLsp joinLines")
      end,
      desc = "Rust: Join lines",
    },
    {
      mode = "n",
      lhs = "<leader>lD",
      rhs = function()
        vim.cmd("RustLsp debuggables")
      end,
      desc = "Rust: Debuggables",
    },
    {
      mode = "n",
      lhs = "<leader>lR",
      rhs = function()
        vim.cmd("RustLsp runnables")
      end,
      desc = "Rust: Runnables",
    },
  },

  plugins = {
    {
      "mrcjkb/rustaceanvim",
      version = "^5",
      lazy = false,
      ft = { "rust" },
      init = function()
        local pack = require("anvil.lang.rust")
        vim.g.rustaceanvim = {
          tools = {
            hover_actions = { replace_builtin_hover = false },
          },
          server = {
            default_settings = pack.lsp.settings,
            on_attach = function(_, bufnr)
              -- Apply language-specific keymaps
              local lang_ok, lang = pcall(require, "anvil.lang")
              if lang_ok then
                lang.apply_keys(bufnr)
              end
            end,
          },
          dap = {
            adapter = pack.dap.adapter_config,
          },
        }
      end,
    },
    {
      "saecki/crates.nvim",
      event = { "BufRead Cargo.toml" },
      dependencies = { "nvim-lua/plenary.nvim" },
      opts = {
        completion = { cmp = { enabled = true } },
      },
    },
  },
}

return M
```

| `<leader>le` | n | Explain error (rustaceanvim) |
| `<leader>lE` | n | Render diagnostic |
| `<leader>lp` | n | Parent module |
| `<leader>lj` | n | Join lines |
| `<leader>lD` | n | List debuggables |
| `<leader>lR` | n | List runnables |
