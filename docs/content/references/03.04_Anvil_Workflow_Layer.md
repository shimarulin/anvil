# Anvil — Workflow Layer

Task management, terminals, sessions, project management, and file marks.

## Architecture

```text
lua/anvil/workflow/
├── init.lua       # workflow subsystem setup
├── tasks.lua      # overseer task configuration
├── terminal.lua   # terminal profiles
├── project.lua    # project detection and switching
├── session.lua    # session management
├── marks.lua      # file marks (harpoon is in editor layer)
└── notes.lua      # project-local notes

lua/plugins/
└── workflow.lua   # workflow plugin specs
```

## Workflow Subsystem Entry Point

### `lua/anvil/workflow/init.lua`

<!-- file: lua/anvil/workflow/init.lua -->
```lua
--- Anvil workflow subsystem setup
--- Coordinates tasks, terminals, sessions, projects
--- Depends on: core/events (optional)
---@class AnvilWorkflow
local M = {}

--- Setup workflow subsystem
function M.setup()
  local events_ok, events = pcall(require, "anvil.core.events")
  if events_ok then
    events.emit("workflow:ready")
  end
end

return M
```

## Task Management

### `lua/anvil/workflow/tasks.lua`

<!-- file: lua/anvil/workflow/tasks.lua -->
```lua
--- Anvil task management (overseer.nvim)
--- Tasks come from language packs + auto-detection
---@class AnvilWorkflowTasks
local M = {}

--- Get overseer opts
---@return table
function M.opts()
  return {
    strategy = "toggleterm",
    task_list = {
      direction = "bottom",
      min_height = 15,
      max_height = 25,
      default_detail = 1,
      bindings = {
        ["<C-l>"] = false,
        ["<C-h>"] = false,
      },
    },
    form = { border = "rounded", win_opts = { winblend = 0 } },
    confirm = { border = "rounded", win_opts = { winblend = 0 } },
    task_win = { border = "rounded" },
  }
end

--- Register language pack tasks with overseer
--- Called after overseer is loaded
function M.register_lang_tasks()
  local lang_ok, lang = pcall(require, "anvil.lang")
  if not lang_ok then return end

  local overseer_ok, overseer = pcall(require, "overseer")
  if not overseer_ok then return end

  local tasks = lang.tasks()
  for _, task in ipairs(tasks) do
    overseer.register_template({
      name = task.name,
      builder = function()
        return {
          cmd  = type(task.cmd) == "table" and task.cmd or { task.cmd },
          name = task.name,
          cwd  = vim.fn.getcwd(),
        }
      end,
      desc = task.name .. " (" .. task.lang .. ")",
      condition = {
        filetype = task.filetypes,
      },
    })
  end
end

return M
```

## Terminal Profiles

### `lua/anvil/workflow/terminal.lua`

<!-- file: lua/anvil/workflow/terminal.lua -->
```lua
--- Anvil terminal profiles
--- Predefined terminal configurations for common tasks
--- Depends on: toggleterm (lazy, only when called)
---@class AnvilWorkflowTerminal
local M = {}

---@type table<string, table>
M.profiles = {}

--- Register built-in terminal profiles
function M.setup()
  M.profiles = {
    default = {
      direction = "horizontal",
      size = 20,
    },
    vertical = {
      direction = "vertical",
      size = function() return math.floor(vim.o.columns * 0.4) end,
    },
    float = {
      direction = "float",
      float_opts = {
        border = "curved",
        width = function() return math.floor(vim.o.columns * 0.85) end,
        height = function() return math.floor(vim.o.lines * 0.85) end,
      },
    },
    bottom = {
      direction = "horizontal",
      size = 15,
    },
  }
end

--- Open a terminal with a named profile
---@param profile_name? string profile name (default: "default")
---@param cmd? string command to run
function M.open(profile_name, cmd)
  profile_name = profile_name or "default"
  local profile = M.profiles[profile_name] or M.profiles.default

  local Terminal = require("toggleterm.terminal").Terminal
  local opts = vim.tbl_extend("force", {
    close_on_exit = false,
    start_in_insert = true,
  }, profile)

  if cmd then
    opts.cmd = cmd
  end

  Terminal:new(opts):toggle()
end

--- Open a lazygit terminal
function M.lazygit()
  M.open("float", "lazygit")
end

--- Open a bottom terminal
function M.bottom()
  M.open("bottom")
end

return M
```

```lua
--- Anvil session management
--- Auto-save and restore sessions per project
---@class AnvilWorkflowSession
local M = {}

--- Get persistence.nvim opts
---@return table
function M.opts()
  return {
    dir = vim.fn.stdpath("state") .. "/sessions/",
    options = {
      "buffers", "curdir", "tabpages", "winsize",
      "help", "globals", "skiprtp",
    },
    pre_save = function()
      -- Close ephemeral windows before saving
      for _, win in ipairs(vim.api.nvim_list_wins()) do
        local buf = vim.api.nvim_win_get_buf(win)
        local bt = vim.bo[buf].buftype
        if bt == "nofile" or bt == "terminal" or bt == "prompt" then
          pcall(vim.api.nvim_win_close, win, true)
        end
      end
    end,
  }
end

return M
```

## Project Detection

### `lua/anvil/workflow/project.lua`

<!-- file: lua/anvil/workflow/project.lua -->
```lua
--- Anvil project detection
--- Finds project root via markers
---@class AnvilWorkflowProject
local M = {}

--- Root markers in priority order
---@type string[]
M.markers = {
  ".git",
  "Cargo.toml",
  "go.mod",
  "package.json",
  "pyproject.toml",
  "Makefile",
  "CMakeLists.txt",
  ".editorconfig",
}

--- Find project root from a given path
---@param path? string starting path (default: current file or cwd)
---@return string root project root path
function M.find_root(path)
  path = path or vim.fn.expand("%:p:h")
  if path == "" then path = vim.fn.getcwd() end

  -- Use vim.fs.find (Neovim 0.10+)
  local found = vim.fs.find(M.markers, {
    path = path,
    upward = true,
    stop = vim.uv.os_homedir(),
  })

  if #found > 0 then
    return vim.fn.fnamemodify(found[1], ":h")
  end

  return vim.fn.getcwd()
end

--- Get project name from root
---@param root? string project root
---@return string name
function M.name(root)
  root = root or M.find_root()
  return vim.fn.fnamemodify(root, ":t")
end

--- Get project type based on marker files
---@param root? string project root
---@return string type e.g. "rust", "node", "python", "go", "unknown"
function M.type(root)
  root = root or M.find_root()

  local checks = {
    { "Cargo.toml",    "rust" },
    { "go.mod",        "go" },
    { "package.json",  "node" },
    { "pyproject.toml", "python" },
    { "setup.py",      "python" },
    { "CMakeLists.txt", "cmake" },
    { "Makefile",      "make" },
  }

  for _, check in ipairs(checks) do
    if vim.fn.filereadable(root .. "/" .. check[1]) == 1 then
      return check[2]
    end
  end

  return "unknown"
end

return M
```

## File Marks

### `lua/anvil/workflow/marks.lua`

<!-- file: lua/anvil/workflow/marks.lua -->
```lua
--- Anvil file marks
--- Simple per-project file bookmarks (complements harpoon)
--- Depends on: core/cache
---@class AnvilWorkflowMarks
local M = {}

local cache = require("anvil.core.cache")

--- Get cache key for current project
---@return string
local function cache_key()
  local project_ok, project = pcall(require, "anvil.workflow.project")
  local root = project_ok and project.find_root() or vim.fn.getcwd()
  return "marks:" .. root
end

--- Get marked files for current project
---@return string[]
function M.list()
  return cache.get(cache_key()) or {}
end

--- Add current file to marks
function M.add()
  local file = vim.fn.expand("%:p")
  if file == "" then return end

  local marks = M.list()
  -- Don't add duplicates
  for _, m in ipairs(marks) do
    if m == file then
      vim.notify("Already marked: " .. vim.fn.fnamemodify(file, ":~:."), vim.log.levels.INFO)
      return
    end
  end

  marks[#marks + 1] = file
  cache.set(cache_key(), marks, 0) -- No TTL
  vim.notify("Marked: " .. vim.fn.fnamemodify(file, ":~:."), vim.log.levels.INFO)
end

--- Remove current file from marks
function M.remove()
  local file = vim.fn.expand("%:p")
  local marks = M.list()
  local new_marks = {}
  local found = false

  for _, m in ipairs(marks) do
    if m == file then
      found = true
    else
      new_marks[#new_marks + 1] = m
    end
  end

  if found then
    cache.set(cache_key(), new_marks, 0)
    vim.notify("Unmarked: " .. vim.fn.fnamemodify(file, ":~:."), vim.log.levels.INFO)
  end
end

--- Pick a marked file to open
function M.pick()
  local marks = M.list()
  if #marks == 0 then
    vim.notify("No marks for this project", vim.log.levels.INFO)
    return
  end

  local items = {}
  for _, m in ipairs(marks) do
    items[#items + 1] = vim.fn.fnamemodify(m, ":~:.")
  end

  vim.ui.select(items, { prompt = "Open marked file:" }, function(_, idx)
    if idx then
      vim.cmd("edit " .. marks[idx])
    end
  end)
end

return M
```

## Project-Local Notes

### `lua/anvil/workflow/notes.lua`

<!-- file: lua/anvil/workflow/notes.lua -->
```lua
--- Anvil project-local notes
--- Quick scratch notes tied to the current project
--- Depends on: workflow/project
---@class AnvilWorkflowNotes
local M = {}

--- Get the notes directory for the current project
---@return string
function M.notes_dir()
  local project_ok, project = pcall(require, "anvil.workflow.project")
  local root = project_ok and project.find_root() or vim.fn.getcwd()
  return root .. "/.anvil-notes"
end

--- Get the path to the main project note
---@return string
function M.main_note_path()
  return M.notes_dir() .. "/notes.md"
end

--- Open the project notes file
--- Creates it if it doesn't exist
function M.open()
  local dir = M.notes_dir()
  local path = M.main_note_path()

  if vim.fn.isdirectory(dir) == 0 then
    vim.fn.mkdir(dir, "p")
  end

  if vim.fn.filereadable(path) ~= 1 then
    local project_ok, project = pcall(require, "anvil.workflow.project")
    local name = project_ok and project.name() or "Project"
    vim.fn.writefile({
      "# " .. name .. " — Notes",
      "",
      "## TODO",
      "",
      "- ",
      "",
      "## Notes",
      "",
    }, path)
  end

  vim.cmd("edit " .. path)
end

--- Open a named note
---@param name string note name (without extension)
function M.open_named(name)
  local dir = M.notes_dir()
  if vim.fn.isdirectory(dir) == 0 then
    vim.fn.mkdir(dir, "p")
  end
  vim.cmd("edit " .. dir .. "/" .. name .. ".md")
end

--- List all notes for the current project
function M.list()
  local dir = M.notes_dir()
  if vim.fn.isdirectory(dir) == 0 then
    vim.notify("No notes directory", vim.log.levels.INFO)
    return
  end

  local files = vim.fn.glob(dir .. "/*.md", false, true)
  if #files == 0 then
    vim.notify("No notes found", vim.log.levels.INFO)
    return
  end

  local items = {}
  for _, f in ipairs(files) do
    items[#items + 1] = vim.fn.fnamemodify(f, ":t:r")
  end

  vim.ui.select(items, { prompt = "Open note:" }, function(choice)
    if choice then
      M.open_named(choice)
    end
  end)
end

return M
```

```lua
--- Anvil workflow plugin specs
---@type LazySpec
return {
  -- ╭──────────────────────────────────────────────────────────╮
  -- │ Overseer — task runner                                   │
  -- ╰──────────────────────────────────────────────────────────╯
  {
    "stevearc/overseer.nvim",
    cmd = {
      "OverseerRun", "OverseerToggle", "OverseerInfo",
      "OverseerBuild", "OverseerQuickAction",
    },
    opts = function()
      return require("anvil.workflow.tasks").opts()
    end,
    config = function(_, opts)
      require("overseer").setup(opts)
      -- Register language pack tasks after overseer loads
      vim.defer_fn(function()
        require("anvil.workflow.tasks").register_lang_tasks()
      end, 100)
    end,
    keys = {
      { "<leader>or", "<cmd>OverseerRun<cr>",         desc = "Run task" },
      { "<leader>ot", "<cmd>OverseerToggle<cr>",       desc = "Toggle task list" },
      { "<leader>ob", "<cmd>OverseerBuild<cr>",        desc = "Build task" },
      { "<leader>oq", "<cmd>OverseerQuickAction<cr>",  desc = "Quick action" },
      { "<leader>oi", "<cmd>OverseerInfo<cr>",         desc = "Overseer info" },
    },
  },

  -- ╭──────────────────────────────────────────────────────────╮
  -- │ persistence.nvim — session management                    │
  -- ╰──────────────────────────────────────────────────────────╯
  {
    "folke/persistence.nvim",
    event = "BufReadPre",
    opts = function()
      return require("anvil.workflow.session").opts()
    end,
    keys = {
      { "<leader>qs", function() require("persistence").load() end,                desc = "Restore session" },
      { "<leader>ql", function() require("persistence").load({ last = true }) end, desc = "Restore last session" },
      { "<leader>qd", function() require("persistence").stop() end,                desc = "Don't save session" },
    },
  },

  -- ╭──────────────────────────────────────────────────────────╮
  -- │ project.nvim — project detection + telescope             │
  -- ╰──────────────────────────────────────────────────────────╯
  {
    "ahmedkhalf/project.nvim",
    event = "VeryLazy",
    config = function()
      require("project_nvim").setup({
        detection_methods = { "pattern", "lsp" },
        patterns = require("anvil.workflow.project").markers,
        show_hidden = true,
        silent_chdir = true,
        scope_chdir = "global",
      })
      -- Register telescope extension
      pcall(function()
        require("telescope").load_extension("projects")
      end)
    end,
    keys = {
      { "<leader>fp", "<cmd>Telescope projects<cr>", desc = "Projects" },
    },
  },

  -- ╭──────────────────────────────────────────────────────────╮
  -- │ which-key groups                                         │
  -- ╰──────────────────────────────────────────────────────────╯
  {
    "folke/which-key.nvim",
    optional = true,
    opts = {
      spec = {
        { "<leader>o", group = "Overseer" },
        { "<leader>q", group = "Session" },
      },
    },
  },
}
```
