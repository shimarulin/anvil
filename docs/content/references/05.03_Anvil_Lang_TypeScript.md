# Anvil — Lang: TypeScript

TypeScript and JavaScript language pack. Covers `.ts`, `.tsx`, `.js`, `.jsx` filetypes.

## Summary

| Feature | Tool |
|---|---|
| LSP | `typescript-tools.nvim` (managed by plugin) |
| Treesitter | `typescript`, `tsx`, `javascript`, `jsdoc` |
| Formatter | `prettierd` |
| Linter | — (ESLint via LSP) |
| DAP | `js-debug-adapter` |
| Test | `neotest-vitest` |
| Extra plugins | `typescript-tools.nvim` |

## Design Decisions

- **typescript-tools.nvim** — provides extra commands (organize imports, rename file, go to source definition) beyond what plain tsserver offers (ADR-014). Manages the LSP server directly, hence `managed_by_plugin = true`.
- **Critical rule**: never run typescript-tools AND vtsls simultaneously.
- **prettierd** — daemon-mode Prettier for fast formatting. Formatting is NOT done by the LSP (ts_ls formatting is disabled in `editor/lsp.lua`).
- **No standalone linter** — ESLint runs as an LSP server via `vscode-eslint-language-server`. If the project has an ESLint config, it activates automatically.
- **js-debug-adapter** — Microsoft's debug adapter for Node.js, Chrome, and Edge.
- **neotest-vitest** — Vitest is the modern test runner for Vite-based projects. For Jest projects, swap the adapter.

### `lua/anvil/lang/typescript.lua`

<!-- file: lua/anvil/lang/typescript.lua -->
```lua
--- Anvil language pack: TypeScript
--- Covers TypeScript, TSX, JavaScript, JSX
---@type AnvilLangPack
local M = {
  name = "typescript",
  filetypes = { "typescript", "typescriptreact", "javascript", "javascriptreact" },

  treesitter = { "typescript", "tsx", "javascript", "jsdoc" },

  lsp = {
    server = "typescript-tools",
    managed_by_plugin = true, -- typescript-tools.nvim manages the server
    settings = {
      -- Settings passed to typescript-tools.nvim setup
      tsserver_file_preferences = {
        includeInlayParameterNameHints = "all",
        includeInlayParameterNameHintsWhenArgumentMatchesName = false,
        includeInlayFunctionParameterTypeHints = true,
        includeInlayVariableTypeHints = true,
        includeInlayPropertyDeclarationTypeHints = true,
        includeInlayFunctionLikeReturnTypeHints = true,
        includeInlayEnumMemberValueHints = true,
      },
      tsserver_format_options = {
        allowIncompleteCompletions = false,
        allowRenameOfImportPath = false,
      },
    },
  },

  formatter = {
    tool = "prettierd",
  },

  -- ESLint runs as LSP, not as nvim-lint linter
  -- linter = nil,

  dap = {
    adapter = "pwa-node",
    adapter_config = {
      type = "server",
      host = "localhost",
      port = "${port}",
      executable = {
        command = "js-debug-adapter",
        args = { "${port}" },
      },
    },
    configurations = {
      {
        type = "pwa-node",
        request = "launch",
        name = "Launch file",
        program = "${file}",
        cwd = "${workspaceFolder}",
      },
      {
        type = "pwa-node",
        request = "attach",
        name = "Attach to process",
        processId = function()
          return require("dap.utils").pick_process()
        end,
        cwd = "${workspaceFolder}",
      },
    },
    dependencies = { "js-debug-adapter" },
  },

  test = {
    adapter = "neotest-vitest",
    config = {},
  },

  tasks = {
    {
      name = "TS: Dev server",
      cmd = "npm run dev",
      lang = "typescript",
      filetypes = { "typescript", "typescriptreact", "javascript", "javascriptreact" },
    },
    {
      name = "TS: Build",
      cmd = "npm run build",
      lang = "typescript",
      filetypes = { "typescript", "typescriptreact", "javascript", "javascriptreact" },
    },
    {
      name = "TS: Test",
      cmd = "npm test",
      lang = "typescript",
      filetypes = { "typescript", "typescriptreact", "javascript", "javascriptreact" },
    },
    {
      name = "TS: Lint",
      cmd = "npm run lint",
      lang = "typescript",
      filetypes = { "typescript", "typescriptreact", "javascript", "javascriptreact" },
    },
    {
      name = "TS: Type check",
      cmd = "npx tsc --noEmit",
      lang = "typescript",
      filetypes = { "typescript", "typescriptreact" },
    },
  },

  ai = {
    system_prompt = "This is a TypeScript/JavaScript project. "
      .. "Use modern ES2022+ syntax, strict TypeScript types, and async/await. "
      .. "Prefer named exports. Use Prettier formatting conventions.",
    file_patterns = {
      "package.json", "tsconfig.json", "tsconfig.*.json",
      ".eslintrc.*", "eslint.config.*", ".prettierrc*",
      "vite.config.*", "next.config.*", "vitest.config.*",
    },
    conventions = "camelCase for variables/functions, PascalCase for types/components. "
      .. "Prefer interfaces over type aliases for object shapes. "
      .. "Use 'const' by default, 'let' when reassignment is needed.",
  },

  keys = {
    {
      mode = "n",
      lhs = "<leader>lo",
      rhs = "<cmd>TSToolsOrganizeImports<cr>",
      desc = "TS: Organize imports",
    },
    {
      mode = "n",
      lhs = "<leader>lR",
      rhs = "<cmd>TSToolsRenameFile<cr>",
      desc = "TS: Rename file",
    },
    {
      mode = "n",
      lhs = "<leader>lm",
      rhs = "<cmd>TSToolsAddMissingImports<cr>",
      desc = "TS: Add missing imports",
    },
    {
      mode = "n",
      lhs = "<leader>lu",
      rhs = "<cmd>TSToolsRemoveUnused<cr>",
      desc = "TS: Remove unused imports",
    },
    {
      mode = "n",
      lhs = "<leader>lD",
      rhs = "<cmd>TSToolsGoToSourceDefinition<cr>",
      desc = "TS: Go to source definition",
    },
  },

  plugins = {
    {
      "pmizio/typescript-tools.nvim",
      dependencies = { "nvim-lua/plenary.nvim", "neovim/nvim-lspconfig" },
      ft = { "typescript", "typescriptreact", "javascript", "javascriptreact" },
      opts = function()
        local pack = require("anvil.lang.typescript")
        return {
          settings = pack.lsp.settings,
          on_attach = function(client, _)
            -- Disable formatting — prettierd handles it
            client.server_capabilities.documentFormattingProvider = false
            client.server_capabilities.documentRangeFormattingProvider = false
          end,
        }
      end,
    },
  },
}

return M
```

## Keymaps

| Key | Mode | Description |
|---|---|---|
| `<leader>lo` | n | Organize imports |
| `<leader>lR` | n | Rename file (updates imports) |
| `<leader>lm` | n | Add missing imports |
| `<leader>lu` | n | Remove unused imports |
| `<leader>lD` | n | Go to source definition |
