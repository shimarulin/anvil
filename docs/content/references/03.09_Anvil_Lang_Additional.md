# Anvil — Additional Language Packs

Language packs for C/C++, HTML/CSS, Docker, SQL, Zig, Nix, Protobuf, and Terraform.

### `lua/anvil/lang/c.lua`

<!-- lua/anvil/lang/c.lua -->
```lua
--- Anvil language pack: C/C++
--- Uses clangd for LSP and codelldb for debugging
---@type AnvilLangPack
local M = {
  name = "c",
  filetypes = { "c", "cpp", "objc", "objcpp", "cuda", "proto" },

  treesitter = { "c", "cpp", "cmake" },

  lsp = {
    server = "clangd",
    settings = {},
    on_attach = function(client, _)
      -- clangd has its own formatting; disable if using clang-format via conform
      client.server_capabilities.documentFormattingProvider = false
    end,
  },

  formatter = {
    tool = "clang-format",
  },

  dap = {
    adapter = "codelldb",
    adapter_config = {
      type = "server",
      port = "${port}",
      executable = {
        command = "codelldb",
        args = { "--port", "${port}" },
      },
    },
    configurations = {
      {
        type = "codelldb",
        request = "launch",
        name = "Launch (select binary)",
        program = function()
          return vim.fn.input("Binary path: ", vim.fn.getcwd() .. "/", "file")
        end,
        cwd = "${workspaceFolder}",
        stopOnEntry = false,
      },
      {
        type = "codelldb",
        request = "attach",
        name = "Attach to process",
        pid = function()
          return require("dap.utils").pick_process()
        end,
        cwd = "${workspaceFolder}",
      },
    },
    dependencies = { "codelldb" },
  },

  tasks = {
    {
      name = "C: Build (make)",
      cmd = "make",
      lang = "c",
      filetypes = { "c", "cpp" },
    },
    {
      name = "C: Build (cmake)",
      cmd = "cmake --build build",
      lang = "c",
      filetypes = { "c", "cpp" },
    },
    {
      name = "C: Clean",
      cmd = "make clean",
      lang = "c",
      filetypes = { "c", "cpp" },
    },
  },

  ai = {
    system_prompt = "This is a C/C++ project. Follow modern C++ conventions (C++17/20). "
      .. "Use RAII, smart pointers, and const correctness. "
      .. "Prefer standard library containers over raw arrays.",
    file_patterns = { "CMakeLists.txt", "Makefile", "compile_commands.json", ".clang-format", ".clang-tidy" },
    conventions = "snake_case for C functions, PascalCase for C++ classes. "
      .. "Use #pragma once for header guards. "
      .. "Prefer constexpr over #define for constants.",
  },

  keys = {
    {
      mode = "n",
      lhs = "<leader>lh",
      rhs = "<cmd>ClangdSwitchSourceHeader<cr>",
      desc = "Switch header/source",
    },
  },
}

return M
```

```lua
--- Anvil language pack: Docker
--- Dockerfile and Docker Compose
---@type AnvilLangPack
local M = {
  name = "docker",
  filetypes = { "dockerfile", "yaml.docker-compose" },

  treesitter = { "dockerfile" },

  lsp = {
    server = "dockerls",
    extra_servers = {
      { server = "docker_compose_language_service" },
    },
  },

  formatter = {
    tool = "prettierd",  -- for docker-compose.yml
    managed_by_lsp = true,  -- Dockerfile has no standard formatter
  },

  linter = {
    tool = "hadolint",
  },

  tasks = {
    {
      name = "Docker: Build",
      cmd = "docker build -t $(basename $(pwd)) .",
      lang = "docker",
      filetypes = { "dockerfile" },
    },
    {
      name = "Docker: Compose Up",
      cmd = "docker compose up -d",
      lang = "docker",
      filetypes = { "dockerfile", "yaml.docker-compose" },
    },
    {
      name = "Docker: Compose Down",
      cmd = "docker compose down",
      lang = "docker",
      filetypes = { "dockerfile", "yaml.docker-compose" },
    },
  },

  ai = {
    system_prompt = "This is a Docker project. "
      .. "Use multi-stage builds to minimize image size. "
      .. "Pin base image versions. Use COPY instead of ADD. "
      .. "Run as non-root user. Use .dockerignore.",
    file_patterns = { "Dockerfile", "Dockerfile.*", "docker-compose.yml", "docker-compose.*.yml", ".dockerignore" },
    conventions = "Use official base images. Pin versions (e.g. node:20-alpine, not node:latest). "
      .. "Order instructions from least to most frequently changing for layer caching.",
  },
}

return M
```
