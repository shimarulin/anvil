# Anvil — Installation & Setup

User-facing guide: from zero to a working Anvil environment.

## Requirements

| Dependency | Minimum | Recommended | Why |
|---|---|---|---|
| Neovim | 0.9 | **0.10+** | NVIM_APPNAME isolation, vim.uv, treesitter foldexpr |
| Git | 2.19 | latest | Partial clone, sparse checkout |
| Node.js | 18 | 20+ | Mason LSP servers, MCP servers |
| npm | — | latest | Comes with Node.js |
| ripgrep | — | latest | Telescope live grep |
| fd | — | latest | Telescope file finder |
| Nerd Font | — | any | Icons in statusline, explorer, completion |
| Terminal | — | kitty / alacritty / wezterm | True color, ligatures |

### Optional

| Tool | Purpose |
|---|---|
| lazygit | Git TUI (launched from Anvil) |
| delta | Better git diff rendering |
| pandoc | URL content conversion for AI chat |
| aider | Multi-file AI editing |
| opencode | AI planning agent |

## System Dependencies

### `scripts/preinstall.sh`

Installs system-level dependencies. Run once on a fresh machine.

<!-- file: scripts/preinstall.sh -->
```bash
#!/usr/bin/env bash
set -euo pipefail

# ─── Colors ───
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m'

log_info() { echo -e "${BLUE}[INFO]${NC} $*"; }
log_ok()   { echo -e "${GREEN}[ OK ]${NC} $*"; }
log_warn() { echo -e "${YELLOW}[WARN]${NC} $*"; }
log_err()  { echo -e "${RED}[ERR ]${NC} $*"; }

# ─── Detect OS ───
detect_os() {
  if [[ -f /etc/os-release ]]; then
    . /etc/os-release
    echo "$ID"
  elif [[ "$(uname -s)" == "Darwin" ]]; then
    echo "macos"
  else
    echo "unknown"
  fi
}

OS=$(detect_os)
log_info "Detected OS: $OS"

# ─── Package managers ───

install_apt() {
  log_info "Installing via apt..."
  sudo apt-get update -qq

  # Neovim — PPA for latest stable
  if ! command -v nvim &>/dev/null || [[ "$(nvim --version | head -1 | grep -oP '\d+\.\d+')" < "0.10" ]]; then
    log_info "Adding Neovim PPA..."
    sudo add-apt-repository -y ppa:neovim-ppa/stable 2>/dev/null || true
    sudo apt-get update -qq
  fi

  sudo apt-get install -y \
    neovim \
    git \
    curl \
    wget \
    unzip \
    ripgrep \
    fd-find \
    nodejs \
    npm \
    python3 \
    python3-pip \
    python3-venv \
    build-essential \
    cmake

  # fd-find installs as fdfind on Debian/Ubuntu
  if command -v fdfind &>/dev/null && ! command -v fd &>/dev/null; then
    sudo ln -sf "$(which fdfind)" /usr/local/bin/fd
    log_ok "Symlinked fdfind → fd"
  fi

  # Optional tools
  sudo apt-get install -y lazygit 2>/dev/null || log_warn "lazygit not in repos — install manually"
  sudo apt-get install -y pandoc 2>/dev/null || log_warn "pandoc not available"
}

install_brew() {
  log_info "Installing via Homebrew..."

  if ! command -v brew &>/dev/null; then
    log_err "Homebrew not found. Install from https://brew.sh"
    exit 1
  fi

  brew install \
    neovim \
    git \
    ripgrep \
    fd \
    node \
    lazygit \
    pandoc \
    cmake
}

install_dnf() {
  log_info "Installing via dnf..."
  sudo dnf install -y \
    neovim \
    git \
    ripgrep \
    fd-find \
    nodejs \
    npm \
    python3 \
    python3-pip \
    gcc \
    gcc-c++ \
    cmake

  sudo dnf install -y lazygit 2>/dev/null || log_warn "lazygit not in repos"
}

install_pacman() {
  log_info "Installing via pacman..."
  sudo pacman -Syu --noconfirm \
    neovim \
    git \
    ripgrep \
    fd \
    nodejs \
    npm \
    python \
    python-pip \
    base-devel \
    cmake \
    lazygit \
    pandoc
}

# ─── Main ───

echo -e "${BLUE}"
echo "  ╔═══════════════════════════════════════╗"
echo "  ║     ⚒  ANVIL — System Setup  ⚒       ║"
echo "  ╚═══════════════════════════════════════╝"
echo -e "${NC}"

case "$OS" in
  ubuntu|debian|pop|linuxmint)
    install_apt ;;
  macos)
    install_brew ;;
  fedora|rhel|centos)
    install_dnf ;;
  arch|manjaro|endeavouros)
    install_pacman ;;
  *)
    log_err "Unsupported OS: $OS"
    log_info "Install manually: neovim git ripgrep fd nodejs npm"
    exit 1 ;;
esac

# ─── Verify ───

echo ""
log_info "Verifying installation..."

REQUIRED=("nvim" "git" "node" "npm")
RECOMMENDED=("rg" "fd" "lazygit" "pandoc")

all_ok=true
for cmd in "${REQUIRED[@]}"; do
  if command -v "$cmd" &>/dev/null; then
    version=$("$cmd" --version 2>/dev/null | head -1 || echo "installed")
    log_ok "$cmd — $version"
  else
    log_err "$cmd — NOT FOUND (required)"
    all_ok=false
  fi
done

for cmd in "${RECOMMENDED[@]}"; do
  if command -v "$cmd" &>/dev/null; then
    version=$("$cmd" --version 2>/dev/null | head -1 || echo "installed")
    log_ok "$cmd — $version"
  else
    log_warn "$cmd — not found (recommended)"
  fi
done

echo ""
if $all_ok; then
  log_ok "All required dependencies installed"
  echo ""
  echo -e "  Next: ${GREEN}./install.sh${NC}"
else
  log_err "Some required dependencies are missing"
  exit 1
fi
```

## Anvil Installation

### `scripts/install.sh`

Main installation script. Creates symlinks, sets up environment, bootstraps plugins.

<!-- file: scripts/install.sh -->
```bash
#!/usr/bin/env bash
set -euo pipefail

# ─── Colors ───
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
CYAN='\033[0;36m'
NC='\033[0m'

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
PROJECT_ROOT="$(dirname "$SCRIPT_DIR")"

# ─── Platform paths ───
NVIM_CONFIG_HOME="${XDG_CONFIG_HOME:-$HOME/.config}"
NVIM_DATA_HOME="${XDG_DATA_HOME:-$HOME/.local/share}"
NVIM_STATE_HOME="${XDG_STATE_HOME:-$HOME/.local/state}"
NVIM_CACHE_HOME="${XDG_CACHE_HOME:-$HOME/.cache}"

APPNAME="${ANVIL_APPNAME:-anvil}"
CONFIG_TARGET="$NVIM_CONFIG_HOME/$APPNAME"

log_info() { echo -e "${BLUE}[INFO]${NC} $*"; }
log_ok()   { echo -e "${GREEN}[ OK ]${NC} $*"; }
log_warn() { echo -e "${YELLOW}[WARN]${NC} $*"; }
log_err()  { echo -e "${RED}[ERR ]${NC} $*"; }

print_header() {
  echo -e "${CYAN}"
  echo "  ╔═══════════════════════════════════════╗"
  echo "  ║       ⚒  ANVIL Installer  ⚒           ║"
  echo "  ╚═══════════════════════════════════════╝"
  echo -e "${NC}"
}

# ─── Dependency check ───

check_deps() {
  local missing=()
  command -v git  &>/dev/null || missing+=("git")
  command -v nvim &>/dev/null || missing+=("nvim")
  command -v node &>/dev/null || missing+=("node")
  command -v npm  &>/dev/null || missing+=("npm")

  if [[ ${#missing[@]} -gt 0 ]]; then
    log_err "Missing dependencies: ${missing[*]}"
    log_info "Run: ./scripts/preinstall.sh"
    exit 1
  fi

  # Check Neovim version
  local nvim_ver
  nvim_ver=$(nvim --version | head -1 | grep -oP 'v\K[0-9]+\.[0-9]+' || true)
  if [[ -n "$nvim_ver" ]]; then
    local major minor
    major=$(echo "$nvim_ver" | cut -d. -f1)
    minor=$(echo "$nvim_ver" | cut -d. -f2)
    if [[ $major -eq 0 && $minor -lt 9 ]]; then
      log_err "Neovim v0.9+ required (found v$nvim_ver)"
      exit 1
    fi
    if [[ $major -eq 0 && $minor -lt 10 ]]; then
      log_warn "Neovim v$nvim_ver — v0.10+ recommended for full features"
    fi
  fi

  log_ok "Dependencies OK (nvim v${nvim_ver:-unknown})"
}

# ─── Symlink ───

create_symlink() {
  local source="$1"
  local target="$2"

  if [[ -L "$target" ]]; then
    local current
    current=$(readlink -f "$target" 2>/dev/null || readlink "$target")
    local resolved
    resolved=$(readlink -f "$source" 2>/dev/null || echo "$source")

    if [[ "$current" == "$resolved" ]]; then
      log_ok "Symlink OK: $target"
      return 0
    else
      log_warn "Updating symlink: $target"
      rm "$target"
    fi
  elif [[ -e "$target" ]]; then
    local backup="${target}.bak.$(date +%Y%m%d%H%M%S)"
    log_warn "Backing up existing: $target → $backup"
    mv "$target" "$backup"
  fi

  mkdir -p "$(dirname "$target")"
  ln -s "$source" "$target"
  log_ok "Symlink: $target → $source"
}

# ─── Environment file ───

setup_env() {
  local env_file="$PROJECT_ROOT/.env"
  local env_example="$PROJECT_ROOT/.env.example"

  if [[ -f "$env_file" ]]; then
    log_ok ".env already exists"
    return
  fi

  if [[ -f "$env_example" ]]; then
    cp "$env_example" "$env_file"
    log_ok ".env created from .env.example"
    log_warn "Edit .env and set your AI_API_KEY"
  else
    cat > "$env_file" << 'ENVEOF'
# Anvil environment — edit these values
AI_ENDPOINT=https://openrouter.ai/api/v1
AI_API_KEY=
AI_MODEL=anthropic/claude-sonnet-4
ANVIL_COLORSCHEME=catppuccin
ENVEOF
    log_ok ".env created with defaults"
    log_warn "Edit .env and set your AI_API_KEY"
  fi
}

# ─── MCP config ───

setup_mcp() {
  local mcp_dir="$NVIM_CONFIG_HOME/$APPNAME"
  local mcp_file="$mcp_dir/mcp-servers.json"

  if [[ -f "$mcp_file" ]]; then
    log_ok "MCP config already exists"
    return
  fi

  # MCP config is in the project, symlink will handle it
  log_info "MCP config will be available via symlink"
}

# ─── Bootstrap plugins ───

bootstrap() {
  log_info "Bootstrapping Neovim plugins (this may take a minute)..."
  if NVIM_APPNAME="$APPNAME" nvim --headless +"Lazy! sync" +qall 2>&1; then
    log_ok "Plugins installed"
  else
    log_warn "Plugin bootstrap had warnings (this is usually fine on first run)"
  fi
}

# ─── Shell aliases ───

print_shell_setup() {
  echo ""
  echo -e "${CYAN}─── Shell Setup (add to ~/.bashrc or ~/.zshrc) ───${NC}"
  echo ""
  echo "  # Anvil"
  echo "  alias anvil='NVIM_APPNAME=$APPNAME nvim'"
  echo ""
  echo "  # Or make it the default editor:"
  echo "  # export EDITOR='NVIM_APPNAME=$APPNAME nvim'"
  echo ""
}

# ─── Uninstall ───

cmd_uninstall() {
  echo -e "${YELLOW}Uninstalling Anvil...${NC}"

  if [[ -L "$CONFIG_TARGET" ]]; then
    rm "$CONFIG_TARGET"
    log_ok "Removed symlink: $CONFIG_TARGET"
  elif [[ -e "$CONFIG_TARGET" ]]; then
    log_warn "$CONFIG_TARGET is not a symlink — refusing to remove"
    log_info "Remove manually: rm -rf $CONFIG_TARGET"
    return 1
  fi

  read -rp "Clean data/state/cache for '$APPNAME'? [y/N] " answer
  if [[ "$answer" =~ ^[Yy]$ ]]; then
    rm -rf "$NVIM_DATA_HOME/$APPNAME"
    rm -rf "$NVIM_STATE_HOME/$APPNAME"
    rm -rf "$NVIM_CACHE_HOME/$APPNAME"
    log_ok "Data cleaned"
  fi

  log_ok "Anvil uninstalled"
}

# ─── Status ───

cmd_status() {
  echo -e "${CYAN}Anvil Status${NC}"
  echo ""

  if [[ -L "$CONFIG_TARGET" ]]; then
    echo -e "  Config: ${GREEN}$CONFIG_TARGET → $(readlink "$CONFIG_TARGET")${NC}"
  elif [[ -d "$CONFIG_TARGET" ]]; then
    echo -e "  Config: ${YELLOW}$CONFIG_TARGET (exists, not symlink)${NC}"
  else
    echo -e "  Config: ${RED}not installed${NC}"
  fi

  echo -e "  Data:   $NVIM_DATA_HOME/$APPNAME"
  echo -e "  State:  $NVIM_STATE_HOME/$APPNAME"
  echo -e "  Cache:  $NVIM_CACHE_HOME/$APPNAME"
  echo ""

  if [[ -f "$PROJECT_ROOT/.env" ]]; then
    if grep -q "^AI_API_KEY=.\+" "$PROJECT_ROOT/.env" 2>/dev/null; then
      echo -e "  API Key: ${GREEN}configured${NC}"
    else
      echo -e "  API Key: ${YELLOW}not set (edit .env)${NC}"
    fi
  else
    echo -e "  .env: ${RED}missing${NC}"
  fi
}

# ─── Usage ───

usage() {
  echo "Usage: $0 [command]"
  echo ""
  echo "Commands:"
  echo "  install     Install Anvil (default)"
  echo "  uninstall   Remove Anvil config and optionally data"
  echo "  status      Show installation status"
  echo "  help        Show this help"
  echo ""
  echo "Environment:"
  echo "  ANVIL_APPNAME   Neovim app name (default: anvil)"
}

# ─── Main ───

main() {
  print_header

  local cmd="${1:-install}"

  case "$cmd" in
    install)
      check_deps
      echo ""

      log_info "Installing Anvil as NVIM_APPNAME='$APPNAME'"
      echo ""

      # 1. Symlink project root → Neovim config
      create_symlink "$PROJECT_ROOT" "$CONFIG_TARGET"

      # 2. Create .env
      setup_env

      # 3. Bootstrap plugins
      echo ""
      read -rp "Bootstrap plugins now? [Y/n] " answer
      if [[ ! "$answer" =~ ^[Nn]$ ]]; then
        bootstrap
      else
        log_info "Skipped. Run later: make bootstrap"
      fi

      # 4. Print next steps
      echo ""
      echo -e "${GREEN}═══ Installation complete ═══${NC}"
      print_shell_setup
      echo -e "${CYAN}─── Next Steps ───${NC}"
      echo ""
      echo "  1. Edit .env and set AI_API_KEY"
      echo "  2. Launch: NVIM_APPNAME=$APPNAME nvim"
      echo "  3. Verify: :Anvil health"
      echo "  4. Optional: make sync-refs  (download framework docs)"
      echo ""
      ;;
    uninstall)
      cmd_uninstall ;;
    status)
      cmd_status ;;
    help|--help|-h)
      usage ;;
    *)
      log_err "Unknown command: $cmd"
      usage
      exit 1 ;;
  esac
}

main "$@"
```

## Step-by-Step Installation

### Quick Start (3 commands)

```bash
# 1. Clone
git clone https://github.com/your-username/anvil ~/.local/src/anvil
cd ~/.local/src/anvil

# 2. Install system deps (if needed)
./scripts/preinstall.sh

# 3. Install Anvil
./scripts/install.sh
```

### What `install.sh` Does

```
1. Checks dependencies (nvim, git, node, npm)
2. Creates symlink:
     ~/.config/anvil → ~/.local/src/anvil
3. Creates .env from .env.example
4. Bootstraps plugins (Lazy.nvim sync)
```

### After Installation

```bash
# Launch Anvil
NVIM_APPNAME=anvil nvim

# Or add alias to shell config
echo "alias anvil='NVIM_APPNAME=anvil nvim'" >> ~/.bashrc
source ~/.bashrc

# Then just:
anvil
```

### First Launch Checklist

1. **`:Anvil health`** — verify everything is working
2. **`:Lazy`** — check plugin status
3. **`:Mason`** — verify LSP servers are installing
4. **Edit `.env`** — set `AI_API_KEY` for AI features
5. **`:AnvilChat`** — test AI (requires API key)

## Configuration

### Environment Variables

All configuration is via `.env` file in the project root:

```bash
# ─── AI Provider ───
AI_ENDPOINT=https://openrouter.ai/api/v1
AI_API_KEY=sk-or-v1-xxxxxxxxxxxxx
AI_MODEL=anthropic/claude-sonnet-4

# ─── Theme ───
ANVIL_COLORSCHEME=catppuccin

# ─── Log level ───
ANVIL_LOG_LEVEL=info
```

Priority: OS environment > `.env.local` > `.env`

### Boot Configuration

Edit `init.lua` to customize:

```lua
require("anvil").boot({
  -- Languages to enable (nil = all)
  languages = { "typescript", "python", "lua", "json", "markdown" },

  -- Theme
  ui = {
    colorscheme = "tokyonight",  -- or "catppuccin", "rose-pine", "kanagawa"
    strategy = "auto",           -- "auto", "standalone", "nvchad"
  },

  -- Vim options override
  options = {
    opt = { relativenumber = false, shiftwidth = 4 },
  },
})
```

### Adding a Language

Create a single file:

```bash
cp lua/anvil/lang/_template.lua lua/anvil/lang/elixir.lua
# Edit the file, fill in LSP/formatter/linter/test/DAP
```

Restart Neovim — Anvil picks it up automatically.

## NVIM_APPNAME Isolation

Anvil uses Neovim's `NVIM_APPNAME` (v0.9+) for complete isolation:

```
NVIM_APPNAME=anvil
  config: ~/.config/anvil/
  data:   ~/.local/share/anvil/
  state:  ~/.local/state/anvil/
  cache:  ~/.cache/anvil/

nvim (default, untouched)
  config: ~/.config/nvim/
  data:   ~/.local/share/nvim/
```

Your existing Neovim configuration is never touched.

## Troubleshooting

### Plugins fail to install

```bash
# Clean and retry
make clean
make bootstrap
```

### LSP servers not found

```bash
# Open Anvil and run Mason
NVIM_APPNAME=anvil nvim
:MasonToolsInstall
```

### AI features not working

```bash
# Check API key
grep AI_API_KEY .env

# Test endpoint
curl -s https://openrouter.ai/api/v1/models \
  -H "Authorization: Bearer $AI_API_KEY" | head -20

# Check health
:checkhealth anvil_ai
```

### Treesitter parsers failing to compile

```bash
# Ensure build tools are installed
# Ubuntu/Debian:
sudo apt-get install build-essential

# macOS:
xcode-select --install
```

### Wrong theme / UI looks broken

```bash
# Verify terminal supports true color
echo -e "\033[38;2;255;100;0mTRUE COLOR TEST\033[0m"
# Should show orange text

# Check Nerd Font is installed
# In Neovim:
:echo &guifont
```

### Symlink issues on macOS

macOS `readlink` doesn't support `-f`. The install script handles this,
but if you create symlinks manually:

```bash
# Use greadlink from coreutils
brew install coreutils
greadlink -f ~/.config/anvil
```

## Updating

```bash
cd ~/.local/src/anvil

# Update Anvil config
git pull

# Update plugins
make update

# Update Mason tools
NVIM_APPNAME=anvil nvim -c "MasonToolsUpdate" -c "qall"
```

## Uninstalling

```bash
./scripts/install.sh uninstall
```

This removes the symlink and optionally cleans data/state/cache.
The cloned repository is not deleted — remove it manually if desired.
