# Anvil — Editor Layer

Core editing experience: completion, search, navigation, text objects, and all the plugins that make Neovim feel like a modern IDE regardless of which language you're working with.

## Architecture

```text
lua/anvil/editor/
├── init.lua         # editor subsystem setup
├── lsp.lua          # LSP orchestrator (strategy point)
├── treesitter.lua   # treesitter base configuration
├── cmp.lua          # completion (nvim-cmp)
├── format.lua       # formatting (conform.nvim)
├── lint.lua         # linting (nvim-lint)
├── dap.lua          # debugging (nvim-dap)
├── test.lua         # testing (neotest)
├── pairs.lua        # auto-pairs, surround
└── snippets.lua     # snippet engine (LuaSnip)

lua/plugins/
├── editor.lua       # editor plugin specs
├── navigation.lua   # telescope, harpoon, flash
└── lsp.lua          # LSP + mason specs
```

## Strategy Points

The editor layer uses standard Neovim plugin APIs directly. For future multi-framework support (Path C), certain functions serve as strategy points — module-level functions that a framework bridge can override before the lang engine runs.

### Strategy point contract

```lua
-- A strategy point is a function assigned to a module field.
-- Default implementation uses standard Neovim plugin APIs.
-- A bridge can override it before lang.setup() runs.

-- editor/lsp.lua
M.register_server = function(server, opts)
  require("lspconfig")[server].setup(opts)
end

-- bridge/astronvim/init.lua (future)
require("anvil.editor.lsp").register_server = function(server, opts)
  -- AstroNvim: merge into AstroLSP config
  require("astrolsp").config[server] = opts
end
```

### List of strategy points

| Module | Function | Default behavior |
|---|---|---|
| editor/lsp | `register_server()` | `lspconfig[server].setup(opts)` |
| editor/lsp | `build_capabilities()` | `cmp_nvim_lsp.default_capabilities()` |
| editor/lsp | `build_on_attach()` | Returns Anvil's on_attach chain |
| editor/dap | `register_adapter()` | `dap.adapters[name] = config` |
| editor/dap | `register_config()` | `dap.configurations[ft] = configs` |
| editor/test | `register_adapter()` | `table.insert(neotest_opts.adapters, ...)` |

## LSP Orchestrator

### `lua/anvil/editor/lsp.lua`

This is the key module that replaces AstroLSP. It builds capabilities, manages on_attach, and registers servers from language packs.

<!-- file: lua/anvil/editor/lsp.lua -->
```lua
--- Anvil LSP orchestrator
--- Replaces AstroLSP: builds capabilities, on_attach, registers servers
--- Strategy points: register_server, build_capabilities, build_on_attach
---@class AnvilEditorLSP
local M = {}

local icons = require("anvil.core.icons")

--- Strategy point: build LSP capabilities
--- Override this in a framework bridge if the framework manages capabilities
---@return table capabilities
M.build_capabilities = function()
  local capabilities = vim.lsp.protocol.make_client_capabilities()

  -- Merge cmp-nvim-lsp capabilities
  local cmp_ok, cmp_lsp = pcall(require, "cmp_nvim_lsp")
  if cmp_ok then
    capabilities = vim.tbl_deep_extend(
      "force",
      capabilities,
      cmp_lsp.default_capabilities()
    )
  end

  -- Enable folding capabilities for nvim-ufo (if used)
  capabilities.textDocument.foldingRange = {
    dynamicRegistration = false,
    lineFoldingOnly = true,
  }

  return capabilities
end

--- Servers that should NOT format (conform handles it)
---@type string[]
M.disable_formatting = {
  "ts_ls", "vtsls", "lua_ls", "html", "cssls", "jsonls", "yamlls",
}

--- Strategy point: build on_attach function
--- Override this in a framework bridge if the framework has its own on_attach
---@return function(client, bufnr)
M.build_on_attach = function()
  return function(client, bufnr)
    -- Disable formatting for servers where conform handles it
    if vim.tbl_contains(M.disable_formatting, client.name) then
      client.server_capabilities.documentFormattingProvider = false
      client.server_capabilities.documentRangeFormattingProvider = false
    end

    -- Apply language-specific keymaps
    local lang_ok, lang = pcall(require, "anvil.lang")
    if lang_ok then
      lang.apply_keys(bufnr)
    end

    -- Attach navic for breadcrumbs
    local navic_ok, navic = pcall(require, "nvim-navic")
    if navic_ok and client.server_capabilities.documentSymbolProvider then
      navic.attach(client, bufnr)
    end

    -- Emit event
    local events_ok, events = pcall(require, "anvil.core.events")
    if events_ok then
      events.emit("lang:attached", {
        client = client.name,
        bufnr = bufnr,
      })
    end
  end
end

--- Strategy point: register an LSP server
--- Override this in a framework bridge if the framework manages LSP setup
---@param server string lspconfig server name
---@param opts table server options (settings, on_attach, capabilities, etc.)
M.register_server = function(server, opts)
  local lspconfig_ok, lspconfig = pcall(require, "lspconfig")
  if not lspconfig_ok then
    vim.notify("[Anvil:LSP] lspconfig not available", vim.log.levels.ERROR)
    return
  end
  lspconfig[server].setup(opts)
end

--- Setup all LSP servers from language packs
--- Called after lang.setup() has registered all packs
function M.setup()
  local lang_ok, lang = pcall(require, "anvil.lang")
  if not lang_ok then return end

  local capabilities = M.build_capabilities()
  local on_attach = M.build_on_attach()
  local servers = lang.lsp_servers()

  for server, settings in pairs(servers) do
    local opts = vim.tbl_deep_extend("force", {
      capabilities = capabilities,
      on_attach = on_attach,
    }, settings)

    M.register_server(server, opts)
  end

  -- Configure diagnostics
  M.setup_diagnostics()

  local logger_ok, logger = pcall(require, "anvil.core.logger")
  if logger_ok then
    logger.info("LSP: %d servers configured", vim.tbl_count(servers))
  end
end

--- Configure vim.diagnostic
function M.setup_diagnostics()
  vim.diagnostic.config({
    virtual_text = {
      prefix = "●",
      spacing = 4,
    },
    signs = {
      text = {
        [vim.diagnostic.severity.ERROR] = icons.diagnostics.error,
        [vim.diagnostic.severity.WARN]  = icons.diagnostics.warn,
        [vim.diagnostic.severity.HINT]  = icons.diagnostics.hint,
        [vim.diagnostic.severity.INFO]  = icons.diagnostics.info,
      },
    },
    update_in_insert = false,
    underline = true,
    severity_sort = true,
    float = {
      focused = false,
      style = "minimal",
      border = "rounded",
      source = true,
      header = "",
      prefix = "",
    },
  })
end

--- LSP keymaps (applied to every buffer with LSP attached)
--- These are set in lua/plugins/lsp.lua via LspAttach autocmd
---@param bufnr number
function M.set_keymaps(bufnr)
  local map = function(mode, lhs, rhs, desc)
    vim.keymap.set(mode, lhs, rhs, { buffer = bufnr, desc = desc })
  end

  map("n", "gd", function() require("telescope.builtin").lsp_definitions() end, "Go to definition")
  map("n", "gD", function() vim.lsp.buf.declaration() end, "Go to declaration")
  map("n", "gr", function() require("telescope.builtin").lsp_references() end, "Go to references")
  map("n", "gi", function() require("telescope.builtin").lsp_implementations() end, "Go to implementations")
  map("n", "gy", function() require("telescope.builtin").lsp_type_definitions() end, "Go to type definition")
  map("n", "K",  function() vim.lsp.buf.hover() end, "Hover documentation")
  map("n", "<leader>la", function() vim.lsp.buf.code_action() end, "Code action")
  map("n", "<leader>lr", function() vim.lsp.buf.rename() end, "Rename symbol")
  map("n", "<leader>lf", function() vim.lsp.buf.format({ async = true }) end, "Format document")
  map("n", "<leader>lh", function() vim.lsp.inlay_hint.enable(not vim.lsp.inlay_hint.is_enabled()) end, "Toggle inlay hints")
  map("n", "<leader>ls", function() require("telescope.builtin").lsp_document_symbols() end, "Document symbols")
  map("n", "<leader>lS", function() require("telescope.builtin").lsp_dynamic_workspace_symbols() end, "Workspace symbols")
  map("n", "<leader>li", "<cmd>LspInfo<cr>", "LSP info")
  map("n", "<leader>ld", function() vim.diagnostic.open_float() end, "Line diagnostics")
end

return M
```

```lua
--- Anvil treesitter base configuration
--- Parsers are added by language packs; this provides shared settings
---@class AnvilEditorTreesitter
local M = {}

--- Base parsers always installed (not language-specific)
---@type string[]
M.base_parsers = {
  "vim", "vimdoc", "query", "regex",
  "lua", "comment", "diff",
  "gitcommit", "git_rebase",
}

--- Get treesitter plugin opts
--- Merges base parsers with language pack parsers
---@return table opts for nvim-treesitter
function M.opts()
  local lang_ok, lang = pcall(require, "anvil.lang")
  local lang_parsers = lang_ok and lang.treesitter_ensure_installed() or {}

  -- Merge and deduplicate
  local parsers = vim.list_extend(vim.deepcopy(M.base_parsers), lang_parsers)
  local seen = {}
  parsers = vim.tbl_filter(function(p)
    if seen[p] then return false end
    seen[p] = true
    return true
  end, parsers)

  return {
    ensure_installed = parsers,
    highlight = {
      enable = true,
      additional_vim_regex_highlighting = false,
      disable = function(_, buf)
        local max_filesize = 1024 * 1024 -- 1MB
        local ok, stats = pcall(vim.uv.fs_stat, vim.api.nvim_buf_get_name(buf))
        if ok and stats and stats.size > max_filesize then
          return true
        end
      end,
    },
    indent = { enable = true },
    incremental_selection = {
      enable = true,
      keymaps = {
        init_selection    = "<C-space>",
        node_incremental  = "<C-space>",
        scope_incremental = false,
        node_decremental  = "<bs>",
      },
    },
    textobjects = {
      select = {
        enable = true,
        lookahead = true,
        keymaps = {
          ["af"] = "@function.outer",
          ["if"] = "@function.inner",
          ["ac"] = "@class.outer",
          ["ic"] = "@class.inner",
          ["aa"] = "@parameter.outer",
          ["ia"] = "@parameter.inner",
          ["ai"] = "@conditional.outer",
          ["ii"] = "@conditional.inner",
          ["al"] = "@loop.outer",
          ["il"] = "@loop.inner",
          ["ab"] = "@block.outer",
          ["ib"] = "@block.inner",
          ["a/"] = "@comment.outer",
        },
      },
      move = {
        enable = true,
        set_jumps = true,
        goto_next_start     = { ["]f"] = "@function.outer", ["]c"] = "@class.outer", ["]a"] = "@parameter.inner" },
        goto_next_end       = { ["]F"] = "@function.outer", ["]C"] = "@class.outer" },
        goto_previous_start = { ["[f"] = "@function.outer", ["[c"] = "@class.outer", ["[a"] = "@parameter.inner" },
        goto_previous_end   = { ["[F"] = "@function.outer", ["[C"] = "@class.outer" },
      },
      swap = {
        enable = true,
        swap_next     = { ["<leader>sa"] = "@parameter.inner" },
        swap_previous = { ["<leader>sA"] = "@parameter.inner" },
      },
    },
  }
end

return M
```

```lua
--- Anvil completion configuration
--- nvim-cmp with LSP, snippets, buffer, path sources
---@class AnvilEditorCmp
local M = {}

--- Get nvim-cmp opts
---@return table
function M.opts()
  local cmp = require("cmp")
  local luasnip = require("luasnip")
  local lspkind = require("lspkind")
  local icons = require("anvil.core.icons")

  return {
    snippet = {
      expand = function(args)
        luasnip.lsp_expand(args.body)
      end,
    },

    window = {
      completion = cmp.config.window.bordered({
        winhighlight = "Normal:NormalFloat,FloatBorder:FloatBorder,"
          .. "CursorLine:PmenuSel,Search:None",
        col_offset = -3,
        side_padding = 1,
      }),
      documentation = cmp.config.window.bordered({
        winhighlight = "Normal:NormalFloat,FloatBorder:FloatBorder,"
          .. "CursorLine:PmenuSel,Search:None",
        max_width = 80,
        max_height = 20,
      }),
    },

    formatting = {
      fields = { "kind", "abbr", "menu" },
      format = lspkind.cmp_format({
        mode = "symbol_text",
        maxwidth = 50,
        ellipsis_char = "...",
        symbol_map = icons.kinds,
      }),
    },

    mapping = cmp.mapping.preset.insert({
      ["<C-k>"]     = cmp.mapping.select_prev_item({ behavior = cmp.SelectBehavior.Insert }),
      ["<C-j>"]     = cmp.mapping.select_next_item({ behavior = cmp.SelectBehavior.Insert }),
      ["<C-b>"]     = cmp.mapping.scroll_docs(-4),
      ["<C-f>"]     = cmp.mapping.scroll_docs(4),
      ["<CR>"]      = cmp.mapping.confirm({ select = false }),
      ["<C-y>"]     = cmp.mapping.confirm({ select = true }),
      ["<C-e>"]     = cmp.mapping.abort(),
      ["<C-Space>"] = cmp.mapping.complete(),

      ["<Tab>"] = cmp.mapping(function(fallback)
        if cmp.visible() then
          cmp.select_next_item()
        elseif luasnip.expand_or_locally_jumpable() then
          luasnip.expand_or_jump()
        else
          fallback()
        end
      end, { "i", "s" }),

      ["<S-Tab>"] = cmp.mapping(function(fallback)
        if cmp.visible() then
          cmp.select_prev_item()
        elseif luasnip.jumpable(-1) then
          luasnip.jump(-1)
        else
          fallback()
        end
      end, { "i", "s" }),
    }),

    sources = cmp.config.sources({
      { name = "nvim_lsp", priority = 1000 },
      { name = "luasnip",  priority = 750 },
      { name = "path",     priority = 500 },
    }, {
      {
        name = "buffer",
        priority = 250,
        option = {
          get_bufnrs = function()
            local bufs = {}
            for _, win in ipairs(vim.api.nvim_list_wins()) do
              bufs[vim.api.nvim_win_get_buf(win)] = true
            end
            return vim.tbl_keys(bufs)
          end,
        },
      },
    }),

    sorting = {
      priority_weight = 2,
      comparators = {
        cmp.config.compare.offset,
        cmp.config.compare.exact,
        cmp.config.compare.score,
        cmp.config.compare.recently_used,
        cmp.config.compare.locality,
        cmp.config.compare.kind,
        cmp.config.compare.length,
        cmp.config.compare.order,
      },
    },

    experimental = {
      ghost_text = { hl_group = "CmpGhostText" },
    },

    performance = {
      debounce = 60,
      throttle = 30,
      fetching_timeout = 500,
      max_view_entries = 50,
    },
  }
end

--- Setup cmdline completions (called after cmp.setup)
function M.setup_cmdline()
  local cmp = require("cmp")

  cmp.setup.cmdline("/", {
    mapping = cmp.mapping.preset.cmdline(),
    sources = { { name = "buffer" } },
  })

  cmp.setup.cmdline(":", {
    mapping = cmp.mapping.preset.cmdline(),
    sources = cmp.config.sources(
      { { name = "path" } },
      { { name = "cmdline" } }
    ),
  })
end

return M
```

```lua
--- Anvil formatting configuration (conform.nvim)
--- Formatters per filetype come from language packs
---@class AnvilEditorFormat
local M = {}

--- Get conform.nvim opts
---@return table
function M.opts()
  local lang_ok, lang = pcall(require, "anvil.lang")
  local formatters_by_ft = lang_ok and lang.formatters_by_ft() or {}

  return {
    formatters_by_ft = formatters_by_ft,

    format_on_save = function(bufnr)
      -- Disable for large files
      if vim.b[bufnr].large_buf then return end

      -- Check if language pack wants format on save
      if lang_ok then
        local pack = lang.current()
        if pack and pack.formatter and pack.formatter.format_on_save == false then
          return
        end
      end

      return {
        timeout_ms = 3000,
        lsp_fallback = true,
      }
    end,

    notify_on_error = true,
  }
end

return M
```

```lua
--- Anvil DAP configuration
--- Debug adapters and configurations come from language packs
--- Strategy points: register_adapter, register_config
---@class AnvilEditorDAP
local M = {}

local icons = require("anvil.core.icons")

--- Strategy point: register a DAP adapter
---@param name string adapter name
---@param config table adapter configuration
M.register_adapter = function(name, config)
  local dap = require("dap")
  dap.adapters[name] = config
end

--- Strategy point: register DAP configurations for a filetype
---@param ft string filetype
---@param configs table[] DAP launch configurations
M.register_config = function(ft, configs)
  local dap = require("dap")
  dap.configurations[ft] = configs
end

--- Setup DAP from language packs
function M.setup()
  local lang_ok, lang = pcall(require, "anvil.lang")
  if not lang_ok then return end

  local dap_configs = lang.dap_configurations()
  for adapter_name, config in pairs(dap_configs) do
    if config.adapter then
      M.register_adapter(adapter_name, config.adapter)
    end
    if config.configurations then
      for _, ft in ipairs(config.filetypes or {}) do
        M.register_config(ft, config.configurations)

  vim.fn.sign_define("DapBreakpoint",          { text = icons.dap.breakpoint,           texthl = "DiagnosticError" })
```

```lua
local M = {}

--- Strategy point: register a neotest adapter
--- Override in framework bridge if needed

---@param adapters table[] target adapters list

--- Build neotest adapter list from language packs
---@return table[] adapters
function M.build_adapters()
  local adapters = {}

    local pack = lang.get(name)
    if pack and pack.test then
      local adapter_ok, adapter = pcall(require, pack.test.adapter)
      if adapter_ok then

      end
    end
  end

  return adapters
end
```
