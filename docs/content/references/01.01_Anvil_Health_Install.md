# Anvil — Health Check & Installation

Bootstrap scripts, dependency management, health diagnostics, and
a Makefile that ties everything together. Run `make install` on a
fresh machine and get a fully working Anvil setup.

## File Map

```
anvil/
├── Makefile                    # make install, make health, make update
├── scripts/
│   ├── install.sh              # full installation orchestrator
│   ├── install-deps.sh         # system dependencies (OS-aware)
│   ├── install-toolchains.sh   # language toolchains
│   ├── install-mason.sh        # mason packages (headless nvim)
│   ├── install-mcp.sh          # MCP servers
│   ├── install-ai-tools.sh     # external AI tools
│   ├── install-fonts.sh        # Nerd Fonts
│   ├── health-check.sh         # pre-flight check (no nvim needed)
│   ├── update.sh               # update everything
│   └── sync-ref.sh             # update .ref/ files (from 00.00)
├── lua/anvil/health.lua        # :checkhealth anvil
└── config/
    └── servers.json            # MCP server configuration
```

```makefile
# ╔═══════════════════════════════════════════════════════╗
# ║  Anvil — Makefile                                     ║
# ║  make install   → full setup from scratch             ║
# ║  make health    → check everything                    ║
# ║  make update    → update all components               ║
# ╚═══════════════════════════════════════════════════════╝

SHELL := /bin/bash
.DEFAULT_GOAL := help

# ─── Paths ───
NVIM_APPNAME   ?= astronvim
CONFIG_DIR     := $$(HOME)/.config/$$(NVIM_APPNAME)
DATA_DIR       := $$(HOME)/.local/share/$$(NVIM_APPNAME)
STATE_DIR      := $$(HOME)/.local/state/$$(NVIM_APPNAME)
CACHE_DIR      := $$(HOME)/.cache/$$(NVIM_APPNAME)
SCRIPTS_DIR    := $$(shell dirname $$(realpath $$(firstword $$(MAKEFILE_LIST))))/scripts

# ─── Colors ───

YELLOW := \033[0;33m
RED    := \033[0;31m
CYAN   := \033[0;36m
RESET  := \033[0m

# ═══════════════════════════════════════════════════════
# Help
# ═══════════════════════════════════════════════════════

.PHONY: help
help: ## Show this help
	@echo ""
	@echo "  $$(CYAN)Anvil$$(RESET) — AstroNvim Configuration"
	@echo ""
	@grep -E '^[a-zA-Z_-]+:.*?## .*$$$' $$(MAKEFILE_LIST) | sort | \
		awk 'BEGIN {FS = ":.*?## "}; {printf "  $$(GREEN)%-20s$$(RESET) %s\n", $$1, $$2}'
	@echo ""

# ═══════════════════════════════════════════════════════
# Install
# ═══════════════════════════════════════════════════════

.PHONY: install
install: ## Full installation (deps + toolchains + plugins + mason + AI)
	@echo "$$(CYAN)═══ Anvil: Full Installation ═══$$(RESET)"
	@$(SCRIPTS_DIR)/install.sh

.PHONY: install-deps
install-deps: ## Install system dependencies only
	@$(SCRIPTS_DIR)/install-deps.sh

.PHONY: install-toolchains

	@$(SCRIPTS_DIR)/install-toolchains.sh

.PHONY: install-mason
install-mason: ## Install Mason packages (headless nvim)
	@$(SCRIPTS_DIR)/install-mason.sh

.PHONY: install-mcp
install-mcp: ## Install MCP servers
	@$(SCRIPTS_DIR)/install-mcp.sh

.PHONY: install-ai
install-ai: ## Install external AI tools
	@$(SCRIPTS_DIR)/install-ai-tools.sh

.PHONY: install-fonts
install-fonts: ## Install Nerd Fonts
	@$(SCRIPTS_DIR)/install-fonts.sh

# ═══════════════════════════════════════════════════════
# Health
# ═══════════════════════════════════════════════════════

.PHONY: health
health: ## Run health check (shell, no nvim needed)
	@$(SCRIPTS_DIR)/health-check.sh

.PHONY: health-nvim
health-nvim: ## Run :checkhealth anvil inside Neovim
	@NVIM_APPNAME=$(NVIM_APPNAME) nvim --headless "+checkhealth anvil" "+qa" 2>&1

# ═══════════════════════════════════════════════════════
# Update
# ═══════════════════════════════════════════════════════

.PHONY: update
update: ## Update everything (plugins, mason, tools)

.PHONY: update-plugins
update-plugins: ## Update Neovim plugins only

	@echo "$$(GREEN)✓ Plugins updated$$(RESET)"

.PHONY: update-mason
update-mason: ## Update Mason packages only

# Development
# ═══════════════════════════════════════════════════════

.PHONY: lint
lint: ## Lint Anvil configuration

	@selene lua/ || true
	@echo "$$(CYAN)→ Running stylua check...$$(RESET)"
	@stylua --check lua/ || true
	@echo "$$(CYAN)→ Running luacheck...$$(RESET)"

.PHONY: format
format: ## Format Anvil configuration

# ═══════════════════════════════════════════════════════

.PHONY: clean
clean: ## Remove all generated data (keep config)

	@rm -rf $(DATA_DIR)

	@rm -rf $(CACHE_DIR)

.PHONY: nuke
nuke: ## Remove EVERYTHING (full reset)
	@echo "$$(RED)This will remove ALL Anvil data including plugins.$$(RESET)"
	@read -p "Are you sure? [y/N] " confirm && [ "$$confirm" = "y" ] || exit 1
	@rm -rf $(DATA_DIR)
	@rm -rf $(STATE_DIR)
	@rm -rf $(CACHE_DIR)
	@echo "$$(GREEN)✓ Nuked. Run 'make install' for fresh setup.$$(RESET)"
```

## Installation Scripts

### `scripts/install.sh` — Orchestrator

```bash
#!/usr/bin/env bash
# ╔═══════════════════════════════════════════════════════╗
# ║  Anvil — Full Installation                            ║
# ║  Orchestrates all installation steps                   ║
# ╚═══════════════════════════════════════════════════════╝

set -euo pipefail

SCRIPT_DIR="$$(cd "$$(dirname "${BASH_SOURCE[0]}")" && pwd)"
CONFIG_DIR="$$(cd "$$SCRIPT_DIR/.." && pwd)"

# Colors
GREEN='\033[0;32m'
YELLOW='\033[0;33m'
RED='\033[0;31m'
CYAN='\033[0;36m'
RESET='\033[0m'

step() { echo -e "\n$${CYAN}═══ \$1 ═══$${RESET}\n"; }
ok()   { echo -e "$${GREEN}✓ \$1$${RESET}"; }
warn() { echo -e "$${YELLOW}⚠ \$1$${RESET}"; }
err()  { echo -e "$${RED}✗ \$1$${RESET}"; }

# ─── Detect OS ───
detect_os() {
  case "$(uname -s)" in
    Linux*)  OS="linux" ;;
    Darwin*) OS="macos" ;;
    *)       OS="unknown" ;;
  esac

  if [ "$OS" = "linux" ] && grep -qi microsoft /proc/version 2>/dev/null; then
    OS="wsl"
  fi

  echo "$OS"
}

OS=$(detect_os)
echo -e "$${CYAN}Anvil Installation$${RESET}"
echo -e "OS: $${GREEN}$$OS${RESET}"
echo -e "Config: $${GREEN}$$CONFIG_DIR${RESET}"
echo ""

# ─── Pre-flight ───
step "Pre-flight Check"
"$SCRIPT_DIR/health-check.sh" --quick || true

# ─── Step 1: System Dependencies ───
step "Step 1/6: System Dependencies"
"$SCRIPT_DIR/install-deps.sh"

# ─── Step 2: Language Toolchains ───
step "Step 2/6: Language Toolchains"
"$SCRIPT_DIR/install-toolchains.sh"

# ─── Step 3: Neovim Plugins ───
step "Step 3/6: Neovim Plugins (Lazy sync)"
NVIM_APPNAME="${NVIM_APPNAME:-astronvim}" nvim --headless "+Lazy! sync" "+qa" 2>&1 || {
  warn "Plugin sync had warnings (this is normal on first run)"
}
ok "Plugins installed"

# ─── Step 4: Mason Packages ───
step "Step 4/6: Mason Packages"
"$SCRIPT_DIR/install-mason.sh"

# ─── Step 5: MCP Servers ───
step "Step 5/6: MCP Servers"
"$SCRIPT_DIR/install-mcp.sh"

# ─── Step 6: AI Tools (optional) ───
step "Step 6/6: AI Tools (optional)"
read -p "Install external AI tools (aider, goose, opencode, plandex)? [y/N] " -n 1 -r
echo
if [[ $$REPLY =~ ^[Yy]$$ ]]; then
  "$SCRIPT_DIR/install-ai-tools.sh"
else
  warn "Skipped AI tools. Run 'make install-ai' later."
fi

# ─── Fonts ───
if ! fc-list 2>/dev/null | grep -qi "nerd"; then
  echo ""
  read -p "Install Nerd Fonts? [y/N] " -n 1 -r
  echo
  if [[ $$REPLY =~ ^[Yy]$$ ]]; then
    "$SCRIPT_DIR/install-fonts.sh"
  fi
fi

# ─── Done ───
step "Installation Complete"
echo -e "
  $${GREEN}Anvil is ready!$${RESET}

  $${CYAN}Quick start:$${RESET}
    astro                    # launch (alias)
    NVIM_APPNAME=astronvim nvim  # launch (explicit)

  $${CYAN}Verify:$${RESET}
    make health              # shell health check
    :checkhealth anvil       # inside Neovim

  $${CYAN}Configure AI:$${RESET}
    Copy $${CONFIG_DIR}/.env.example to $${CONFIG_DIR}/.env
    Set your API keys

  $${CYAN}Next steps:$${RESET}
    make update-ref          # generate reference files
    :Anvil                   # explore Anvil commands
"
```

### `scripts/install-deps.sh` — System Dependencies

```bash
#!/usr/bin/env bash
# System dependencies (OS-aware)

set -euo pipefail

GREEN='\033[0;32m'
YELLOW='\033[0;33m'
CYAN='\033[0;36m'
RESET='\033[0m'

ok()   { echo -e "$${GREEN}✓ \$1$${RESET}"; }
warn() { echo -e "$${YELLOW}⚠ \$1$${RESET}"; }
info() { echo -e "$${CYAN}→ \$1$${RESET}"; }

# ─── Detect OS ───
case "$(uname -s)" in
  Linux*)  OS="linux" ;;
  Darwin*) OS="macos" ;;
  *)       OS="unknown" ;;
esac

# ─── Detect package manager ───
detect_pkg_manager() {
  if [ "$OS" = "macos" ]; then
    echo "brew"
  elif command -v apt-get &>/dev/null; then
    echo "apt"
  elif command -v dnf &>/dev/null; then
    echo "dnf"
  elif command -v pacman &>/dev/null; then
    echo "pacman"
  elif command -v zypper &>/dev/null; then
    echo "zypper"
  elif command -v apk &>/dev/null; then
    echo "apk"
  elif command -v nix-env &>/dev/null; then
    echo "nix"
  else
    echo "unknown"
  fi
}

PKG=$(detect_pkg_manager)
info "Package manager: $PKG"

# ─── Required packages ───
# Format: "command|package_name_apt|package_name_brew|package_name_pacman"
PACKAGES=(
  "nvim|neovim|neovim|neovim"
  "git|git|git|git"
  "curl|curl|curl|curl"
  "node|nodejs|node|nodejs"
  "npm|npm|node|npm"
  "rg|ripgrep|ripgrep|ripgrep"
  "fd|fd-find|fd|fd"
  "fzf|fzf|fzf|fzf"
  "bat|bat|bat|bat"
  "delta|git-delta|git-delta|git-delta"
  "lazygit|lazygit|lazygit|lazygit"
  "make|make|make|make"
  "unzip|unzip|unzip|unzip"
  "wget|wget|wget|wget"
  "tree|tree|tree|tree"
  "jq|jq|jq|jq"
)

install_package() {
  local cmd="\$1"
  local apt_name="\$2"
  local brew_name="\$3"
  local pacman_name="\$4"

  if command -v "$cmd" &>/dev/null; then
    ok "$cmd already installed"
    return 0
  fi

  info "Installing $cmd..."
  case "$PKG" in
    brew)   brew install "$$brew_name" 2>/dev/null || warn "Failed: $$brew_name" ;;
    apt)    sudo apt-get install -y "$$apt_name" 2>/dev/null || warn "Failed: $$apt_name" ;;
    pacman) sudo pacman -S --noconfirm "$$pacman_name" 2>/dev/null || warn "Failed: $$pacman_name" ;;
    dnf)    sudo dnf install -y "$$apt_name" 2>/dev/null || warn "Failed: $$apt_name" ;;
    *)      warn "Cannot install $cmd — unknown package manager" ;;
  esac
}

# ─── Install Homebrew if macOS and missing ───
if [ "$OS" = "macos" ] && ! command -v brew &>/dev/null; then
  info "Installing Homebrew..."
  /bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/HEAD/install.sh)"
fi

# ─── Install lazygit (special handling) ───
install_lazygit() {
  if command -v lazygit &>/dev/null; then
    ok "lazygit already installed"
    return
  fi

  case "$PKG" in
    brew)   brew install lazygit ;;
    pacman) sudo pacman -S --noconfirm lazygit ;;
    apt)
      # lazygit not in default apt repos
      LAZYGIT_VERSION=$(curl -s "https://api.github.com/repos/jesseduffield/lazygit/releases/latest" | jq -r '.tag_name' | sed 's/^v//')
      curl -Lo /tmp/lazygit.tar.gz "https://github.com/jesseduffield/lazygit/releases/latest/download/lazygit_${LAZYGIT_VERSION}_Linux_x86_64.tar.gz"
      tar xf /tmp/lazygit.tar.gz -C /tmp lazygit
      sudo install /tmp/lazygit /usr/local/bin
      rm -f /tmp/lazygit /tmp/lazygit.tar.gz
      ;;
    *)
      if command -v go &>/dev/null; then
        go install github.com/jesseduffield/lazygit@latest
      else
        warn "Cannot install lazygit — install manually"
      fi
      ;;
  esac
}

# ─── Install all ───
for pkg_spec in "${PACKAGES[@]}"; do
  IFS='|' read -r cmd apt_name brew_name pacman_name <<< "$pkg_spec"
  if [ "$cmd" = "lazygit" ]; then
    install_lazygit
  else
    install_package "$$cmd" "$$apt_name" "$$brew_name" "$$pacman_name"
  fi
done

# ─── Verify Neovim version ───
if command -v nvim &>/dev/null; then
  NVIM_VERSION=$(nvim --version | head -1 | grep -oP '\d+\.\d+\.\d+')
  NVIM_MAJOR=$$(echo "$$NVIM_VERSION" | cut -d. -f1)
  NVIM_MINOR=$$(echo "$$NVIM_VERSION" | cut -d. -f2)
  if [ "$$NVIM_MAJOR" -eq 0 ] && [ "$$NVIM_MINOR" -lt 10 ]; then
    warn "Neovim $NVIM_VERSION detected — Anvil requires 0.10+"
    warn "Please update Neovim: https://github.com/neovim/neovim/releases"
  else
    ok "Neovim $NVIM_VERSION"
  fi
fi

echo ""
ok "System dependencies installed"
```

### `scripts/install-toolchains.sh` — Language Toolchains

```bash
#!/usr/bin/env bash
# Language toolchains: Rust, Go, Python, Lua, Node

set -euo pipefail

GREEN='\033[0;32m'
YELLOW='\033[0;33m'
CYAN='\033[0;36m'
RESET='\033[0m'

ok()   { echo -e "$${GREEN}✓ \$1$${RESET}"; }
warn() { echo -e "$${YELLOW}⚠ \$1$${RESET}"; }
info() { echo -e "$${CYAN}→ \$1$${RESET}"; }

# ═══════════════════════════════════════
# Rust
# ═══════════════════════════════════════
install_rust() {
  info "Rust toolchain"

  if command -v rustup &>/dev/null; then
    ok "rustup installed ($(rustc --version 2>/dev/null | awk '{print \$2}'))"
    info "Updating Rust..."
    rustup update stable 2>/dev/null || true
  else
    info "Installing Rust via rustup..."
    curl --proto '=https' --tlsv1.2 -sSf https://sh.rustup.rs | sh -s -- -y --default-toolchain stable
    source "$HOME/.cargo/env" 2>/dev/null || true
  fi

  # Components
  rustup component add rust-analyzer rustfmt clippy rust-src 2>/dev/null || true
  ok "Rust components: rust-analyzer, rustfmt, clippy, rust-src"

  # Cargo tools
  local cargo_tools=(
    "cargo-edit"
    "cargo-watch"
    "cargo-expand"
    "cargo-audit"
    "cargo-nextest"
  )
  for tool in "${cargo_tools[@]}"; do
    if cargo install --list 2>/dev/null | grep -q "^$tool "; then
      ok "$tool installed"
    else
      info "Installing $tool..."
      cargo install "$$tool" 2>/dev/null || warn "Failed to install $$tool"
    fi
  done
}

# ═══════════════════════════════════════
# Go
# ═══════════════════════════════════════
install_go() {
  info "Go toolchain"

  if command -v go &>/dev/null; then
    ok "Go installed ($(go version | awk '{print \$3}'))"
  else
    warn "Go not found — install from https://go.dev/doc/install"
    return
  fi

  # Go tools
  local go_tools=(
    "golang.org/x/tools/gopls@latest"
    "mvdan.cc/gofumpt@latest"
    "golang.org/x/tools/cmd/goimports@latest"
    "github.com/cweill/gotests/gotests@latest"
    "github.com/fatih/gomodifytags@latest"
    "github.com/josharian/impl@latest"
    "github.com/go-delve/delve/cmd/dlv@latest"
    "golang.org/x/vuln/cmd/govulncheck@latest"
  )
  for tool in "${go_tools[@]}"; do
    local bin_name
    bin_name=$$(basename "$${tool%%@*}")
    if command -v "$bin_name" &>/dev/null; then
      ok "$bin_name installed"
    else
      info "Installing $bin_name..."
      go install "$$tool" 2>/dev/null || warn "Failed: $$tool"
    fi
  done

  # golangci-lint (special install)
  if command -v golangci-lint &>/dev/null; then
    ok "golangci-lint installed"
  else
    info "Installing golangci-lint..."
    curl -sSfL https://raw.githubusercontent.com/golangci/golangci-lint/master/install.sh | \
      sh -s -- -b "$(go env GOPATH)/bin" 2>/dev/null || warn "Failed: golangci-lint"
  fi
}

# ═══════════════════════════════════════
# Python
# ═══════════════════════════════════════
install_python() {
  info "Python toolchain"

  if command -v python3 &>/dev/null; then
    ok "Python installed ($(python3 --version 2>&1 | awk '{print \$2}'))"
  else
    warn "Python 3 not found — install from https://www.python.org/"
    return
  fi

  # pipx
  if ! command -v pipx &>/dev/null; then
    info "Installing pipx..."
    python3 -m pip install --user pipx 2>/dev/null || true
    python3 -m pipx ensurepath 2>/dev/null || true
  fi
  ok "pipx available"

  # Python tools via pipx
  local py_tools=("ruff" "mypy" "ipython")
  for tool in "${py_tools[@]}"; do
    if command -v "$tool" &>/dev/null; then
      ok "$tool installed"
    else
      info "Installing $tool..."
      pipx install "$tool" 2>/dev/null || \
        python3 -m pip install --user "$tool" 2>/dev/null || \
        warn "Failed: $tool"
    fi
  done

  # uv (fast package installer)
  if command -v uv &>/dev/null; then
    ok "uv installed"
  else
    info "Installing uv..."
    curl -LsSf https://astral.sh/uv/install.sh | sh 2>/dev/null || warn "Failed: uv"
  fi
}

# ═══════════════════════════════════════
# Lua
# ═══════════════════════════════════════
install_lua() {
  info "Lua toolchain"

  if command -v lua &>/dev/null || command -v luajit &>/dev/null; then
    local lua_ver
    lua_ver=$(lua -v 2>&1 | head -1 || luajit -v 2>&1 | head -1 || echo "unknown")
    ok "Lua installed ($lua_ver)"
  else
    warn "Lua/LuaJIT not found — install via package manager"
  fi

  # stylua
  if command -v stylua &>/dev/null; then
    ok "stylua installed"
  else
    info "Installing stylua..."
    if command -v cargo &>/dev/null; then
      cargo install stylua 2>/dev/null || warn "Failed: stylua"
    else
      warn "Install stylua manually: cargo install stylua"
    fi
  fi

  # selene
  if command -v selene &>/dev/null; then
    ok "selene installed"
  else
    info "Installing selene..."
    if command -v cargo &>/dev/null; then
      cargo install selene 2>/dev/null || warn "Failed: selene"
    else
      warn "Install selene manually: cargo install selene"
    fi
  fi

  # luarocks + busted
  if command -v luarocks &>/dev/null; then
    ok "luarocks installed"
    if luarocks show busted &>/dev/null 2>&1; then
      ok "busted installed"
    else
      info "Installing busted..."
      luarocks install busted 2>/dev/null || warn "Failed: busted"
    fi
  else
    warn "luarocks not found — install via package manager"
  fi
}

# ═══════════════════════════════════════
# Node.js
# ═══════════════════════════════════════
install_node() {
  info "Node.js toolchain"

  if command -v node &>/dev/null; then
    ok "Node.js installed ($(node --version))"
  else
    warn "Node.js not found — install from https://nodejs.org/"
    return
  fi

  if command -v npm &>/dev/null; then
    ok "npm installed ($(npm --version))"
  fi

  # Global npm packages (minimal)
  local npm_tools=("neovim" "tree-sitter-cli")
  for tool in "${npm_tools[@]}"; do
    if npm list -g "$tool" &>/dev/null; then
      ok "npm: $tool installed"
    else
      info "Installing npm: $tool..."
      npm install -g "$$tool" 2>/dev/null || warn "Failed: npm $$tool"
    fi
  done
}

# ═══════════════════════════════════════
# Run all
# ═══════════════════════════════════════
install_rust
echo ""
install_go
echo ""
install_python
echo ""
install_lua
echo ""
install_node

echo ""
ok "Language toolchains installed"
```

### `scripts/install-mason.sh` — Mason Packages

```bash
#!/usr/bin/env bash
# Install Mason packages via headless Neovim

set -euo pipefail

CYAN='\033[0;36m'
GREEN='\033[0;32m'
YELLOW='\033[0;33m'
RESET='\033[0m'

NVIM_APPNAME="${NVIM_APPNAME:-astronvim}"

echo -e "$${CYAN}→ Installing Mason packages...$${RESET}"
echo -e "$${YELLOW}  This runs Neovim headless and may take a few minutes.$${RESET}"

# MasonToolsInstall will install everything from mason-tool-installer
```

```lua
  }

  for _, t in ipairs(ai_tools) do
    if vim.fn.executable(t[1]) == 1 then
      health.ok(t[2])
    else
      health.info(t[2] .. " not installed (optional)")
    end
  end

  -- MCP
  local mcp_config = vim.fn.expand("~/.config/mcphub/servers.json")
  if vim.fn.filereadable(mcp_config) == 1 then
    health.ok("MCP config: " .. mcp_config)
    local config_ok, config = pcall(function()
      local content = table.concat(vim.fn.readfile(mcp_config), "\n")
      return vim.json.decode(content)
    end)
    if config_ok and config and config.mcpServers then
      local count = vim.tbl_count(config.mcpServers)
      health.ok(count .. " MCP server(s) configured")
    end
  else
    health.info("MCP config not found (run 'make install-mcp')")
  end

  -- ═══════════════════════════════════════
  -- Mason
  -- ═══════════════════════════════════════
  health.start("Anvil Mason")

  local mason_ok, mason_registry = pcall(require, "mason-registry")
  if mason_ok then
    local installed = mason_registry.get_installed_packages()
    health.ok(#installed .. " Mason packages installed")

    -- Check critical packages
    local critical_mason = {
      "lua-language-server",
      "stylua",
    }

    -- Add from enabled lang packs
    if lang_ok then
      for _, name in ipairs(lang.enabled()) do
        local pack = lang.get(name)
        if pack and pack.lsp and not pack.lsp.managed_by_plugin then
          table.insert(critical_mason, pack.lsp.server)
        end
      end
    end

    for _, pkg_name in ipairs(critical_mason) do
      local pkg_ok, pkg = pcall(mason_registry.get_package, pkg_name)
      if pkg_ok and pkg:is_installed() then
        health.ok("Mason: " .. pkg_name)
      else
        health.warn("Mason: " .. pkg_name .. " not installed", { ":MasonInstall " .. pkg_name })
      end
    end
  else
    health.warn("Mason registry not available")
  end
end

return M
```
