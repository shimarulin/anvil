# Anvil — Git Layer

Complete Git integration: inline signs, blame, diff viewer, conflict
resolution, worktrees, and a full TUI client — all without leaving Neovim.

## Architecture

```
lua/anvil/git/
├── init.lua        # git subsystem setup
├── signs.lua       # gitsigns configuration
├── diff.lua        # diffview configuration
├── conflict.lua    # conflict resolution
└── worktree.lua    # git worktrees

lua/plugins/
└── git.lua         # all git plugin specs
```

### Integration Points

```
┌─────────────────────────────────────────────────────────────┐
│                       Git Layer                              │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  gitsigns ──→ sign column (add/change/delete indicators)    │
│           ──→ inline blame (current line)                   │
│           ──→ hunk operations (stage, reset, preview)       │
│           ──→ hunk navigation (]h / [h)                     │
│           ──→ buffer diff                                   │
│           ──→ text object (ih = inner hunk)                 │
│                                                             │
│  lazygit  ──→ full TUI git client (via toggleterm)          │
│                                                             │
│  neogit   ──→ magit-style git client (native nvim)          │
│                                                             │
│  diffview ──→ side-by-side diff viewer                      │
│           ──→ file history                                  │
│           ──→ merge tool                                    │
│                                                             │
│  git-conflict ──→ conflict markers highlighting             │
│               ──→ choose ours/theirs/both/none              │
│                                                             │
│  git-worktree ──→ manage worktrees from telescope           │
│                                                             │
│  AI Layer ──→ AIReviewStaged (reads git diff)               │
│           ──→ AICommit (reads git diff --cached)            │
│           ──→ AIExplainCommit (reads git log + diff)        │
│           ──→ /diff slash command (reads git diff HEAD)     │
│                                                             │
│  core/icons ──→ git sign symbols                            │
│             ──→ neo-tree git status symbols                 │
└─────────────────────────────────────────────────────────────┘
```

```lua
--- Anvil gitsigns configuration
--- Inline git status, blame, hunk operations
---@class AnvilGitSigns
local M = {}

---@return LazyPluginSpec
function M.spec()
  return {
    "lewis6991/gitsigns.nvim",
    event = { "BufReadPre", "BufNewFile" },
    opts = {
      -- ─── Sign column indicators ───
      signs = {
        add          = { text = "▎" },
        change       = { text = "▎" },
        delete       = { text = "" },
        topdelete    = { text = "" },
        changedelete = { text = "▎" },
        untracked    = { text = "▎" },
      },
      signs_staged = {
        add          = { text = "▎" },
        change       = { text = "▎" },
        delete       = { text = "" },
        topdelete    = { text = "" },
        changedelete = { text = "▎" },
      },

      -- ─── Inline blame ───
      current_line_blame = true,
      current_line_blame_opts = {
        virt_text = true,
        virt_text_pos = "eol",
        delay = 300,
        ignore_whitespace = true,
      },
      current_line_blame_formatter = "<author>, <author_time:%R> — <summary>",

      -- ─── Performance ───
      max_file_length = 40000,
      watch_gitdir = { follow_files = true },

      -- ─── Keymaps via on_attach ───
      on_attach = function(bufnr)
        local gs = package.loaded.gitsigns

        local function map(mode, l, r, opts)
          opts = opts or {}
          opts.buffer = bufnr
          vim.keymap.set(mode, l, r, opts)
        end

        -- ── Navigation ──
        map("n", "]h", function()
          if vim.wo.diff then return "]c" end
          vim.schedule(function() gs.next_hunk() end)
          return "<Ignore>"
        end, { expr = true, desc = "Next hunk" })

        map("n", "[h", function()
          if vim.wo.diff then return "[c" end
          vim.schedule(function() gs.prev_hunk() end)
          return "<Ignore>"
        end, { expr = true, desc = "Previous hunk" })

        -- ── Stage / Reset ──
        map("n", "<leader>gs", gs.stage_hunk,  { desc = "Stage hunk" })
        map("n", "<leader>gr", gs.reset_hunk,  { desc = "Reset hunk" })

        map("v", "<leader>gs", function()
          gs.stage_hunk({ vim.fn.line("."), vim.fn.line("v") })
        end, { desc = "Stage hunk (visual)" })

        map("v", "<leader>gr", function()
          gs.reset_hunk({ vim.fn.line("."), vim.fn.line("v") })
        end, { desc = "Reset hunk (visual)" })

        -- ── Buffer operations ──
        map("n", "<leader>gS", gs.stage_buffer,      { desc = "Stage buffer" })
        map("n", "<leader>gR", gs.reset_buffer,      { desc = "Reset buffer" })
        map("n", "<leader>gu", gs.undo_stage_hunk,   { desc = "Undo stage hunk" })

        -- ── Preview ──
        map("n", "<leader>gp", gs.preview_hunk,        { desc = "Preview hunk" })
        map("n", "<leader>gP", gs.preview_hunk_inline,  { desc = "Preview hunk inline" })

        -- ── Blame ──
        map("n", "<leader>gb", function()
          gs.blame_line({ full = true })
        end, { desc = "Blame line (full)" })

        map("n", "<leader>gB", gs.toggle_current_line_blame,
          { desc = "Toggle line blame" })

        -- ── Diff ──
        map("n", "<leader>gd", gs.diffthis,   { desc = "Diff this" })
        map("n", "<leader>gD", function()
          gs.diffthis("~")
        end, { desc = "Diff this (~)" })

        -- ── Text object ──
        map({ "o", "x" }, "ih", ":<C-U>Gitsigns select_hunk<CR>",
          { desc = "Select hunk" })
      end,
    },
  }
end

return M
```

```lua
--- Anvil git conflict resolution
---@class AnvilGitConflict
local M = {}

---@return LazyPluginSpec
function M.spec()
  return {
    "akinsho/git-conflict.nvim",
    version = "*",
    event = "BufReadPre",
    opts = {
      default_mappings = false, -- we define our own below
      default_commands = true,
      disable_diagnostics = true,
      list_opener = "copen",
      highlights = {
        incoming = "DiffAdd",
        current  = "DiffText",
        ancestor = "DiffDelete",
      },
    },
    config = function(_, opts)
      require("git-conflict").setup(opts)

      -- ─── Keymaps (only active in conflict buffers) ───
      vim.api.nvim_create_autocmd("User", {
        pattern = "GitConflictDetected",
        callback = function(args)
          local bufnr = args.buf or vim.api.nvim_get_current_buf()
          local map = function(lhs, rhs, desc)
            vim.keymap.set("n", lhs, rhs, { buffer = bufnr, desc = desc })
          end

          map("<leader>gco", "<Plug>(git-conflict-ours)",          "Conflict: Choose ours")
          map("<leader>gct", "<Plug>(git-conflict-theirs)",        "Conflict: Choose theirs")
          map("<leader>gcb", "<Plug>(git-conflict-both)",          "Conflict: Choose both")
          map("<leader>gc0", "<Plug>(git-conflict-none)",          "Conflict: Choose none")
          map("]x",          "<Plug>(git-conflict-next-conflict)", "Next conflict")
          map("[x",          "<Plug>(git-conflict-prev-conflict)", "Previous conflict")

          vim.notify(
            "Git conflict detected — use <leader>gc{o,t,b,0} to resolve",
            vim.log.levels.WARN,
            { title = "Git Conflict" }
          )
        end,
      })
    end,
  }
end

return M
```

## Git Subsystem Init

### `lua/anvil/git/init.lua`

<!-- file: lua/anvil/git/init.lua -->
```lua
--- Anvil git subsystem
---@class AnvilGit
local M = {}

--- Check if current directory is a git repository
---@return boolean
function M.is_repo()
  local result = vim.fn.system("git rev-parse --is-inside-work-tree 2>/dev/null")
  return vim.v.shell_error == 0 and vim.trim(result) == "true"
end

--- Get current branch name
---@return string
function M.branch()
  if not M.is_repo() then return "" end
  local result = vim.fn.system("git branch --show-current 2>/dev/null")
  return vim.trim(result)
end

--- Get short HEAD hash
---@return string
function M.head_short()
  if not M.is_repo() then return "" end
  local result = vim.fn.system("git rev-parse --short HEAD 2>/dev/null")
  return vim.trim(result)
end

--- Get number of uncommitted changes
---@return { staged: number, unstaged: number, untracked: number }
function M.status_counts()
  if not M.is_repo() then
    return { staged = 0, unstaged = 0, untracked = 0 }
  end

  local status = vim.fn.system("git status --porcelain 2>/dev/null")
  local staged, unstaged, untracked = 0, 0, 0

  for line in status:gmatch("[^\n]+") do
    local x, y = line:sub(1, 1), line:sub(2, 2)
    if x == "?" then
      untracked = untracked + 1
    else
      if x ~= " " and x ~= "?" then staged = staged + 1 end
      if y ~= " " and y ~= "?" then unstaged = unstaged + 1 end
    end
  end

  return { staged = staged, unstaged = unstaged, untracked = untracked }
end

--- Get staged diff (for AI review)
---@return string
function M.staged_diff()
  if not M.is_repo() then return "" end
  return vim.fn.system("git diff --cached 2>/dev/null")
end

--- Get working tree diff (for AI context)
---@param max_chars? number truncate after this many chars
---@return string
function M.working_diff(max_chars)
  if not M.is_repo() then return "" end
  local diff = vim.fn.system("git diff HEAD 2>/dev/null")
  max_chars = max_chars or 15000
  if #diff > max_chars then
    diff = diff:sub(1, max_chars) .. "\n\n... (truncated at " .. max_chars .. " chars)"
  end
  return diff
end

--- Get commit info
---@param ref? string git ref (default: HEAD)
---@return { hash: string, subject: string, body: string, author: string, date: string }
function M.commit_info(ref)
  ref = ref or "HEAD"
  if not M.is_repo() then
    return { hash = "", subject = "", body = "", author = "", date = "" }
  end

  local fmt = "%H%n%s%n%b%n%an%n%ar"
  local result = vim.fn.system(
    string.format("git log -1 --format='%s' %s 2>/dev/null", fmt, ref)
  )
  local lines = vim.split(vim.trim(result), "\n")

  return {
    hash    = lines[1] or "",
    subject = lines[2] or "",
    body    = table.concat(vim.list_slice(lines, 3, #lines - 2), "\n"),
    author  = lines[#lines - 1] or "",
    date    = lines[#lines] or "",
  }
end

return M
```

## Workflow Examples

### Workflow: Review changes before commit

```
1. Make changes to files
2. <leader>gv         → Diffview opens, shows all changes
   - Navigate files in left panel
   - Review diffs side-by-side
   - Press 's' to stage individual files
3. <leader>gV         → Close diffview
4. <leader>aGr        → AI reviews staged changes
5. Fix any issues AI found
6. <leader>aGc        → AI generates commit message
7. <leader>gc         → Neogit commit (paste AI message)
```

### Workflow: Resolve merge conflicts

```
1. git merge feature-branch   (in terminal, creates conflicts)
2. Open conflicted file
   → git-conflict highlights markers automatically
   → notification: "Git conflict detected"
3. ]x / [x            → navigate between conflicts
4. <leader>gco        → choose ours
   <leader>gct        → choose theirs
   <leader>gcb        → choose both
   <leader>gc0        → choose none (delete both)
5. Repeat for all conflicts
6. <leader>gS         → stage buffer
7. <leader>gc         → commit
```

### Workflow: Compare feature branch with main

```
1. <leader>gm         → Diffview: compare HEAD with main
   - See all files that differ
   - Review each change side-by-side
2. <leader>gF         → Branch history (all commits)
3. <leader>gV         → Close when done
```

### Workflow: Work with git worktrees

```
1. <leader>gW         → Create worktree
   - Enter branch name and path
   - New worktree created
2. <leader>gw         → List worktrees (telescope)
   - Select a worktree to switch to
   - Neovim changes directory
3. Work in the worktree independently
4. <leader>gw         → Switch back
```
