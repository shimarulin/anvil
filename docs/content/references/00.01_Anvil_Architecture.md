# Anvil — Architecture & Core

Anvil is a comprehensive AstroNvim user configuration where AI is a first-class
subsystem alongside LSP, languages, debugging, UI, and workflow.

## Philosophy

```
┌─────────────────────────────────────────────────────┐
│                      ANVIL                          │
│           "One config to rule them all"             │
│                                                     │
│  1. Language-first  — language defines everything   │
│  2. Consistency     — same patterns everywhere      │
│  3. AI-native       — AI is woven into every layer  │
│  4. Portable        — works on Linux and macOS      │
│  5. Declarative     — describe WHAT, not HOW        │
│  6. Discoverable    — find everything via :Anvil    │
└─────────────────────────────────────────────────────┘
```

## Core Layer

### `lua/anvil/core/events.lua`

```lua
--- Anvil event bus
--- Lightweight pub/sub for decoupled communication between layers
---@class AnvilEvents
local M = {}

---@type table<string, function[]>
M._listeners = {}

--- Subscribe to an event
---@param event string event name (e.g. "anvil:ready", "lang:attached")
---@param callback function(data: any)
---@return function unsubscribe function
function M.on(event, callback)
  if not M._listeners[event] then
    M._listeners[event] = {}
  end
  table.insert(M._listeners[event], callback)

  -- Return unsubscribe function
  return function()
    M._listeners[event] = vim.tbl_filter(function(cb)
      return cb ~= callback
    end, M._listeners[event] or {})
  end
end

--- Subscribe to an event (fires only once)
---@param event string
---@param callback function(data: any)
function M.once(event, callback)
  local unsub
  unsub = M.on(event, function(data)
    unsub()
    callback(data)
  end)
end

--- Emit an event
---@param event string
---@param data? any
function M.emit(event, data)
  for _, callback in ipairs(M._listeners[event] or {}) do
    local ok, err = pcall(callback, data)
    if not ok then
      -- Avoid circular dependency: use vim.notify directly
      vim.notify(
        string.format("[Anvil:Events] Error in '%s' handler: %s", event, err),
        vim.log.levels.ERROR
      )
    end
  end
end

--- Remove all listeners for an event (or all events)
---@param event? string if nil, clears everything
function M.clear(event)
  if event then
    M._listeners[event] = nil
  else
    M._listeners = {}
  end
end

--- List registered events (for debugging)
---@return table<string, number> event name → listener count
function M.list()
  local result = {}
  for event, listeners in pairs(M._listeners) do
    result[event] = #listeners
  end
  return result
end

--- Setup: no-op for now, exists for boot() consistency
function M.setup() end

return M
```

### `lua/anvil/core/logger.lua`

```lua
--- Anvil structured logger
--- Provides leveled logging with optional file output
---@class AnvilLogger
local M = {}

---@class AnvilLogConfig
---@field level? string "debug"|"info"|"warn"|"error" (default: "info")
---@field to_file? boolean write to log file (default: false)
---@field file_path? string log file path
---@field prefix? string log prefix (default: "Anvil")

---@type AnvilLogConfig
M._config = {
  level = "info",
  to_file = false,
  file_path = nil,
  prefix = "Anvil",
}

local LEVELS = {
  debug = { priority = 0, vim_level = vim.log.levels.DEBUG },
  info  = { priority = 1, vim_level = vim.log.levels.INFO },
  warn  = { priority = 2, vim_level = vim.log.levels.WARN },
  error = { priority = 3, vim_level = vim.log.levels.ERROR },
}

---@param config? AnvilLogConfig
function M.setup(config)
  M._config = vim.tbl_deep_extend("force", M._config, config or {})
  if M._config.to_file and not M._config.file_path then
    M._config.file_path = vim.fn.stdpath("log") .. "/anvil.log"
  end
end

---@param level string
---@param msg string
---@param ... any format arguments
local function log(level, msg, ...)
  local level_info = LEVELS[level]
  local config_level = LEVELS[M._config.level]

  if not level_info or not config_level then return end
  if level_info.priority < config_level.priority then return end

  -- Format message
  local formatted
  if select("#", ...) > 0 then
    local ok, result = pcall(string.format, msg, ...)
    formatted = ok and result or (msg .. " " .. vim.inspect({ ... }))
  else
    formatted = msg
  end

  local full_msg = string.format("[%s] %s", M._config.prefix, formatted)

  -- Notify in nvim
  vim.schedule(function()
    vim.notify(full_msg, level_info.vim_level, { title = M._config.prefix })
  end)

  -- Write to file
  if M._config.to_file and M._config.file_path then
    local timestamp = os.date("%Y-%m-%d %H:%M:%S")
    local line = string.format("[%s] [%s] %s\n", timestamp, level:upper(), formatted)
    local file = io.open(M._config.file_path, "a")
    if file then
      file:write(line)
      file:close()
    end
  end
end

function M.debug(msg, ...) log("debug", msg, ...) end
function M.info(msg, ...)  log("info", msg, ...) end
function M.warn(msg, ...)  log("warn", msg, ...) end
function M.error(msg, ...) log("error", msg, ...) end

return M
```

### `lua/anvil/core/cache.lua`

```lua
--- Anvil cache with TTL support
--- Simple in-memory key-value store with automatic expiration
---@class AnvilCache
local M = {}

---@class AnvilCacheEntry
---@field value any
---@field expires_at number timestamp (0 = never expires)

---@type table<string, AnvilCacheEntry>
M._store = {}

--- Get a value from cache
---@param key string
---@return any? value nil if not found or expired
function M.get(key)
  local entry = M._store[key]
  if not entry then return nil end

  if entry.expires_at > 0 and os.time() > entry.expires_at then
    M._store[key] = nil
    return nil
  end

  return entry.value
end

--- Set a value in cache
---@param key string
---@param value any
---@param ttl_seconds? number time to live in seconds (nil = forever)
function M.set(key, value, ttl_seconds)
  M._store[key] = {
    value = value,
    expires_at = ttl_seconds and (os.time() + ttl_seconds) or 0,
  }
end

--- Delete a key from cache
---@param key string
function M.del(key)
  M._store[key] = nil
end

--- Check if key exists and is not expired
---@param key string
---@return boolean
function M.has(key)
  return M.get(key) ~= nil
end

--- Get or compute: returns cached value, or calls fn() and caches the result
---@param key string
---@param fn function(): any
---@param ttl_seconds? number
---@return any
function M.get_or_set(key, fn, ttl_seconds)
  local value = M.get(key)
  if value ~= nil then return value end

  value = fn()
  M.set(key, value, ttl_seconds)
  return value
end

--- Remove all expired entries
function M.cleanup()
  local now = os.time()
  for key, entry in pairs(M._store) do
    if entry.expires_at > 0 and now > entry.expires_at then
      M._store[key] = nil
    end
  end
end

--- Clear entire cache
function M.clear()
  M._store = {}
end

--- Get cache statistics
---@return { total: number, expired: number }
function M.stats()
  local total = 0
  local expired = 0
  local now = os.time()

  for _, entry in pairs(M._store) do
    total = total + 1
    if entry.expires_at > 0 and now > entry.expires_at then
      expired = expired + 1
    end
  end

  return { total = total, expired = expired }
end

return M
```

### `lua/anvil/core/env.lua`

```lua
--- Anvil environment variable loader
--- Loads .env files with proper parsing (quotes, comments, empty lines)
---@class AnvilEnv
local M = {}

---@type table<string, string>
M._vars = {}

---@type boolean
M._loaded = false

--- Parse a .env file and return key-value pairs
---@param filepath string
---@return table<string, string>
local function parse_dotenv(filepath)
  local vars = {}

  if vim.fn.filereadable(filepath) ~= 1 then
    return vars
  end

  for line in io.lines(filepath) do
    -- Skip empty lines and comments
    if not line:match("^%s*$") and not line:match("^%s*#") then
      local key, value = line:match("^([%w_]+)%s*=%s*(.*)$")
      if key and value then
        -- Strip surrounding quotes
        value = value:match('^"(.*)"$')
          or value:match("^'(.*)'$")
          or value

        -- Strip inline comments (only outside quotes)
        if not value:match('^"') and not value:match("^'") then
          value = value:match("^(.-)%s+#") or value
        end

        value = vim.trim(value)
        vars[key] = value
      end
    end
  end

  return vars
end

--- Load environment variables from .env files
--- Priority: OS env > project .env > global .env > defaults
function M.load()
  if M._loaded then return end

  -- Global .env (in nvim config directory)
  local global_env = vim.fn.stdpath("config") .. "/.env"
  local global_vars = parse_dotenv(global_env)

  -- Project .env (in current working directory)
  local project_env = vim.fn.getcwd() .. "/.env"
  local project_vars = parse_dotenv(project_env)

  -- Merge: project overrides global
  M._vars = vim.tbl_extend("force", global_vars, project_vars)
  M._loaded = true
end

--- Get an environment variable
--- Priority: OS env > loaded .env > default
---@param name string
---@param default? string
---@return string
function M.get(name, default)
  -- OS environment has highest priority
  local os_val = os.getenv(name)
  if os_val and os_val ~= "" then
    return os_val
  end

  -- Then .env files
  local env_val = M._vars[name]
  if env_val and env_val ~= "" then
    return env_val
  end

  return default or ""
end

--- Get a numeric environment variable
---@param name string
---@param default? number
---@return number
function M.get_number(name, default)
  local val = M.get(name)
  return tonumber(val) or default or 0
end

--- Get a boolean environment variable
---@param name string
---@param default? boolean
---@return boolean
function M.get_bool(name, default)
  local val = M.get(name):lower()
  if val == "true" or val == "1" or val == "yes" then return true end
  if val == "false" or val == "0" or val == "no" then return false end
  return default or false
end

--- Check if a variable is set (in any source)
---@param name string
---@return boolean
function M.has(name)
  return M.get(name) ~= ""
end

--- Reload .env files
function M.reload()
  M._loaded = false
  M._vars = {}
  M.load()
end

return M
```

### `lua/anvil/core/autocmds.lua`

```lua
--- Anvil global autocommands
---@class AnvilAutocmds
local M = {}

function M.setup()
  local augroup = vim.api.nvim_create_augroup("Anvil", { clear = true })

  -- Highlight on yank
  vim.api.nvim_create_autocmd("TextYankPost", {
    group = augroup,
    desc = "Highlight on yank",
    callback = function()
      vim.highlight.on_yank({ higroup = "IncSearch", timeout = 200 })
    end,
  })

  -- Resize splits on window resize
  vim.api.nvim_create_autocmd("VimResized", {
    group = augroup,
    desc = "Equalize splits on resize",
    callback = function()
      vim.cmd("tabdo wincmd =")
    end,
  })

  -- Remove trailing whitespace on save
  vim.api.nvim_create_autocmd("BufWritePre", {
    group = augroup,
    desc = "Remove trailing whitespace",
    pattern = "*",
    callback = function()
      -- Skip certain filetypes
      local skip_ft = { "markdown", "diff", "gitcommit" }
      if vim.tbl_contains(skip_ft, vim.bo.filetype) then return end

      local cursor = vim.api.nvim_win_get_cursor(0)
      vim.cmd([[%s/\s\+$//e]])
      vim.api.nvim_win_set_cursor(0, cursor)
    end,
  })

  -- Return to last edit position
  vim.api.nvim_create_autocmd("BufReadPost", {
    group = augroup,
    desc = "Return to last edit position",
    callback = function()
      local mark = vim.api.nvim_buf_get_mark(0, '"')
      local line_count = vim.api.nvim_buf_line_count(0)
      if mark[1] > 0 and mark[1] <= line_count then
        pcall(vim.api.nvim_win_set_cursor, 0, mark)
      end
    end,
  })

  -- Close certain filetypes with q
  vim.api.nvim_create_autocmd("FileType", {
    group = augroup,
    desc = "Close with q",
    pattern = {
      "help", "lspinfo", "man", "notify", "qf",
      "startuptime", "checkhealth", "neotest-output",
      "neotest-summary", "neotest-output-panel",
    },
    callback = function(event)
      vim.bo[event.buf].buflisted = false
      vim.keymap.set("n", "q", "<cmd>close<cr>", { buffer = event.buf, silent = true })
    end,
  })

  -- Auto-create parent directories on save
  vim.api.nvim_create_autocmd("BufWritePre", {
    group = augroup,
    desc = "Create parent directories",
    callback = function(event)
      if event.match:match("^%w%w+://") then return end
      local file = vim.uv.fs_realpath(event.match) or event.match
      vim.fn.mkdir(vim.fn.fnamemodify(file, ":p:h"), "p")
    end,
  })
```

### Language Engine: `lua/anvil/lang/init.lua`

```lua
--- Anvil language engine
--- Registry, aggregation, and plugin spec generation for all language packs
---@class AnvilLang
local M = {}

---@type table<string, AnvilLangPack>
M._registry = {}

---@type string[]
M._enabled = {}

-- ═══════════════════════════════════════════════════
-- Helpers
-- ═══════════════════════════════════════════════════

--- Deduplicate a list of strings
---@param list string[]
---@return string[]
local function dedup(list)
  local seen = {}
  local result = {}
  for _, item in ipairs(list) do
    if not seen[item] then
      seen[item] = true
      result[#result + 1] = item
    end
  end
  return result
end

-- ═══════════════════════════════════════════════════
-- Setup & Registry
-- ═══════════════════════════════════════════════════

--- Register all available language packs
---@param enabled_langs? string[] list of languages to enable (nil = all)
function M.setup(enabled_langs)
  local lang_dir = vim.fn.stdpath("config") .. "/lua/anvil/lang"
  local files = vim.fn.glob(lang_dir .. "/*.lua", false, true)

  for _, file in ipairs(files) do
    local name = vim.fn.fnamemodify(file, ":t:r")
    if name ~= "init" and name ~= "_template" then
      local ok, pack = pcall(require, "anvil.lang." .. name)
      if ok and type(pack) == "table" and pack.name then
        M._registry[pack.name] = pack
      elseif not ok then
        vim.notify(
          string.format("[Anvil:Lang] Failed to load '%s': %s", name, pack),
          vim.log.levels.WARN
        )
      end
    end
  end

  -- Enable requested or all
  M._enabled = enabled_langs or vim.tbl_keys(M._registry)

  local logger_ok, logger = pcall(require, "anvil.core.logger")
  if logger_ok then
    logger.info(
      "Registered %d languages, %d enabled",
      vim.tbl_count(M._registry),
      #M._enabled
    )
  end
end

--- Get a language pack by name
---@param name string
---@return AnvilLangPack?
function M.get(name)
  return M._registry[name]
end

--- Get the language pack for the current buffer's filetype
---@return AnvilLangPack?
function M.current()
  local ft = vim.bo.filetype
  for _, pack in pairs(M._registry) do
    if vim.tbl_contains(pack.filetypes, ft) then
      return pack
    end
  end
  return nil
end

--- List of enabled language names
---@return string[]
function M.enabled()
  return M._enabled
end

--- List of all registered language names
---@return string[]
function M.available()
  return vim.tbl_keys(M._registry)
end

-- ═══════════════════════════════════════════════════
-- Plugin Spec Generators
-- ═══════════════════════════════════════════════════

--- Collect all treesitter parsers from enabled packs
---@return string[]
function M.treesitter_ensure_installed()
  local parsers = {}
  for _, name in ipairs(M._enabled) do
    local pack = M._registry[name]
    if pack and pack.treesitter then
      vim.list_extend(parsers, pack.treesitter)
    end
  end
  return dedup(parsers)
end

--- Collect all LSP servers and their settings
---@return table<string, table> server_name → settings
function M.lsp_servers()
  local servers = {}
  for _, name in ipairs(M._enabled) do
    local pack = M._registry[name]
    if pack and pack.lsp then
      servers[pack.lsp.server] = pack.lsp.settings or {}
    end
  end
  return servers
end

--- Collect all mason packages to install
---@return string[]
function M.mason_ensure_installed()
  local tools = {}
  for _, name in ipairs(M._enabled) do
    local pack = M._registry[name]
    if not pack then goto continue end

    if pack.lsp then
      tools[#tools + 1] = pack.lsp.server
    end
    if pack.dap then
      tools[#tools + 1] = pack.dap.adapter
    end
    if pack.formatter then
      tools[#tools + 1] = pack.formatter.tool
    end
    if pack.linter then
      tools[#tools + 1] = pack.linter.tool
    end

    ::continue::
  end
  return dedup(tools)
end

-- ═══════════════════════════════════════════════════
-- Runtime
-- ═══════════════════════════════════════════════════

--- Apply language-specific keymaps to a buffer
---@param bufnr number
function M.apply_keys(bufnr)
  local pack = M.current()
  if not pack or not pack.keys then return end

  for _, k in ipairs(pack.keys) do
    vim.keymap.set(
      k.mode or "n",
      k.key,
      k.cmd,
      { buffer = bufnr, desc = pack.name .. ": " .. k.desc }
    )
  end
end

--- Show info about a language pack (for :AnvilLang command)
```

```lua
  -- 1. Configuration
  local config = require("anvil.config")
  M.config = config.apply(opts or {})

  -- 2. Core (synchronous, instant)
  require("anvil.core.env").load()
  require("anvil.core.events").setup()
  require("anvil.core.logger").setup(M.config.log)
  require("anvil.core.options").apply(M.config.options)
  require("anvil.core.autocmds").setup()
  require("anvil.core.keys").setup(M.config.keys)

  -- 3. Platform detection
  require("anvil.platform").apply()

  -- 4. Languages (registration only, plugins loaded lazily)
  require("anvil.lang").setup(M.config.languages)

  -- 5. AI (lazy initialization — loads on first :AnvilChat etc.)

  -- 6. Commands
  M._register_commands()

  -- 7. Ready
  M._ready = true
  require("anvil.core.events").emit("anvil:ready", {
    version = M._version,
    languages = #require("anvil.lang").enabled(),
  })
end

function M._register_commands()
  local cmd = vim.api.nvim_create_user_command

  -- Main menu
  cmd("Anvil", function(args)
    require("anvil.ui").open_menu(args)
  end, {
    nargs = "?",
    complete = function()
      return {
        "chat", "ask", "context", "tools", "mcp",
        "provider", "lang", "health", "index",
        "metrics", "version",
      }
    end,
    desc = "Anvil: Main command hub",
  })

  -- AI
  cmd("AnvilChat", function(a)
    require("anvil.ai").chat(a)
  end, { nargs = "?", range = true })

  cmd("AnvilAsk", function(a)
    require("anvil.ai").ask(a.args)
  end, { nargs = "+" })

  cmd("AnvilInline", function(a)
    require("anvil.ai").inline(a)
  end, { nargs = "+", range = true })

  cmd("AnvilContext", function(a)
    require("anvil.ai").show_context(a)
  end, { nargs = "?" })

  cmd("AnvilCompletion", function()
    require("anvil.ai").toggle_completion()
  end, {})

  cmd("AnvilTool", function(a)
    require("anvil.ai").launch_tool(a.args)
  end, { nargs = 1 })

  cmd("AnvilMCP", function(a)
    require("anvil.ai").manage_mcp(a)
  end, { nargs = "?" })

  cmd("AnvilProvider", function(a)
    require("anvil.ai").switch_provider(a.args)
  end, { nargs = "?" })

  cmd("AnvilIndex", function()
    require("anvil.ai").index_project()
  end, {})

  -- Language
  cmd("AnvilLang", function(a)
    require("anvil.lang").info(a.args)
  end, {
    nargs = "?",
    complete = function()
      return require("anvil.lang").available()
    end,
    desc = "Anvil: Language pack info",
  })

  -- Utility
  cmd("AnvilHealth", function()
    vim.cmd("checkhealth anvil")
  end, {})

  cmd("AnvilVersion", function()
    vim.notify("Anvil v" .. M._version)
  end, {})
end

return M
```

### `lua/anvil/config.lua`

```lua
--- Anvil global configuration
--- Merges defaults with user overrides
---@class AnvilConfig
local M = {}

---@class AnvilUserConfig
---@field languages? string[]
---@field options? table
---@field keys? table
---@field log? AnvilLogConfig

---@type AnvilUserConfig
M.defaults = {
  languages = nil, -- nil = all available
  options = {},
  keys = {},
  log = {
    level = "info",
    to_file = false,
  },
}

--- Apply user config over defaults
---@param user_config AnvilUserConfig
---@return AnvilUserConfig
function M.apply(user_config)
  return vim.tbl_deep_extend("force", M.defaults, user_config)
end

return M
```

### `lua/anvil/platform/init.lua`

```lua
--- Anvil platform detection and OS-specific settings
---@class AnvilPlatform
local M = {}

--- Detect current platform
---@return "linux"|"macos"|"wsl"
function M.detect()
  if vim.fn.has("wsl") == 1 then
    return "wsl"
  elseif vim.fn.has("mac") == 1 then
    return "macos"
  else
    return "linux"
  end
end

--- Apply platform-specific settings
function M.apply()
  local platform = M.detect()
  local ok, platform_mod = pcall(require, "anvil.platform." .. platform)
  if ok and platform_mod.apply then
    platform_mod.apply()
  end
end

return M
```
