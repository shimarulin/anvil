# Anvil — Developer Setup

Guide for contributors and anyone modifying the Anvil configuration itself. Covers: tooling, type checking, linting, testing, CI, and the development workflow.

## Prerequisites

Everything from **01.04 Installation**, plus:

| Tool          | Purpose                          | Install                              |
|---------------|----------------------------------|--------------------------------------|
| stylua        | Lua formatter                    | `cargo install stylua` or Mason      |
| selene        | Lua linter                       | `cargo install selene` or Mason      |
| shellcheck    | Shell script linter              | `apt install shellcheck` or `brew install shellcheck` |
| actionlint    | GitHub Actions linter (optional) | `go install github.com/rhysd/actionlint/cmd/actionlint@latest` |
| plenary.nvim  | Test framework                   | Installed as plugin dependency       |
| pre-commit    | Git hook manager (optional)      | `pip install pre-commit` or `brew install pre-commit` |

## Developer Dependencies

### `scripts/dev-setup.sh`

Installs developer tooling on top of the base system dependencies from `preinstall.sh`. Run once after cloning the repository.

<!-- file: scripts/dev-setup.sh -->
```bash
#!/usr/bin/env bash
set -euo pipefail

# ─── Colors ───
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
CYAN='\033[0;36m'
NC='\033[0m'

log_info()  { echo -e "${BLUE}[INFO]${NC} $*"; }
log_ok()    { echo -e "${GREEN}[ OK ]${NC} $*"; }
log_warn()  { echo -e "${YELLOW}[WARN]${NC} $*"; }
log_err()   { echo -e "${RED}[ERR ]${NC} $*"; }

# ─── Detect OS ───
detect_os() {
  if [[ -f /etc/os-release ]]; then
    . /etc/os-release
    echo "$ID"
  elif [[ "$(uname -s)" == "Darwin" ]]; then
    echo "macos"
  else
    echo "unknown"
  fi
}

OS=$(detect_os)

# ─── Check base dependencies ───
check_base() {
  local missing=()
  command -v nvim &>/dev/null || missing+=("nvim")
  command -v git  &>/dev/null || missing+=("git")
  command -v node &>/dev/null || missing+=("node")

  if [[ ${#missing[@]} -gt 0 ]]; then
    log_err "Missing base dependencies: ${missing[*]}"
    log_info "Run ./scripts/preinstall.sh first"
    exit 1
  fi
  log_ok "Base dependencies present"
}

# ─── Install stylua ───
install_stylua() {
  if command -v stylua &>/dev/null; then
    log_ok "stylua $(stylua --version 2>/dev/null | head -1)"
    return 0
  fi

  log_info "Installing stylua..."

  if command -v cargo &>/dev/null; then
    cargo install stylua
    log_ok "stylua installed via cargo"
  elif [[ "$OS" == "macos" ]] && command -v brew &>/dev/null; then
    brew install stylua
    log_ok "stylua installed via brew"
  else
    # Download prebuilt binary
    local arch
    arch="$(uname -m)"
    local platform
    if [[ "$(uname -s)" == "Darwin" ]]; then
      platform="macos-aarch64"
      [[ "$arch" == "x86_64" ]] && platform="macos-x86_64"
    else
      platform="linux-x86_64"
      [[ "$arch" == "aarch64" ]] && platform="linux-aarch64"
    fi

    local tag
    tag="$(curl -fsSL https://api.github.com/repos/JohnnyMorganz/StyLua/releases/latest \
      | grep '"tag_name"' | head -1 | cut -d'"' -f4)"

    local url="https://github.com/JohnnyMorganz/StyLua/releases/download/${tag}/stylua-${platform}.zip"
    curl -fsSL "$url" -o /tmp/stylua.zip
    unzip -oq /tmp/stylua.zip -d /tmp/stylua-bin
    chmod +x /tmp/stylua-bin/stylua
    mkdir -p "$HOME/.local/bin"
    mv /tmp/stylua-bin/stylua "$HOME/.local/bin/stylua"
    rm -rf /tmp/stylua.zip /tmp/stylua-bin
    log_ok "stylua installed to ~/.local/bin/"
  fi
}

# ─── Install selene ───
install_selene() {
  if command -v selene &>/dev/null; then
    log_ok "selene $(selene --version 2>/dev/null | head -1)"
    return 0
  fi

  log_info "Installing selene..."

  if command -v cargo &>/dev/null; then
    cargo install selene
    log_ok "selene installed via cargo"
  elif [[ "$OS" == "macos" ]] && command -v brew &>/dev/null; then
    brew install selene
    log_ok "selene installed via brew"
  else
    # Download prebuilt binary
    local arch
    arch="$(uname -m)"
    local suffix="linux"
    [[ "$arch" == "aarch64" ]] && suffix="linux-aarch64"

    local tag
    tag="$(curl -fsSL https://api.github.com/repos/Kampfkarren/selene/releases/latest \
      | grep '"tag_name"' | head -1 | cut -d'"' -f4)"

    local url="https://github.com/Kampfkarren/selene/releases/download/${tag}/selene-${tag}-${suffix}.zip"
    curl -fsSL "$url" -o /tmp/selene.zip
    unzip -oq /tmp/selene.zip -d /tmp/selene-bin
    chmod +x /tmp/selene-bin/selene
    mkdir -p "$HOME/.local/bin"
    mv /tmp/selene-bin/selene "$HOME/.local/bin/selene"
    rm -rf /tmp/selene.zip /tmp/selene-bin
    log_ok "selene installed to ~/.local/bin/"
  fi
}

# ─── Install shellcheck ───
install_shellcheck() {
  if command -v shellcheck &>/dev/null; then
    log_ok "shellcheck $(shellcheck --version 2>/dev/null | grep '^version:' || echo 'installed')"
    return 0
  fi

  log_info "Installing shellcheck..."

  case "$OS" in
    ubuntu|debian|pop|linuxmint)
      sudo apt-get install -y shellcheck
      ;;
    fedora|rhel|centos)
      sudo dnf install -y ShellCheck
      ;;
    arch|manjaro|endeavouros)
      sudo pacman -S --noconfirm shellcheck
      ;;
    macos)
      brew install shellcheck
      ;;
    *)
      log_warn "shellcheck: install manually from https://github.com/koalaman/shellcheck"
      return 0
      ;;
  esac
  log_ok "shellcheck installed"
}

# ─── Install pre-commit ───
install_precommit() {
  if command -v pre-commit &>/dev/null; then
    log_ok "pre-commit $(pre-commit --version 2>/dev/null)"
    return 0
  fi

  log_info "Installing pre-commit..."

  if [[ "$OS" == "macos" ]] && command -v brew &>/dev/null; then
    brew install pre-commit
  elif command -v pip3 &>/dev/null; then
    pip3 install --user pre-commit 2>/dev/null || pip3 install pre-commit
  elif command -v pipx &>/dev/null; then
    pipx install pre-commit
  else
    log_warn "pre-commit: install manually — pip install pre-commit"
    return 0
  fi
  log_ok "pre-commit installed"
}

# ─── Install actionlint (optional) ───
install_actionlint() {
  if command -v actionlint &>/dev/null; then
    log_ok "actionlint $(actionlint --version 2>/dev/null | head -1)"
    return 0
  fi

  if [[ "$OS" == "macos" ]] && command -v brew &>/dev/null; then
    log_info "Installing actionlint..."
    brew install actionlint
    log_ok "actionlint installed"
  elif command -v go &>/dev/null; then
    log_info "Installing actionlint..."
    go install github.com/rhysd/actionlint/cmd/actionlint@latest
    log_ok "actionlint installed via go"
  else
    log_warn "actionlint: skipped (optional — install go or brew to enable)"
  fi
}

# ─── Setup pre-commit hooks ───
setup_hooks() {
  local project_root
  project_root="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"

  if [[ ! -f "$project_root/.pre-commit-config.yaml" ]]; then
    log_warn "No .pre-commit-config.yaml found — skipping hook setup"
    return 0
  fi

  if ! command -v pre-commit &>/dev/null; then
    log_warn "pre-commit not installed — skipping hook setup"
    return 0
  fi

  log_info "Installing git hooks..."
  cd "$project_root"
  pre-commit install
  pre-commit install --hook-type pre-push
  log_ok "Git hooks installed (pre-commit + pre-push)"
}

# ─── Install plenary for tests ───
ensure_plenary() {
  local plenary_dir="${XDG_DATA_HOME:-$HOME/.local/share}/anvil/lazy/plenary.nvim"

  if [[ -d "$plenary_dir" ]]; then
    log_ok "plenary.nvim present"
    return 0
  fi

  log_info "Installing plenary.nvim for tests..."
  mkdir -p "$(dirname "$plenary_dir")"
  git clone --depth 1 https://github.com/nvim-lua/plenary.nvim "$plenary_dir"
  log_ok "plenary.nvim installed"
}

# ─── Verify ───
verify() {
  echo ""
  log_info "Verifying developer tools..."
  echo ""

  local tools=(
    "stylua:required:Lua formatter"
    "selene:required:Lua linter"
    "shellcheck:required:Shell linter"
    "pre-commit:recommended:Git hook manager"
    "actionlint:optional:GitHub Actions linter"
  )

  local all_required=true
  for entry in "${tools[@]}"; do
    IFS=':' read -r cmd level desc <<< "$entry"
    if command -v "$cmd" &>/dev/null; then
      local ver
      ver=$("$cmd" --version 2>/dev/null | head -1 || echo "installed")
      log_ok "$cmd — $ver ($desc)"
    else
      if [[ "$level" == "required" ]]; then
        log_err "$cmd — NOT FOUND ($desc) [required]"
        all_required=false
      elif [[ "$level" == "recommended" ]]; then
        log_warn "$cmd — not found ($desc) [recommended]"
      else
        log_warn "$cmd — not found ($desc) [optional]"
      fi
    fi
  done

  # Check plenary
  local plenary_dir="${XDG_DATA_HOME:-$HOME/.local/share}/anvil/lazy/plenary.nvim"
  if [[ -d "$plenary_dir" ]]; then
    log_ok "plenary.nvim — $plenary_dir"
  else
    log_warn "plenary.nvim — not found (run make bootstrap)"
  fi

  # Check git hooks
  local project_root
  project_root="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
  if [[ -f "$project_root/.git/hooks/pre-commit" ]]; then
    log_ok "Git pre-commit hook installed"
  else
    log_warn "Git pre-commit hook not installed (run: pre-commit install)"
  fi

  echo ""
  if $all_required; then
    log_ok "Developer environment ready"
    echo ""
    echo -e "  Quick check:  ${CYAN}make check${NC}"
    echo -e "  Run tests:    ${CYAN}make test-all${NC}"
    echo -e "  Launch Anvil: ${CYAN}make dev${NC}"
  else
    log_err "Some required tools are missing"
    exit 1
  fi
}

# ─── Main ───
echo -e "${CYAN}"
echo "  ╔═══════════════════════════════════════╗"
echo "  ║     ⚒  ANVIL — Developer Setup  ⚒     ║"
echo "  ╚═══════════════════════════════════════╝"
echo -e "${NC}"

check_base
echo ""

install_stylua
install_selene
install_shellcheck
install_precommit
install_actionlint
echo ""

ensure_plenary
setup_hooks

verify
```

### What `dev-setup.sh` Does

- Verifies base dependencies from `preinstall.sh` are present
- Installs **stylua** (Lua formatter) — via cargo, brew, or prebuilt binary
- Installs **selene** (Lua linter) — via cargo, brew, or prebuilt binary
- Installs **shellcheck** (shell linter) — via system package manager
- Installs **pre-commit** (git hook manager) — via brew, pip, or pipx
- Installs **actionlint** (GitHub Actions linter, optional) — via brew or go
- Ensures **plenary.nvim** is available for tests
- Installs **git hooks** (pre-commit + pre-push)
- Verifies all tools and reports status

### Quick Start for Contributors

```bash
# 1. System dependencies (if not already done)
./scripts/preinstall.sh

# 2. Install Anvil
./scripts/install.sh

# 3. Developer tools
./scripts/dev-setup.sh

# 4. Verify everything
make check
make test-all
```

## Tooling Configuration

### `.luarc.json`

`lua_ls` workspace configuration. Provides `vim.*` API types, `vim.uv` (luv), and busted globals for tests.

### ``.luarc.json``

<!-- file: .luarc.json -->
```json
{
  "$schema": "https://raw.githubusercontent.com/LuaLS/vscode-lua/master/setting/schema.json",
  "runtime": {
    "version": "LuaJIT",
    "path": ["?.lua", "?/init.lua"],
    "pathStrict": true
  },
  "workspace": {
    "library": [
      "${3rd}/luv/library",
      "${3rd}/busted/library"
    ],
    "checkThirdParty": false,
    "ignoreDir": [
      "refs",
      ".git",
      "node_modules"
    ]
  },
  "diagnostics": {
    "globals": ["vim", "describe", "it", "before_each", "after_each", "assert"],
    "disable": ["missing-fields", "incomplete-signature-doc"]
  },
  "completion": {
    "callSnippet": "Replace",
    "keywordSnippet": "Replace"
  },
  "type": {
    "castNumberToInteger": true
  },
  "hint": {
    "enable": true,
    "setType": true,
    "paramName": "All",
    "paramType": true,
    "arrayIndex": "Disable"
  },
  "format": {
    "enable": false
  }
}
```

Key decisions:

- `format.enable = false` — stylua handles formatting, not lua_ls
- `checkThirdParty = false` — lazydev.nvim handles plugin types at runtime
- `refs` in `ignoreDir` — reference sources are too large for lua_ls to index
- `busted` library — enables type info for `describe`, `it`, `assert` in tests

Optional: framework types. If you need go-to-definition into AstroNvim or NvChad source code, add to `workspace.library`:

```json
"library": [
  "${3rd}/luv/library",
  "${3rd}/busted/library",
  "refs/astronvim/core/lua"
]
```

This slows down lua_ls. Only add what you actively need.

## Makefile

### `Makefile`

<!-- file: Makefile -->
```makefile
.PHONY: help install dev-setup health health-ai sync-refs sync-refs-nvchad \
        sync-refs-astro clean-refs lint lint-lua lint-sh lint-actions \
        fmt fmt-check check test test-unit test-smoke test-all \
        bootstrap update clean dev

SHELL   := /bin/bash
APPNAME ?= anvil

help: ## Show this help
	@grep -E '^[a-zA-Z_-]+:.*?## .*$$' $(MAKEFILE_LIST) | \
	  awk 'BEGIN {FS = ":.*?## "}; {printf "  \033[36m%-20s\033[0m %s\n", $$1, $$2}'

# ─── Setup ───

install: ## Install Anvil (verify location + bootstrap)
	@./scripts/install.sh

dev-setup: ## Install developer tools (stylua, selene, shellcheck, hooks)
	@./scripts/dev-setup.sh

bootstrap: ## Bootstrap Neovim plugins (headless)
	@NVIM_APPNAME=$(APPNAME) nvim --headless +"Lazy! sync" +qall 2>&1

# ─── References ───

sync-refs: ## Download/update framework reference sources
	@./scripts/sync-refs.sh

sync-refs-nvchad: ## Update NvChad references only
	@./scripts/sync-refs.sh nvchad

sync-refs-astro: ## Update AstroNvim references only
	@./scripts/sync-refs.sh astronvim

clean-refs: ## Remove all reference sources
	@rm -rf refs/
	@echo "refs/ removed"

# ─── Lint ───

lint: lint-lua lint-sh ## Run all linters

lint-lua: ## Run selene on Lua files
	@selene lua/

lint-sh: ## Run shellcheck on shell scripts
	@shellcheck scripts/*.sh tests/smoke/*.sh

lint-actions: ## Run actionlint on GitHub Actions workflows
	@actionlint

# ─── Format ───

fmt: ## Run stylua formatter
	@stylua lua/

fmt-check: ## Check formatting without modifying files
	@stylua --check lua/

# ─── Check ───

check: lint fmt-check ## Run all checks (lint + format)

# ─── Test ───

test: test-unit ## Run unit tests (default)

test-unit: ## Run unit tests (plenary.busted)
	@NVIM_APPNAME=$(APPNAME) nvim --headless \
		-u tests/minimal_init.lua \
		-c "PlenaryBustedDirectory tests/unit" 2>&1

test-smoke: ## Run smoke tests (headless Neovim boot)
	@bash tests/smoke/smoke_spec.sh

test-all: lint fmt-check test-unit test-smoke ## Run everything

# ─── Health ───

health: ## Run Anvil health check
	@NVIM_APPNAME=$(APPNAME) nvim --headless \
		+"checkhealth anvil" +qall 2>&1

health-ai: ## Run AI health check
	@NVIM_APPNAME=$(APPNAME) nvim --headless \
		+"checkhealth anvil_ai" +qall 2>&1

# ─── Dev ───

dev: ## Launch Anvil for development
	@NVIM_APPNAME=$(APPNAME) nvim

# ─── Maintenance ───

update: ## Update plugins
	@NVIM_APPNAME=$(APPNAME) nvim --headless +"Lazy! update" +qall 2>&1

clean: ## Clean Neovim state/cache for anvil
	@rm -rf "$$HOME/.local/share/$(APPNAME)"
	@rm -rf "$$HOME/.local/state/$(APPNAME)"
	@rm -rf "$$HOME/.cache/$(APPNAME)"
	@echo "$(APPNAME) data cleaned (includes Python venv)"
```

### `tests/smoke/smoke_spec.sh`

Headless smoke tests — verifies Anvil boots without errors.

<!-- file: tests/smoke/smoke_spec.sh -->
```bash
#!/usr/bin/env bash
set -euo pipefail

# ─── Colors ───
GREEN='\033[0;32m'
RED='\033[0;31m'
CYAN='\033[0;36m'
NC='\033[0m'

APPNAME="${ANVIL_APPNAME:-anvil}"
PASS=0
FAIL=0

smoke() {
  local name="$1"; shift
  local output
  if output=$(NVIM_APPNAME="$APPNAME" nvim --headless "$@" 2>&1); then
    echo -e "  ${GREEN}✅${NC} $name"
    ((PASS++))
  else
    echo -e "  ${RED}❌${NC} $name"
    echo "     $output" | head -5
    ((FAIL++))
  fi
}

echo -e "${CYAN}═══ Anvil Smoke Tests ═══${NC}"
echo ""

# ─── Boot ───
smoke "Anvil boots without errors" \
  -u tests/smoke_init.lua \
  -c "lua assert(require('anvil')._ready, 'Boot not completed')" \
  -c "qall!"

# ─── Core modules load ───
smoke "All core modules load" \
  -u tests/smoke_init.lua \
  -c "lua local mods = {'anvil.core.python','anvil.core.env','anvil.core.cache','anvil.core.logger','anvil.core.events','anvil.core.options','anvil.core.keys','anvil.core.icons','anvil.config'}; for _,m in ipairs(mods) do assert(pcall(require, m), m..' failed to load') end" \
  -c "qall!"

# ─── Python provider ───
smoke "Python provider is configured" \
  -u tests/smoke_init.lua \
  -c "lua assert(vim.g.python3_host_prog ~= nil, 'python3_host_prog not set')" \
  -c "qall!"

# ─── Health check ───
smoke "checkhealth anvil runs" \
  -u tests/smoke_init.lua \
  -c "checkhealth anvil" \
  -c "qall!"

# ─── Config module ───
smoke "Config module returns table" \
  -u tests/smoke_init.lua \
  -c "lua local c = require('anvil.config').get(); assert(type(c) == 'table', 'config is not a table')" \
  -c "qall!"

# ─── Results ───
echo ""
echo -e "${CYAN}─── Results ───${NC}"
echo -e "  Passed: ${GREEN}${PASS}${NC}"
echo -e "  Failed: ${RED}${FAIL}${NC}"
echo ""

if [[ $FAIL -gt 0 ]]; then
  echo -e "${RED}FAIL${NC}"
  exit 1
else
  echo -e "${GREEN}ALL PASSED${NC}"
fi
```

### Writing Unit Tests

Tests use plenary.busted (Lua port of busted):

```lua
describe("anvil.core.cache", function()
  local cache = require("anvil.core.cache")

  before_each(function()
    cache.clear()
  end)

  it("stores and retrieves values", function()
    cache.set("key", "value")
    assert.equals("value", cache.get("key"))
  end)

  it("returns nil for missing keys", function()
    assert.is_nil(cache.get("nonexistent"))
  end)
end)
```

Every public function should have at least one test. New core modules must include tests in `tests/unit/core_spec.lua` or a dedicated spec file.

## Type Annotations Convention

All Anvil modules use LuaLS annotations:

```lua
---@class AnvilCoreCache          -- class definition
---@field _store table            -- field

---@param key string              -- parameter
---@param ttl? number             -- optional parameter
---@return any? value             -- return value with name

---@alias AnvilLogLevel "trace"|"debug"|"info"|"warn"|"error"  -- union type
```

Every public function must have `---@param` and `---@return` annotations.
Every module must have a `---@class` annotation at the top.

## Reference Sources

Framework source code and documentation, available locally for browsing and AI context. Downloaded into `refs/` (gitignored).

### `scripts/sync-refs.sh`

<!-- file: scripts/sync-refs.sh -->
``````bash
#!/usr/bin/env bash
set -euo pipefail

# ─── Colors ───
GREEN='\033[0;32m'
BLUE='\033[0;34m'
YELLOW='\033[1;33m'
NC='\033[0m'

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
PROJECT_ROOT="$(dirname "$SCRIPT_DIR")"
REFS_DIR="$PROJECT_ROOT/refs"

log_info() { echo -e "${BLUE}[INFO]${NC} $*"; }
log_ok()   { echo -e "${GREEN}[ OK ]${NC} $*"; }
log_warn() { echo -e "${YELLOW}[WARN]${NC} $*"; }

# ─── Repository definitions ───
# Format: "target_dir|repo_url|branch"
REPOS=(
  "nvchad/core|https://github.com/NvChad/NvChad.git|v2.5"
  "nvchad/docs|https://github.com/NvChad/nvchad.github.io.git|main"
  "nvchad/community|https://github.com/NvChad/nvcommunity.git|main"
  "astronvim/core|https://github.com/AstroNvim/AstroNvim.git|v4"
  "astronvim/docs|https://github.com/AstroNvim/docs.git|main"
  "astronvim/community|https://github.com/AstroNvim/astrocommunity.git|main"
)

# ─── Functions ───
sync_repo() {
  local target="$1"
  local url="$2"
  local branch="$3"
  local full_path="$REFS_DIR/$target"

  if [[ -d "$full_path/.git" ]]; then
    log_info "Updating $target (branch: $branch)..."
    git -C "$full_path" fetch origin "$branch" --depth=1 2>/dev/null
    git -C "$full_path" reset --hard "origin/$branch" 2>/dev/null
    log_ok "$target updated"
  else
    log_info "Cloning $target (branch: $branch)..."
    mkdir -p "$(dirname "$full_path")"
    git clone --depth=1 --branch="$branch" --single-branch "$url" "$full_path" 2>/dev/null
    log_ok "$target cloned"
  fi
}

generate_readme() {
  cat > "$REFS_DIR/README.md" << 'HEREDOC'
# Reference Sources

This directory contains cloned reference repositories for NvChad and AstroNvim.
It is not tracked by git (listed in .gitignore).

## Purpose

- **Developer reference** — browse framework source code without leaving the project
- **AI agent context** — AI tools can read these files for accurate framework knowledge
- **lua_ls workspace** — optional: add paths to .luarc.json for type resolution

## Contents

| Directory              | Repository                 | Branch |
|------------------------|----------------------------|--------|
| `nvchad/core/`         | NvChad/NvChad              | v2.5   |
| `nvchad/docs/`         | NvChad/nvchad.github.io    | main   |
| `nvchad/community/`    | NvChad/nvcommunity         | main   |
| `astronvim/core/`      | AstroNvim/AstroNvim        | v4     |
| `astronvim/docs/`      | AstroNvim/docs             | main   |
| `astronvim/community/` | AstroNvim/astrocommunity   | main   |

## Update

```bash
make sync-refs          # all
make sync-refs-nvchad   # NvChad only
make sync-refs-astro    # AstroNvim only
./scripts/sync-refs.sh nvchad/core  # single repo
```
HEREDOC
  log_ok "README.md generated"
}

# ─── Main ───
main() {
  echo -e "${BLUE}═══ Anvil Reference Sync ═══${NC}"
  echo ""

  mkdir -p "$REFS_DIR"

  local filter="${1:-}"

  for entry in "${REPOS[@]}"; do
    IFS='|' read -r target url branch <<< "$entry"

    if [[ -n "$filter" && "$target" != *"$filter"* ]]; then
      continue
    fi

    sync_repo "$target" "$url" "$branch"
  done

  generate_readme

  echo ""
  echo -e "${GREEN}═══ Sync complete ═══${NC}"
  echo ""

  if command -v du &>/dev/null; then
    echo "Disk usage:"
    du -sh "$REFS_DIR"/*/ 2>/dev/null | sed 's|./refs/|  refs/|'
    echo ""
    du -sh "$REFS_DIR" | sed 's|.*/refs|  refs/ (total)|'
  fi
}

main "$@"
``````

## Development Workflow

### Daily Workflow

```text
1. Create a feature branch
2. Make changes in lua/ or scripts/
3. make fmt              ← format Lua
4. make lint             ← selene + shellcheck
5. make test-unit        ← fast unit tests
6. make test-smoke       ← headless boot verification
7. git commit            ← pre-commit runs lint automatically
8. git push              ← pre-push runs unit tests
9. CI runs test-all      ← lint + unit + smoke on stable + nightly
10. PR → review → merge
```

### Adding a New Core Module

1. Create `lua/anvil/core/newmodule.lua`
2. Add `---@class` annotation
3. Add to architecture diagram in `02.01_Anvil_Core_Layer.md`
4. Add to health check in `lua/health/anvil.lua` (core modules list)
5. Add tests in `tests/unit/` (new file or extend `core_spec.lua`)
6. Add to smoke test module list in `tests/smoke/smoke_spec.sh`
7. Run `make test-all`

### Adding a New Language Pack

1. Copy `lua/anvil/lang/_template.lua` → `lua/anvil/lang/elixir.lua`
2. Fill in all fields (LSP, formatter, linter, DAP, test, tasks, AI)
3. Restart Neovim — auto-detected
4. Run `:Anvil lang elixir` to verify
5. Add to enabled languages in smoke test if it should be tested

### Adding a New UI Strategy

1. Create `lua/anvil/ui/strategies/newframework.lua`
2. Implement `M.specs()` returning `LazyPluginSpec[]`
3. Add detection logic in `lua/anvil/ui/init.lua`
4. Test with `ANVIL_UI_STRATEGY=newframework`

### Modifying a Plugin Spec

1. Find the spec in `lua/plugins/*.lua`
2. Modify `opts`/`config`/`keys`
3. Run `:Lazy sync` to apply
4. Run `make lint` to verify

## Troubleshooting

### lua_ls is slow

```json
// .luarc.json — increase ignoreDir
"ignoreDir": ["refs", ".git", "node_modules", "tests/fixtures"]
```

Also check: `:LspInfo` → lua_ls should show only `lua/` as workspace root, not the entire project.

### selene reports false positives

```lua
-- Suppress for a line:
local _ = something  -- selene: allow(unused_variable)

-- Or add to selene.toml:
[lints]
unused_variable = "allow"  -- if too noisy
```

### stylua conflicts with existing formatting

Check `.editorconfig` is not overriding `stylua.toml`. stylua respects `.editorconfig` by default. If both exist, stylua wins for Lua files.

### Tests fail with "module not found"

```bash
# Ensure plenary is installed
NVIM_APPNAME=anvil nvim --headless +"Lazy! sync" +qall

# Check test init
cat tests/minimal_init.lua
# Should have: vim.opt.rtp:append(vim.fn.stdpath("data") .. "/lazy/plenary.nvim")
```

### Smoke tests fail in CI

Common causes:
- Python venv not created (check the "Setup Python venv" step)
- Neovim version mismatch (check the matrix)
- Plugin not installed (smoke tests use `smoke_init.lua` which boots fully)

### References out of date

```bash
# Update all
make sync-refs

# Check current versions
cd refs/nvchad/core && git log --oneline -1
cd refs/astronvim/core && git log --oneline -1
```

### Pre-commit hooks not running

```bash
# Verify hooks are installed
pre-commit install
pre-commit install --hook-type pre-push

# Test manually
pre-commit run --all-files

# Skip hooks temporarily (emergency)
git commit --no-verify
```
