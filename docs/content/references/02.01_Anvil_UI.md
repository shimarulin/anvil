# Anvil — UI Layer

Visual presentation: theme, statusline, dashboard, notifications,
tabs, breadcrumbs, and all plugins that shape how Neovim looks.

## Architecture

```
lua/anvil/ui/
├── init.lua           # UI subsystem setup
├── theme.lua          # colorscheme configuration
├── statusline.lua     # heirline / statusline customization
├── dashboard.lua      # start screen
├── notifications.lua  # notification system
└── which_key.lua      # key hint groups

lua/plugins/
└── ui.lua             # all UI plugin specs
```

### Design Principles

1. **One theme, fully integrated** — Catppuccin Mocha with explicit
   integration for every plugin
2. **Rounded borders everywhere** — consistent `border = "rounded"`
3. **Icons from core/icons** — single source of truth
4. **No transparency by default** — uncomment if desired
5. **Minimal chrome** — `cmdheight=0`, `showmode=false`, noice handles the rest

```lua
      dashboard.section.buttons.opts.hl = "DashboardCenter"
      dashboard.section.footer.opts.hl = "DashboardFooter"

      -- Footer: show stats after load
      vim.api.nvim_create_autocmd("User", {
        pattern = "LazyVimStarted",
        once = true,
        callback = function()
          local stats = require("lazy").stats()
          local ms = (math.floor(stats.startuptime * 100 + 0.5) / 100)
          local anvil = require("anvil")
          dashboard.section.footer.val = string.format(
            "  Anvil v%s  │  %d plugins  │  %dms",
            anvil._version or "?",
            stats.loaded,
            ms
          )
          pcall(vim.cmd.AlphaRedraw)
        end,
      })

      return opts
    end,
  }
end

return M
```

```lua
--- Anvil which-key group definitions
--- Uses groups from core/keys for consistency
---@class AnvilUIWhichKey
local M = {}

---@return LazyPluginSpec
function M.spec()
  local key_groups = require("anvil.core.keys").groups

  -- Convert to which-key spec format
  local spec = {}
  for key, name in pairs(key_groups) do
    spec[#spec + 1] = { key, group = name }
  end

  -- Add sub-groups
  local sub_groups = {
    { "<leader>aA", group = "󰚩 AI Agents" },
    { "<leader>aG", group = "󰊢 AI Git" },
    { "<leader>Ac", group = "󰚩 AI Context →" },
    { "<leader>fg", group = "󰊢 Git (telescope)" },
    { "<leader>h",  group = " Harpoon" },
  }
  vim.list_extend(spec, sub_groups)

  return {
    "folke/which-key.nvim",
    opts = {
      plugins = {
        marks = true,
        registers = true,
        spelling = { enabled = true, suggestions = 20 },
      },
      spec = spec,
    },
  }
end

return M
```

```lua
    vim.g.neovide_padding_right = 5
    vim.g.neovide_padding_left = 5
  end

  -- ─── LSP hover/signature borders ───
  vim.lsp.handlers["textDocument/hover"] = vim.lsp.with(
    vim.lsp.handlers.hover,
    { border = "rounded", max_width = 80, max_height = 30 }
  )
  vim.lsp.handlers["textDocument/signatureHelp"] = vim.lsp.with(
    vim.lsp.handlers.signature_help,
    { border = "rounded" }
  )

  -- ─── Diagnostic float borders ───
  vim.diagnostic.config({
    float = { border = "rounded", source = true },
  })
end

return M
```

## Statusline Integration

### `lua/anvil/ui/statusline.lua`

Anvil adds custom components to AstroNvim's heirline statusline:

```lua
--- Anvil statusline customization
--- Adds AI status and language pack info to heirline
---@class AnvilUIStatusline
local M = {}

--- AI status component for heirline
---@return table heirline component
function M.ai_component()
  return {
    condition = function()
      return vim.g.anvil_ai_active or vim.g.anvil_minuet_enabled
    end,
    provider = function()
      local parts = {}

      if vim.g.anvil_minuet_enabled then
        parts[#parts + 1] = "󰚩"
      end

      local err_count = vim.g.anvil_ai_error_count or 0
      if err_count > 0 then
        parts[#parts + 1] = "🔴" .. err_count
      end

      if #parts == 0 then return "" end
      return " " .. table.concat(parts, " ") .. " "
    end,
    hl = { fg = "blue", bold = true },
  }
end

--- Language pack component for heirline
---@return table heirline component
function M.lang_component()
  return {
    condition = function()
      return vim.bo.filetype ~= ""
    end,
    provider = function()
      local ok, lang = pcall(require, "anvil.lang")
      if not ok then return "" end
      local pack = lang.current()
      if not pack then return "" end
      return " " .. pack.name .. " "
    end,
    hl = { fg = "green" },
  }
end

return M
```

> **Note:** These components can be inserted into AstroNvim's heirline
> configuration via the `astroui` statusline customization API. The exact
> integration depends on the AstroNvim version and heirline setup.
