# Anvil — Installation & Setup

User-facing guide: from zero to a working Anvil environment.

---

## Requirements

| Dependency   | Minimum | Recommended   | Why                                          |
|-------------|---------|---------------|----------------------------------------------|
| Neovim      | 0.9     | 0.10+         | `NVIM_APPNAME` isolation, `vim.uv`, treesitter |
| Git         | 2.19    | latest        | Partial clone, sparse checkout               |
| Node.js     | 18      | 20+           | Mason LSP servers, MCP servers               |
| npm         | —       | latest        | Comes with Node.js                           |
| ripgrep     | —       | latest        | Telescope live grep                          |
| fd          | —       | latest        | Telescope file finder                        |
| Nerd Font   | —       | any           | Icons in statusline, explorer, completion    |
| Terminal    | —       | kitty/alacritty/wezterm | True color, ligatures              |
| Clipboard   | —       | xclip/wl-clipboard/pbcopy | System clipboard (`"+y`, `"+p`)   |
| Python 3    | 3.8     | 3.11+         | Python provider for Neovim (some plugins)    |
| Build tools | —       | gcc, make, cmake | Tree-sitter parser compilation            |

> **Nerd Font note:** Set your terminal font to **JetBrainsMono Nerd Font**, _not_ JetBrainsMono Nerd Font **Mono**. The Mono variant constrains icon glyphs to a single character cell, making them noticeably smaller.

## System Dependencies

### `scripts/preinstall.sh`

Installs system-level dependencies. Run once on a fresh machine.

<!-- file: scripts/preinstall.sh -->
```bash
#!/usr/bin/env bash
set -euo pipefail

# ─── Colors ───
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m'

log_info() { echo -e "${BLUE}[INFO]${NC} $*"; }
log_ok()   { echo -e "${GREEN}[ OK ]${NC} $*"; }
log_warn() { echo -e "${YELLOW}[WARN]${NC} $*"; }
log_err()  { echo -e "${RED}[ERR ]${NC} $*"; }

# ─── Detect OS ───
detect_os() {
  if [[ -f /etc/os-release ]]; then
    . /etc/os-release
    echo "$ID"
  elif [[ "$(uname -s)" == "Darwin" ]]; then
    echo "macos"
  else
    echo "unknown"
  fi
}

OS=$(detect_os)
log_info "Detected OS: $OS"

# ─── Package managers ───
install_apt() {
  log_info "Installing via apt..."
  sudo apt-get update -qq

  if ! command -v nvim &>/dev/null || [[ "$(nvim --version | head -1 | grep -oP '\d+\.\d+')" < "0.10" ]]; then
    log_info "Adding Neovim PPA..."
    sudo add-apt-repository -y ppa:neovim-ppa/stable 2>/dev/null || true
    sudo apt-get update -qq
  fi

  sudo apt-get install -y \
    neovim git curl wget unzip ripgrep fd-find \
    nodejs npm python3 python3-pip python3-venv \
    build-essential cmake

  if command -v fdfind &>/dev/null && ! command -v fd &>/dev/null; then
    sudo ln -sf "$(which fdfind)" /usr/local/bin/fd
    log_ok "Symlinked fdfind → fd"
  fi

  # Clipboard tool
  if ! command -v xclip &>/dev/null && ! command -v xsel &>/dev/null && ! command -v wl-copy &>/dev/null; then
    if [[ -n "${WAYLAND_DISPLAY:-}" ]]; then
      sudo apt-get install -y wl-clipboard 2>/dev/null || log_warn "wl-clipboard not available"
    else
      sudo apt-get install -y xclip 2>/dev/null || log_warn "xclip not available"
    fi
  fi

  sudo apt-get install -y lazygit 2>/dev/null || log_warn "lazygit not in repos — install manually"
  sudo apt-get install -y pandoc 2>/dev/null  || log_warn "pandoc not available"
}

install_brew() {
  log_info "Installing via Homebrew..."
  if ! command -v brew &>/dev/null; then
    log_err "Homebrew not found. Install from https://brew.sh"
    exit 1
  fi

  # Remove dead taps that cause warnings
  local dead_taps
  dead_taps="$(brew tap 2>/dev/null | grep -E 'cask-fonts|cask-drivers' || true)"
  for tap in $dead_taps; do
    brew untap "$tap" 2>/dev/null || true
  done

  brew install neovim git ripgrep fd node lazygit pandoc cmake
}

install_dnf() {
  log_info "Installing via dnf..."
  sudo dnf install -y \
    neovim git ripgrep fd-find nodejs npm \
    python3 python3-pip gcc gcc-c++ cmake
  sudo dnf install -y lazygit 2>/dev/null || log_warn "lazygit not in repos"
}

install_pacman() {
  log_info "Installing via pacman..."
  sudo pacman -Syu --noconfirm \
    neovim git ripgrep fd nodejs npm python python-pip \
    base-devel cmake lazygit pandoc
}

# ─── Neovim version fallback ───
# If the installed version is too old, download from GitHub releases.
ensure_nvim_version() {
  local required_minor=9
  if command -v nvim &>/dev/null; then
    local ver
    ver=$(nvim --version | head -1 | grep -oP '\d+\.\d+' || true)
    local major minor
    major=$(echo "$ver" | cut -d. -f1)
    minor=$(echo "$ver" | cut -d. -f2)
    if [[ $major -ge 1 ]] || [[ $major -eq 0 && $minor -ge $required_minor ]]; then
      log_ok "Neovim v${ver} meets minimum requirement (v0.${required_minor}+)"
      return 0
    fi
    log_warn "Neovim v${ver} is too old — downloading latest from GitHub..."
  else
    log_info "Neovim not found — downloading from GitHub..."
  fi

  local tag
  tag="$(curl -fsSL https://api.github.com/repos/neovim/neovim/releases/latest \
    | grep '"tag_name"' | head -1 | cut -d'"' -f4)"

  if [[ "$(uname -s)" == "Darwin" ]]; then
    curl -fsSL "https://github.com/neovim/neovim/releases/download/${tag}/nvim-macos-arm64.tar.gz" -o /tmp/nvim.tar.gz
    sudo rm -rf /opt/nvim
    sudo tar -xzf /tmp/nvim.tar.gz -C /opt/
    sudo ln -sf /opt/nvim-macos-arm64/bin/nvim /usr/local/bin/nvim
  else
    curl -fsSL "https://github.com/neovim/neovim/releases/download/${tag}/nvim-linux64.tar.gz" -o /tmp/nvim.tar.gz
    sudo rm -rf /opt/nvim
    sudo tar -xzf /tmp/nvim.tar.gz -C /opt/
    sudo mv /opt/nvim-linux64 /opt/nvim
    sudo ln -sf /opt/nvim/bin/nvim /usr/local/bin/nvim
  fi

  rm -f /tmp/nvim.tar.gz
  log_ok "Installed Neovim $(nvim --version | head -1)"
}

# ─── Python provider (PEP 668 safe) ───
setup_python_provider() {
  local venv_dir="$HOME/.local/share/nvim-deps-venv"

  if [[ -f "$venv_dir/bin/python3" ]] && "$venv_dir/bin/python3" -c "import pynvim" 2>/dev/null; then
    log_ok "Python provider already configured ($venv_dir)"
    return 0
  fi

  log_info "Creating Python venv for Neovim provider..."
  python3 -m venv "$venv_dir"
  "$venv_dir/bin/pip" install --upgrade pip pynvim

  # Global plugin so ALL Neovim configs find pynvim automatically
  local plugin_dir="$HOME/.local/share/nvim/site/plugin"
  mkdir -p "$plugin_dir"
  cat > "$plugin_dir/python3_host.lua" <<EOF
-- Auto-generated by preinstall.sh
-- Points Neovim Python 3 provider to the venv with pynvim.
-- Applies to ALL Neovim configs (any NVIM_APPNAME).
-- Any config can override by setting vim.g.python3_host_prog itself.
-- To undo: delete this file.
if not vim.g.python3_host_prog then
  vim.g.python3_host_prog = "$venv_dir/bin/python3"
end
EOF

  log_ok "Python provider: $venv_dir/bin/python3"
  log_ok "Global plugin: $plugin_dir/python3_host.lua"
}

# ─── Nerd Font ───
install_nerd_font() {
  local font_name="JetBrainsMono"
  local font_dir

  if [[ "$(uname -s)" == "Darwin" ]]; then
    font_dir="$HOME/Library/Fonts"
  else
    font_dir="$HOME/.local/share/fonts"
  fi

  if ls "$font_dir"/*${font_name}* &>/dev/null 2>&1; then
    log_ok "Nerd Font already installed ($font_name)"
    return 0
  fi

  log_info "Installing $font_name Nerd Font..."
  local tag
  tag="$(curl -fsSL https://api.github.com/repos/ryanoasis/nerd-fonts/releases/latest \
    | grep '"tag_name"' | head -1 | cut -d'"' -f4)"

  mkdir -p "$font_dir"
  curl -fsSL "https://github.com/ryanoasis/nerd-fonts/releases/download/${tag}/${font_name}.zip" -o /tmp/nf.zip
  unzip -oq /tmp/nf.zip -d "$font_dir"
  rm -f /tmp/nf.zip

  if [[ "$(uname -s)" != "Darwin" ]]; then
    fc-cache -f "$font_dir" 2>/dev/null || true
  fi

  log_ok "Installed $font_name Nerd Font to $font_dir"
  log_warn "Set your terminal font to: JetBrainsMono Nerd Font (NOT the Mono variant)"
}

# ─── PATH check ───
ensure_local_bin_in_path() {
  local target="$HOME/.local/bin"
  mkdir -p "$target"

  if echo "$PATH" | tr ':' '\n' | grep -qx "$target"; then
    return 0
  fi

  log_warn "\$HOME/.local/bin is NOT in your \$PATH"
  log_info "Some installed tools (tree-sitter, Python packages) live there."

  local shell_config=""
  local export_line="export PATH=\"\$HOME/.local/bin:\$PATH\""

  if [[ -n "${ZSH_VERSION:-}" ]] || [[ "$SHELL" == */zsh ]]; then
    # zsh hardcodes ~/.zshenv as the first file — NOT $ZDOTDIR/.zshenv
    shell_config="$HOME/.zshenv"
  elif [[ "$SHELL" == */bash ]]; then
    if [[ "$(uname -s)" == "Darwin" ]]; then
      shell_config="$HOME/.bash_profile"
    else
      shell_config="$HOME/.bashrc"
    fi
  elif [[ "$SHELL" == */fish ]]; then
    shell_config="$HOME/.config/fish/config.fish"
    export_line="fish_add_path \"$target\""
  fi

  if [[ -n "$shell_config" ]]; then
    if [[ -f "$shell_config" ]] && grep -qF "$target" "$shell_config"; then
      return 0
    fi
    read -rp "Add $export_line to $shell_config? [y/N] " answer
    if [[ "$answer" =~ ^[Yy]$ ]]; then
      echo "" >> "$shell_config"
      echo "# Added by Anvil install script" >> "$shell_config"
      echo "$export_line" >> "$shell_config"
      export PATH="$target:$PATH"
      log_ok "PATH updated in $shell_config"
    fi
  fi
}

# ─── Main ───
echo -e "${BLUE}"
echo "  ╔═══════════════════════════════════════╗"
echo "  ║       ⚒  ANVIL — System Setup  ⚒      ║"
echo "  ╚═══════════════════════════════════════╝"
echo -e "${NC}"

case "$OS" in
  ubuntu|debian|pop|linuxmint)  install_apt ;;
  macos)                        install_brew ;;
  fedora|rhel|centos)           install_dnf ;;
  arch|manjaro|endeavouros)     install_pacman ;;
  *)
    log_err "Unsupported OS: $OS"
    log_info "Install manually: neovim git ripgrep fd nodejs npm"
    exit 1 ;;
esac

ensure_nvim_version
install_nerd_font
setup_python_provider
ensure_local_bin_in_path

# ─── Verify ───
echo ""
log_info "Verifying installation..."

REQUIRED=("nvim" "git" "node" "npm")
RECOMMENDED=("rg" "fd" "lazygit" "pandoc" "python3")

all_ok=true
for cmd in "${REQUIRED[@]}"; do
  if command -v "$cmd" &>/dev/null; then
    version=$("$cmd" --version 2>/dev/null | head -1 || echo "installed")
    log_ok "$cmd — $version"
  else
    log_err "$cmd — NOT FOUND (required)"
    all_ok=false
  fi
done

for cmd in "${RECOMMENDED[@]}"; do
  if command -v "$cmd" &>/dev/null; then
    version=$("$cmd" --version 2>/dev/null | head -1 || echo "installed")
    log_ok "$cmd — $version"
  else
    log_warn "$cmd — not found (recommended)"
  fi
done

# Clipboard check
if command -v pbcopy &>/dev/null; then
  log_ok "Clipboard: pbcopy (macOS)"
elif command -v wl-copy &>/dev/null; then
  log_ok "Clipboard: wl-clipboard (Wayland)"
elif command -v xclip &>/dev/null; then
  log_ok "Clipboard: xclip (X11)"
elif command -v xsel &>/dev/null; then
  log_ok "Clipboard: xsel (X11)"
else
  log_warn "No clipboard tool found — \"+y and \"+p won't work in Neovim"
fi

echo ""
if $all_ok; then
  log_ok "All required dependencies installed"
  echo ""
  echo -e "  Next: ${GREEN}./install.sh${NC}"
else
  log_err "Some required dependencies are missing"
  exit 1
fi
```

## Anvil Installation

### `scripts/install.sh`

Main installation script. Creates symlinks, bootstraps plugins.

<!-- file: scripts/install.sh -->
```bash
#!/usr/bin/env bash
set -euo pipefail

# ─── Colors ───
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
CYAN='\033[0;36m'
NC='\033[0m'

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
PROJECT_ROOT="$(dirname "$SCRIPT_DIR")"

# ─── Platform paths ───
NVIM_CONFIG_HOME="${XDG_CONFIG_HOME:-$HOME/.config}"
NVIM_DATA_HOME="${XDG_DATA_HOME:-$HOME/.local/share}"
NVIM_STATE_HOME="${XDG_STATE_HOME:-$HOME/.local/state}"
NVIM_CACHE_HOME="${XDG_CACHE_HOME:-$HOME/.cache}"

APPNAME="${ANVIL_APPNAME:-anvil}"
CONFIG_TARGET="$NVIM_CONFIG_HOME/$APPNAME"

log_info() { echo -e "${BLUE}[INFO]${NC} $*"; }
log_ok()   { echo -e "${GREEN}[ OK ]${NC} $*"; }
log_warn() { echo -e "${YELLOW}[WARN]${NC} $*"; }
log_err()  { echo -e "${RED}[ERR ]${NC} $*"; }

print_header() {
  echo -e "${CYAN}"
  echo "  ╔═══════════════════════════════════════╗"
  echo "  ║        ⚒  ANVIL Installer  ⚒          ║"
  echo "  ╚═══════════════════════════════════════╝"
  echo -e "${NC}"
}

# ─── Dependency check ───
check_deps() {
  local missing=()
  command -v git  &>/dev/null || missing+=("git")
  command -v nvim &>/dev/null || missing+=("nvim")
  command -v node &>/dev/null || missing+=("node")
  command -v npm  &>/dev/null || missing+=("npm")

  if [[ ${#missing[@]} -gt 0 ]]; then
    log_err "Missing dependencies: ${missing[*]}"
    log_info "Run: ./scripts/preinstall.sh"
    exit 1
  fi

  local nvim_ver
  nvim_ver=$(nvim --version | head -1 | grep -oP 'v\K[0-9]+\.[0-9]+' || true)
  if [[ -n "$nvim_ver" ]]; then
    local major minor
    major=$(echo "$nvim_ver" | cut -d. -f1)
    minor=$(echo "$nvim_ver" | cut -d. -f2)
    if [[ $major -eq 0 && $minor -lt 9 ]]; then
      log_err "Neovim v0.9+ required (found v$nvim_ver)"
      exit 1
    fi
    if [[ $major -eq 0 && $minor -lt 10 ]]; then
      log_warn "Neovim v$nvim_ver — v0.10+ recommended for full features"
    fi
  fi

  log_ok "Dependencies OK (nvim v${nvim_ver:-unknown})"
}

# ─── Symlink ───
create_symlink() {
  local source="$1"
  local target="$2"

  if [[ -L "$target" ]]; then
    local current
    current=$(readlink -f "$target" 2>/dev/null || readlink "$target")
    local resolved
    resolved=$(readlink -f "$source" 2>/dev/null || echo "$source")

    if [[ "$current" == "$resolved" ]]; then
      log_ok "Symlink OK: $target"
      return 0
    else
      log_warn "Updating symlink: $target"
      rm "$target"
    fi
  elif [[ -e "$target" ]]; then
    local backup="${target}.bak.$(date +%Y%m%d%H%M%S)"
    log_warn "Backing up existing: $target → $backup"
    mv "$target" "$backup"
  fi

  mkdir -p "$(dirname "$target")"
  ln -s "$source" "$target"
  log_ok "Symlink: $target → $source"
}

# ─── Bootstrap plugins ───
bootstrap() {
  log_info "Bootstrapping Neovim plugins (this may take a minute)..."
  if NVIM_APPNAME="$APPNAME" nvim --headless +"Lazy! sync" +qall 2>&1; then
    log_ok "Plugins installed"
  else
    log_warn "Plugin bootstrap had warnings (this is usually fine on first run)"
  fi
}

# ─── Shell aliases ───
print_shell_setup() {
  echo ""
  echo -e "${CYAN}─── Shell Setup (add to ~/.bashrc or ~/.zshrc) ───${NC}"
  echo ""
  echo "  # Anvil"
  echo "  alias anvil='NVIM_APPNAME=$APPNAME nvim'"
  echo ""
  echo "  # Or make it the default editor:"
  echo "  # export EDITOR='NVIM_APPNAME=$APPNAME nvim'"
  echo ""
}

# ─── Uninstall ───
cmd_uninstall() {
  echo -e "${YELLOW}Uninstalling Anvil...${NC}"

  if [[ -L "$CONFIG_TARGET" ]]; then
    rm "$CONFIG_TARGET"
    log_ok "Removed symlink: $CONFIG_TARGET"
  elif [[ -e "$CONFIG_TARGET" ]]; then
    log_warn "$CONFIG_TARGET is not a symlink — refusing to remove"
    log_info "Remove manually: rm -rf $CONFIG_TARGET"
    return 1
  fi

  read -rp "Clean data/state/cache for '$APPNAME'? [y/N] " answer
  if [[ "$answer" =~ ^[Yy]$ ]]; then
    rm -rf "$NVIM_DATA_HOME/$APPNAME"
    rm -rf "$NVIM_STATE_HOME/$APPNAME"
    rm -rf "$NVIM_CACHE_HOME/$APPNAME"
    log_ok "Data cleaned"
  fi

  log_ok "Anvil uninstalled"
}

# ─── Status ───
cmd_status() {
  echo -e "${CYAN}Anvil Status${NC}"
  echo ""

  if [[ -L "$CONFIG_TARGET" ]]; then
    echo -e "  Config: ${GREEN}$CONFIG_TARGET → $(readlink "$CONFIG_TARGET")${NC}"
  elif [[ -d "$CONFIG_TARGET" ]]; then
    echo -e "  Config: ${YELLOW}$CONFIG_TARGET (exists, not symlink)${NC}"
  else
    echo -e "  Config: ${RED}not installed${NC}"
  fi

  echo -e "  Data:   $NVIM_DATA_HOME/$APPNAME"
  echo -e "  State:  $NVIM_STATE_HOME/$APPNAME"
  echo -e "  Cache:  $NVIM_CACHE_HOME/$APPNAME"
  echo ""

  local ai_config="$PROJECT_ROOT/config/ai-providers.json"
  if [[ -f "$ai_config" ]]; then
    echo -e "  AI config: ${GREEN}$ai_config${NC}"
  else
    echo -e "  AI config: ${YELLOW}not yet created (will use defaults)${NC}"
  fi

  # Python provider status
  local venv_dir="$HOME/.local/share/nvim-deps-venv"
  if [[ -f "$venv_dir/bin/python3" ]] && "$venv_dir/bin/python3" -c "import pynvim" 2>/dev/null; then
    echo -e "  Python:  ${GREEN}$venv_dir/bin/python3 (pynvim OK)${NC}"
  else
    echo -e "  Python:  ${YELLOW}pynvim not configured (run preinstall.sh)${NC}"
  fi
}

# ─── Usage ───
usage() {
  echo "Usage: $0 [command]"
  echo ""
  echo "Commands:"
  echo "  install     Install Anvil (default)"
  echo "  uninstall   Remove Anvil config and optionally data"
  echo "  status      Show installation status"
  echo "  help        Show this help"
  echo ""
  echo "Environment:"
  echo "  ANVIL_APPNAME   Neovim app name (default: anvil)"
}

# ─── Main ───
main() {
  print_header

  local cmd="${1:-install}"

  case "$cmd" in
    install)
      check_deps
      echo ""

      log_info "Installing Anvil as NVIM_APPNAME='$APPNAME'"
      echo ""

      # 1. Symlink project root → Neovim config
      create_symlink "$PROJECT_ROOT" "$CONFIG_TARGET"

      # 2. Bootstrap plugins
      echo ""
      read -rp "Bootstrap plugins now? [Y/n] " answer
      if [[ ! "$answer" =~ ^[Nn]$ ]]; then
        bootstrap
      else
        log_info "Skipped. Run later: make bootstrap"
      fi

      # 3. Print next steps
      echo ""
      echo -e "${GREEN}═══ Installation complete ═══${NC}"
      print_shell_setup
      echo -e "${CYAN}─── Next Steps ───${NC}"
      echo ""
      echo "  1. Launch: NVIM_APPNAME=$APPNAME nvim"
      echo "  2. Configure AI: :AnvilProviderEdit (edit config/ai-providers.json)"
      echo "  3. Or pick a preset: :AnvilPreset"
      echo "  4. Verify: :checkhealth anvil"
      echo ""
      ;;
    uninstall)
      cmd_uninstall ;;
    status)
      cmd_status ;;
    help|--help|-h)
      usage ;;
    *)
      log_err "Unknown command: $cmd"
      usage
      exit 1 ;;
  esac
}

main "$@"
```

## Step-by-Step Installation

### Quick Start (3 commands)

```bash
# 1. Clone
git clone https://github.com/your-username/anvil ~/.local/src/anvil
cd ~/.local/src/anvil

# 2. Install system deps (if needed)
./scripts/preinstall.sh

# 3. Install Anvil
./scripts/install.sh
```

### What `install.sh` Does

1. Checks dependencies (nvim, git, node, npm)
2. Creates symlink: `~/.config/anvil` → `~/.local/src/anvil`
3. Bootstraps plugins (Lazy.nvim sync)

### What `preinstall.sh` Does

1. Detects OS and package manager
2. Installs system packages (neovim, git, ripgrep, fd, node, build tools)
3. Ensures Neovim ≥ 0.9 (falls back to GitHub release if repo version is too old)
4. Installs JetBrainsMono Nerd Font
5. Creates a Python venv with `pynvim` (PEP 668 safe)
6. Installs a global Neovim plugin for the Python provider
7. Installs clipboard tool (xclip / wl-clipboard on Linux)
8. Ensures `~/.local/bin` is in `$PATH`
9. Installs optional tools (lazygit, pandoc)

- `:AnvilPreset` — select an AI preset
- `:AnvilChat` — test AI (requires configured provider with API key)

## Architecture & Design Decisions

### Neovim Version Guarantee

Many distribution repositories ship older Neovim versions (especially `apt` on Ubuntu LTS). The `preinstall.sh` script handles this automatically:

```text
Install via package manager
        │
        ▼
  Check version ≥ 0.9?
  ┌─────┴─────┐
  │ YES       │ NO
  │ Done      │ Download latest release
  │           │ from GitHub (prebuilt tarball)
  │           │ Install to /opt/nvim
  │           │ Symlink to /usr/local/bin/nvim
  └───────────┘
```

The GitHub release tag is fetched dynamically via the API so the script always gets the latest stable version.

### Python Provider & PEP 668

PEP 668 marks system Python installations as "externally managed", preventing `pip install` from modifying them. This is enforced on Arch Linux (always), macOS Homebrew Python (always), Ubuntu ≥ 23.04, Fedora ≥ 38, and others.

**Solution:** The script creates a dedicated Python virtual environment:

```text
~/.local/share/nvim-deps-venv/
├── bin/
│   ├── python3        ← isolated Python
│   ├── pip            ← isolated pip
│   └── ...
└── lib/
    └── python3.x/
        └── site-packages/
            └── pynvim/
```

A **global Neovim plugin** is installed at `~/.local/share/nvim/site/plugin/python3_host.lua` so that _all_ Neovim configurations (any `NVIM_APPNAME`) find `pynvim` automatically:

```lua
-- Auto-generated by preinstall.sh
if not vim.g.python3_host_prog then
  vim.g.python3_host_prog = "/home/user/.local/share/nvim-deps-venv/bin/python3"
end
```

The `if not` guard ensures any config that explicitly sets `python3_host_prog` takes priority.

### Clipboard Integration

Neovim requires a clipboard tool for system clipboard integration (`"+y` and `"+p`). Without one, these operations silently fail.

| Platform       | Tool                     | Notes                        |
|---------------|--------------------------|------------------------------|
| macOS         | `pbcopy` / `pbpaste`     | Built-in, nothing to install |
| Linux (X11)   | `xclip` or `xsel`        | Script installs `xclip`      |
| Linux (Wayland) | `wl-clipboard`         | Script detects `$WAYLAND_DISPLAY` |

### PATH & Shell Configuration

Several tools are installed into `~/.local/bin/`. The script checks whether this directory is in `$PATH` and offers to add it.

**Why `~/.zshenv` for zsh (not `~/.zshrc` or `$ZDOTDIR/.zshenv`)?**

zsh **hardcodes** `~/.zshenv` as the first file it reads. The `$ZDOTDIR` variable only takes effect _after_ `~/.zshenv` has been sourced. A file at `$ZDOTDIR/.zshenv` is **never loaded automatically** by zsh.

| File                  | Read by zsh?  | When                          |
|-----------------------|---------------|-------------------------------|
| `~/.zshenv`           | ✅ Always     | First, before anything else   |
| `$ZDOTDIR/.zshenv`    | ❌ Never      | zsh does NOT re-read `.zshenv` from `$ZDOTDIR` |
| `$ZDOTDIR/.zshrc`     | ✅ Interactive | After `~/.zshenv` sets `$ZDOTDIR` |
| `~/.zshrc`            | Only if `$ZDOTDIR` is unset | Falls back to `$HOME` |

Using `~/.zshenv` ensures PATH is available in interactive shells, non-interactive shells, Neovim's `:!command`, and subprocesses.

| Shell          | Config file          | Export syntax                              |
|---------------|----------------------|--------------------------------------------|
| zsh           | `~/.zshenv`          | `export PATH="$HOME/.local/bin:$PATH"`     |
| bash (Linux)  | `~/.bashrc`          | `export PATH="$HOME/.local/bin:$PATH"`     |
| bash (macOS)  | `~/.bash_profile`    | `export PATH="$HOME/.local/bin:$PATH"`     |
| fish          | `~/.config/fish/config.fish` | `fish_add_path "$HOME/.local/bin"` |

## NVIM_APPNAME Isolation

Anvil uses Neovim's `NVIM_APPNAME` (v0.9+) for complete isolation:

```text
NVIM_APPNAME=anvil
  config:  ~/.config/anvil/
  data:    ~/.local/share/anvil/
  state:   ~/.local/state/anvil/
  cache:   ~/.cache/anvil/

nvim (default, untouched)
  config:  ~/.config/nvim/
  data:    ~/.local/share/nvim/
```

Your existing Neovim configuration is never touched.

> **Note:** The global Python provider plugin at `~/.local/share/nvim/site/plugin/` is shared across all `NVIM_APPNAME` values. This is intentional — it provides a working Python provider as a default for every configuration.

## Troubleshooting

### Plugins fail to install

```bash
# Clean and retry
make clean
make bootstrap
```

### LSP servers not found

```bash
# Open Anvil and run Mason
NVIM_APPNAME=anvil nvim
:MasonToolsInstall
```

### AI features not working

```bash
# Check provider config
:AnvilProviderConfig

# Edit config file
:AnvilProviderEdit

# Check health
:checkhealth anvil_ai
```

### Treesitter parsers failing to compile

```bash
# Ensure build tools are installed
# Ubuntu/Debian:
sudo apt-get install build-essential

# macOS:
xcode-select --install
```

### Wrong theme / UI looks broken

```bash
# Verify terminal supports true color
echo -e "\033[38;2;255;100;0mTRUE COLOR TEST\033[0m"
# Should show orange text

# Check Nerd Font is installed
# In Neovim:
:echo &guifont
```

### Icons look small in Neovim

You are using the **Mono** variant of the Nerd Font. Change your terminal font to **JetBrainsMono Nerd Font** (without "Mono" at the end).

### `error: externally-managed-environment` when running `pip install`

This is expected. The system Python is protected by PEP 668. The install script handles this by using a dedicated virtual environment. Do not run `pip install` against the system Python — use the venv instead:

```bash
~/.local/share/nvim-deps-venv/bin/pip install <package>
```

### `homebrew/homebrew-cask-fonts does not exist`

The `preinstall.sh` script auto-removes this dead tap. If you see this error outside the script:

```bash
brew untap homebrew/cask-fonts
```

### Neovim version is too old after install

On `apt`-based systems the repository version is often outdated. The `preinstall.sh` script falls back to the GitHub release automatically. If you installed Neovim through another method, remove it first and re-run the script.

### `tree-sitter` or other tools not found after install

Ensure `~/.local/bin` is in your `$PATH`. The script offers to add it automatically during installation. If you declined, add manually:

```bash
# zsh — add to ~/.zshenv (NOT .zshrc, NOT $ZDOTDIR/.zshenv)
export PATH="$HOME/.local/bin:$PATH"

# bash
export PATH="$HOME/.local/bin:$PATH"

# fish
fish_add_path "$HOME/.local/bin"
```

Then restart your terminal or run `source ~/.zshenv`.

### `:checkhealth provider` still shows Python warnings

1. Verify the global plugin exists and points to the correct path:

   ```bash
   cat ~/.local/share/nvim/site/plugin/python3_host.lua
   ```

2. Verify pynvim is importable:

   ```bash
   ~/.local/share/nvim-deps-venv/bin/python3 -c "import pynvim; print('OK')"
   ```

3. Make sure no other config is setting `python3_host_prog` to a different (broken) path:

   ```bash
   grep -r "python3_host_prog" ~/.config/nvim/
   ```

### Clipboard not working in Neovim

- Check `:checkhealth provider` for the Clipboard section
- On Linux, ensure you have the right tool for your display server:
  - **X11:** `xclip` or `xsel`
  - **Wayland:** `wl-clipboard` (`wl-copy` / `wl-paste`)
- In a headless/SSH session without a display server, clipboard integration requires additional setup (e.g. OSC 52 support in your terminal)

## Uninstalling

### Anvil Config

```bash
./scripts/install.sh uninstall
```

This removes the symlink and optionally cleans data/state/cache. The cloned repository is not deleted — remove it manually if desired.

### Full Cleanup Reference

| Component              | How to remove                                                |
|------------------------|--------------------------------------------------------------|
| Anvil config symlink   | `./scripts/install.sh uninstall`                             |
| Anvil data/state/cache | Included in uninstall (when confirmed)                       |
| Global Python plugin   | `rm ~/.local/share/nvim/site/plugin/python3_host.lua`        |
| Python venv            | `rm -rf ~/.local/share/nvim-deps-venv`                       |
| PATH line in shell config | Remove the `# Added by Anvil install script` block       |
| Nerd Font (macOS)      | Delete `*JetBrainsMono*` files from `~/Library/Fonts/`       |
| Nerd Font (Linux)      | Delete `*JetBrainsMono*` files from `~/.local/share/fonts/`  |
| Neovim (GitHub install)| `sudo rm -rf /opt/nvim /usr/local/bin/nvim`                  |
| Neovim (package manager)| `brew uninstall neovim`, `sudo apt remove neovim`, etc.     |
| lazygit (GitHub install)| `sudo rm /usr/local/bin/lazygit`                            |
| Cloned repository      | `rm -rf ~/.local/src/anvil`                                  |
