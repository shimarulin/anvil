# Anvil — UI Layer

Theme, statusline, bufferline, file explorer, dashboard, notifications,
and zen mode.

## Architecture

```
lua/anvil/ui/
├── init.lua           # UI subsystem setup
├── theme.lua          # colorscheme configuration
├── statusline.lua     # lualine configuration
├── bufferline.lua     # bufferline configuration
├── explorer.lua       # file explorer (neo-tree)
├── notifications.lua  # notification system
├── dashboard.lua      # start screen
└── zen.lua            # zen / focus mode

lua/plugins/
└── ui.lua             # UI plugin specs
```

## Statusline

### `lua/anvil/ui/statusline.lua`

```lua
--- Anvil statusline configuration (lualine)
---@class AnvilUIStatusline
local M = {}

local icons = require("anvil.core.icons")

--- Return lualine opts
---@return table
function M.opts()
  return {
    options = {
      theme = "auto",
      globalstatus = true,
      disabled_filetypes = {
        statusline = { "dashboard", "alpha", "starter" },
      },
      component_separators = { left = "", right = "" },
      section_separators   = { left = "", right = "" },
    },
    sections = {
      lualine_a = { "mode" },
      lualine_b = {
        { "branch", icon = icons.git.branch },
      },
      lualine_c = {
        {
          "diagnostics",
          symbols = {
            error = icons.diagnostics.error,
            warn  = icons.diagnostics.warn,
            info  = icons.diagnostics.info,
            hint  = icons.diagnostics.hint,
          },
        },
        { "filetype", icon_only = true, separator = "", padding = { left = 1, right = 0 } },
        {
          "filename",
          path = 1,
          symbols = {
            modified = icons.file.modified,
            readonly = icons.file.readonly,
            unnamed  = icons.file.unnamed,
          },
        },
        -- Navic breadcrumbs
        {
          function()
            local navic_ok, navic = pcall(require, "nvim-navic")
            if navic_ok and navic.is_available() then
              return navic.get_location()
            end
            return ""
          end,
          cond = function()
            local navic_ok, navic = pcall(require, "nvim-navic")
            return navic_ok and navic.is_available()
          end,
        },
      },
      lualine_x = {
        -- AI completion status
        {
          function()
            local comp_ok, comp = pcall(require, "anvil.ai.completion")
            if comp_ok and comp._enabled then
              return icons.misc.ai .. "ON"
            end
            return ""
          end,
          color = { fg = "#89b4fa" },
        },
        -- Overseer task status
        {
          "overseer",
          label = "",
          colored = true,
        },
        {
          "diff",
          symbols = {
            added    = icons.git.added,
            modified = icons.git.modified,
            removed  = icons.git.removed,
          },
        },
      },
      lualine_y = {
        { "progress", separator = " ", padding = { left = 1, right = 0 } },
        { "location", padding = { left = 0, right = 1 } },
      },
      lualine_z = {
        function() return os.date("%H:%M") end,
      },
    },
    extensions = { "neo-tree", "lazy", "toggleterm", "trouble", "overseer", "quickfix" },
  }
end

return M
```

## Dashboard

### `lua/anvil/ui/dashboard.lua`

```lua
--- Anvil dashboard configuration
---@class AnvilUIDashboard
local M = {}

--- Return dashboard opts (for alpha.nvim or similar)
---@return table
function M.opts()
  local dashboard = require("alpha.themes.dashboard")

  dashboard.section.header.val = {
    [[    ╔═══════════════════════════════╗    ]],
    [[    ║     ⚒  ANVIL  ⚒              ║    ]],
    [[    ║   AI-Powered Forge            ║    ]],
    [[    ║   for Neovim                  ║    ]],
    [[    ╚═══════════════════════════════╝    ]],
  }

  dashboard.section.buttons.val = {
    dashboard.button("f", "  Find file",       "<cmd>Telescope find_files<cr>"),
    dashboard.button("r", "  Recent files",    "<cmd>Telescope oldfiles<cr>"),
    dashboard.button("g", "  Grep text",       "<cmd>Telescope live_grep<cr>"),
    dashboard.button("p", "  Projects",        "<cmd>Telescope projects<cr>"),
    dashboard.button("s", "  Restore session", [[<cmd>lua require("persistence").load()<cr>]]),
    dashboard.button("a", "󰚩  AI Chat",         "<cmd>CodeCompanionChat<cr>"),
    dashboard.button("h", "  Health check",    "<cmd>checkhealth anvil<cr>"),
    dashboard.button("l", "󰒲  Lazy",            "<cmd>Lazy<cr>"),
    dashboard.button("q", "  Quit",            "<cmd>qa<cr>"),
  }

  dashboard.section.footer.val = function()
    local stats = require("lazy").stats()
    local ms = (math.floor(stats.startuptime * 100 + 0.5) / 100)
    return "⚡ " .. stats.loaded .. "/" .. stats.count .. " plugins in " .. ms .. "ms"
  end

  dashboard.section.header.opts.hl = "AlphaHeader"
  dashboard.section.buttons.opts.hl = "AlphaButtons"
  dashboard.section.footer.opts.hl = "AlphaFooter"

  return dashboard.config
end

return M
```

## Notifications

### `lua/anvil/ui/notifications.lua`

```lua
--- Anvil notification configuration
---@class AnvilUINotifications
local M = {}

--- Return noice.nvim opts (or nvim-notify opts)
---@return table
function M.notify_opts()
  return {
    background_colour = "#000000",
    fps = 30,
    level = vim.log.levels.INFO,
    minimum_width = 30,
    render = "default",
    stages = "fade_in_slide_out",
    timeout = 3000,
    top_down = true,
  }
end

--- Return noice.nvim opts
---@return table
function M.noice_opts()
  return {
    lsp = {
      override = {
        ["vim.lsp.util.convert_input_to_markdown_lines"] = true,
        ["vim.lsp.util.stylize_markdown"] = true,
        ["cmp.entry.get_documentation"] = true,
      },
    },
    routes = {
      -- Hide "written" messages
      {
        filter = { event = "msg_show", kind = "", find = "written" },
        opts = { skip = true },
      },
      -- Hide search count
      {
        filter = { event = "msg_show", kind = "search_count" },
        opts = { skip = true },
      },
    },
    presets = {
      bottom_search = true,
      command_palette = true,
      long_message_to_split = true,
      lsp_doc_border = true,
    },
  }
end

return M
```
